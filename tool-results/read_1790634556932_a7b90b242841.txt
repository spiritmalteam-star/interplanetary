   419→   419→/* ---------------- the forge chat — the direct mirror line ---------------- */
   420→   420→
   421→   421→function ForgeBody({ text }: { text: string }) {
   422→   422→  const paras = useMemo(
   423→   423→    () =>
   424→   424→      text
   425→   425→        .split(/\n{2,}/)
   426→   426→        .map((p) => p.trim())
   427→   427→        .filter(Boolean),
   428→   428→    [text]
   429→   429→  );
   430→   430→  return (
   431→   431→    <div className="space-y-2.5">
   432→   432→      {paras.map((p, i) => {
   433→   433→        const isLast = i === paras.length - 1;
   434→   434→        if (isLast && p.startsWith("—")) {
   435→   435→          return (
   436→   436→            <p
   437→   437→              key={i}
   438→   438→              className="mono-label pt-1 text-[11px] leading-relaxed"
   439→   439→              style={{ color: "color-mix(in srgb, var(--scope-b) 82%, white)" }}
   440→   440→            >
   441→   441→              {p}
   442→   442→            </p>
   443→   443→          );
   444→   444→        }
   445→   445→        return (
   446→   446→          <p
   447→   447→            key={i}
   448→   448→            className={cn(
   449→   449→              "text-[14.5px] leading-[1.75] text-foreground/88",
   450→   450→              i === 0 && "font-medium text-foreground/95"
   451→   451→            )}
   452→   452→          >
   453→   453→            {p}
   454→   454→          </p>
   455→   455→        );
   456→   456→      })}
   457→   457→    </div>
   458→   458→  );
   459→   459→}
