   309→   309→
   310→   310→function GroupDossierBody({
   311→   311→  kind,
   312→   312→  id,
   313→   313→}: {
   314→   314→  kind: DossierKind;
   315→   315→  id: string;
   316→   316→}) {
   317→   317→  const t = useT();
   318→   318→  const entry = useMemo(() => findDossier(kind, id), [kind, id]);
   319→   319→  const openModal = useMirror((s) => s.openModal);
   320→   320→  const extras = useMemo(
   321→   321→    () => (entry ? groupProfiles[entry.id] : undefined),
   322→   322→    [entry]
   323→   323→  );
   324→   324→
   325→   325→  const [visible, setVisible] = useState(BATCH);
   326→   326→  const [filter, setFilter] = useState("");
   327→   327→  const [prevGroupKey, setPrevGroupKey] = useState(`${kind}|${id}`);
   328→   328→  const listRef = useRef<HTMLUListElement>(null);
   329→   329→  const sentinelRef = useRef<HTMLDivElement>(null);
   330→   330→
   331→   331→  // Reset reveal state when switching groups — adjusted during render.
   332→   332→  const groupKey = `${kind}|${id}`;
   333→   333→  if (groupKey !== prevGroupKey) {
   334→   334→    setPrevGroupKey(groupKey);
   335→   335→    setVisible(BATCH);
   336→   336→    setFilter("");
   337→   337→  }
   338→   338→
   339→   339→  const filteredReps = useMemo(() => {
   340→   340→    if (!entry) return [];
   341→   341→    const q = filter.trim().toLowerCase();
   342→   342→    if (!q) return entry.representatives;
   343→   343→    return entry.representatives.filter(
   344→   344→      (r) =>
   345→   345→        r.name.toLowerCase().includes(q) ||
   346→   346→        r.origin.toLowerCase().includes(q) ||
   347→   347→        r.specialty.toLowerCase().includes(q)
   348→   348→    );
   349→   349→  }, [entry, filter]);
   350→   350→
   351→   351→  // Auto-reveal as the sentinel scrolls into view
   352→   352→  useEffect(() => {
   353→   353→    const el = sentinelRef.current;
   354→   354→    if (!el) return;
   355→   355→    const io = new IntersectionObserver(
   356→   356→      (entries) => {
   357→   357→        if (entries[0]?.isIntersecting) {
   358→   358→          setVisible((v) =>
   359→   359→            v < filteredReps.length
   360→   360→              ? Math.min(v + BATCH, filteredReps.length)
   361→   361→              : v
   362→   362→          );
   363→   363→        }
   364→   364→      },
   365→   365→      { rootMargin: "300px" }
   366→   366→    );
   367→   367→    io.observe(el);
   368→   368→    return () => io.disconnect();
   369→   369→  }, [filteredReps.length]);
   370→   370→
   371→   371→  if (!entry) return null;
   372→   372→
   373→   373→  const kindLabel =
   374→   374→    kind === "civilization" ? t("Civilization dossier") : t("Interdimensional dossier");
   375→   375→  const extras_ = extras;
   376→   376→
   377→   377→  return (
   378→   378→    <>
   379→   379→      {/* the three context images — banner impression + two context scenes */}
   380→   380→      <div className="px-5 pt-1 sm:px-6">
   381→   381→        <ProfileGallery
   382→   382→          slots={[
   383→   383→            [groupImage(kind, entry.id)],
   384→   384→            [
   385→   385→              sceneImageFor(entry.id, "milieu"),
   386→   386→              groupImage(kind, entry.id),
   387→   387→            ],
   388→   388→            [
   389→   389→              sceneImageForDistinct(entry.id, "homelands", [
   390→   390→                sceneImageFor(entry.id, "milieu"),
   391→   391→                groupImage(kind, entry.id),
   392→   392→              ]),
   393→   393→              sceneImageFor(entry.id, "milieu"),
   394→   394→            ],
   395→   395→          ]}
   396→   396→          alts={[
   397→   397→            t("AI-rendered scene impression of the {name}", { name: entry.name }),
   398→   398→            t("The milieu of {name} — an impressionistic scene", { name: entry.name }),
   399→   399→            t("The homelands of {name} — an impressionistic scene", { name: entry.name }),
   400→   400→          ]}
   401→   401→          testid="group-gallery"
   402→   402→        />
   403→   403→      </div>
   404→   404→
   405→   405→      <div className="flex flex-wrap items-center gap-2 px-5 pt-4 sm:px-6">
   406→   406→        <Badge tone="cy">{kindLabel}</Badge>
   407→   407→        <Badge tone="ok">{t("{n} named representatives", { n: entry.count })}</Badge>
   408→   408→        <Badge tone="pk">{entry.range}</Badge>
   409→   409→      </div>
   410→   410→
   411→   411→      <div className="nice-scroll max-h-[min(60vh,540px)] space-y-5 overflow-y-auto px-5 pb-4 pt-4 sm:px-6">
   412→   412→        <ProfileSection index="01" label={t("Essence")}>
   413→   413→          <p className="text-[14.5px] font-medium leading-relaxed text-foreground/90">{entry.essence}</p>
   414→   414→        </ProfileSection>
   415→   415→        <ProfileSection index="02" label={t("Role in human awakening")}>
   416→   416→          <p className="text-[14.5px] leading-relaxed text-foreground/85">{entry.role}</p>
   417→   417→        </ProfileSection>
   418→   418→        <ProfileSection index="03" label={t("Signs of resonance")}>
   419→   419→          <p className="text-[14.5px] leading-relaxed text-foreground/85">{entry.signal}</p>
   420→   420→        </ProfileSection>
   421→   421→
   422→   422→        {/* deep sections */}
   423→   423→        {extras_ && (
   424→   424→          <>
   425→   425→            <div className="h-px w-full bg-gradient-to-r from-[color-mix(in_srgb,var(--cy)_35%,transparent)] to-transparent" />
   426→   426→            <ProfileSection index="04" label={t("Recorded history")}>
   427→   427→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.history}</p>
   428→   428→            </ProfileSection>
   429→   429→            <ProfileSection index="05" label={t("How their society is organized")}>
   430→   430→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.structure}</p>
   431→   431→            </ProfileSection>
   432→   432→            <ProfileSection index="06" label={t("Ships, temples & artifacts")}>
   433→   433→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.artifacts}</p>
   434→   434→            </ProfileSection>
   435→   435→            <ProfileSection index="07" label={t("Contact protocol")}>
   436→   436→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.contactProtocol}</p>

   620→   620→
   621→   621→export function DossierModal() {
   622→   622→  const modal = useMirror((s) => s.modal);
   623→   623→  const closeModal = useMirror((s) => s.closeModal);
   624→   624→  const t = useT();
   625→   625→
   626→   626→  const isEntity = modal?.type === "entity";
   627→   627→  const isGroup = modal?.type === "dossier";
   628→   628→  const kind = modal && (isEntity || isGroup) ? modal.kind : null;
   629→   629→  const id = modal && (isEntity || isGroup) ? modal.id : null;
   630→   630→
   631→   631→  const groupEntry =
   632→   632→    kind && !isEntity ? findDossier(kind, id as string) : null;
   633→   633→  const open = (isGroup && !!groupEntry) || (isEntity && !!kind && !!id);
   634→   634→
   635→   635→  let title = t("Dossier");
   636→   636→  let description: string | undefined;
   637→   637→  if (isGroup && groupEntry) {
   638→   638→    title = groupEntry.name;
   639→   639→    description = groupEntry.origin;
   640→   640→  } else if (isEntity && kind && id) {
   641→   641→    const rep = findEntity(kind, id);
   642→   642→    const p = rep ? getEntityProfile(rep, kind) : null;
   643→   643→    title = rep?.name ?? t("Representative");
   644→   644→    description = p
   645→   645→      ? `${p.archiveNo} · ${p.rank}`
   646→   646→      : t("A named representative of the archive");
   647→   647→  }
   648→   648→
   649→   649→  return (
   650→   650→    <ModalShell
   651→   651→      open={open}
   652→   652→      onOpenChange={(o) => (o ? undefined : closeModal())}
   653→   653→      title={title}
   654→   654→      description={description}
   655→   655→      widthClass="sm:max-w-[560px]"
   656→   656→    >
   657→   657→      {isEntity && kind && id ? (
   658→   658→        <EntityDossier kind={kind} id={id} />
   659→   659→      ) : isGroup && kind && id ? (
   660→   660→        <GroupDossierBody kind={kind} id={id} />
   661→   661→      ) : null}
   662→   662→    </ModalShell>
   663→   663→  );
   664→   664→}
