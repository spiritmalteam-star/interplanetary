    28→    28→/* ---------- parsing the transmission into rich blocks ---------- */
    29→    29→
    30→    30→type Block =
    31→    31→  | { kind: "opening"; text: string }
    32→    32→  | { kind: "para"; text: string }
    33→    33→  | { kind: "list"; items: string[] }
    34→    34→  | { kind: "signature"; text: string };
    35→    35→
    36→    36→function parseBlocks(text: string): Block[] {
    37→    37→  const raw = text
    38→    38→    .split(/\n{2,}/)
    39→    39→    .map((p) => p.trim())
    40→    40→    .filter(Boolean);
    41→    41→  const blocks: Block[] = [];
    42→    42→
    43→    43→  raw.forEach((p, i) => {
    44→    44→    const isLast = i === raw.length - 1;
    45→    45→    if (isLast && (p.startsWith("—") || p.startsWith("—"))) {
    46→    46→      blocks.push({ kind: "signature", text: p });
    47→    47→      return;
    48→    48→    }
    49→    49→    const lines = p.split("\n").map((l) => l.trim());
    50→    50→    const isList =
    51→    51→      lines.length > 0 &&
    52→    52→      lines.every((l) => l.startsWith("- ") || l.startsWith("• "));
    53→    53→    if (isList) {
    54→    54→      blocks.push({
    55→    55→        kind: "list",
    56→    56→        items: lines.map((l) => l.replace(/^[-•]\s*/, "")),
    57→    57→      });
    58→    58→      return;
    59→    59→    }
    60→    60→    if (i === 0) {
    61→    61→      blocks.push({ kind: "opening", text: p });
    62→    62→      return;
    63→    63→    }
    64→    64→    blocks.push({ kind: "para", text: p });
    65→    65→  });
    66→    66→
    67→    67→  return blocks;
    68→    68→}

   602→   602→/* ---------- main view: one independent channel per scope ---------- */
   603→   603→
   604→   604→export function TransmissionView() {
   605→   605→  const activeMode = useMirror((s) => s.activeMode);
   606→   606→  const session = useMirror((s) => s.sessions[s.activeMode]);
   607→   607→  const remedyStatus = useMirror((s) => s.remedyStatus);
   608→   608→  const t = useT();
   609→   609→
   610→   610→  const scope: Scope = activeMode;
   611→   611→  const meta = SCOPE_META[scope];
   612→   612→
   613→   613→  /* The apothecary serves the concern most recently spoken — the last
   614→   614→     exchange's question, held quietly until the thread rests. */
   615→   615→  const lastSpoken = useMemo(
   616→   616→    () =>
   617→   617→      scope === "healing"
   618→   618→        ? (session.messages.at(-1)?.query ?? null)
   619→   619→        : null,
   620→   620→    [scope, session.messages]
   621→   621→  );
   622→   622→
   623→   623→  /* The thread begins where it begins: opening, reloading or switching
   624→   624→     into a channel keeps the thread at its BEGINNING. New generations
   625→   625→     load from the TOP — the newest exchange settles its first line at
   626→   626→     the top of the view and the visitor reads downward from there.
   627→   627→     One law above all: the moment a suggestion is struck — even after
   628→   628→     the thread already carries conversations — the frame must meet
   629→   629→     THAT exchange, while it forms and again when it lands. */
   630→   630→  const topRef = useRef<HTMLDivElement | null>(null);
   631→   631→  const latestRef = useRef<HTMLDivElement | null>(null);
   632→   632→  const loadingRef = useRef<HTMLDivElement | null>(null);
   633→   633→  const errorRef = useRef<HTMLDivElement | null>(null);
   634→   634→  const messagesLength = session.messages.length;
   635→   635→  const status = session.status;
   636→   636→  const baseline = useRef({
   637→   637→    mode: activeMode,
   638→   638→    len: messagesLength,
   639→   639→    status,
   640→   640→    fresh: true,
   641→   641→  });
   642→   642→
   643→   643→  useEffect(() => {
   644→   644→    const b = baseline.current;
   645→   645→    const sameMode = b.mode === activeMode;
   646→   646→    const grew = sameMode && messagesLength !== b.len;
   647→   647→    const startedLoading =
   648→   648→      sameMode && !grew && status === "loading" && b.status !== "loading";
   649→   649→    const becameError =
   650→   650→      sameMode &&
   651→   651→      !grew &&
   652→   652→      !startedLoading &&
   653→   653→      status === "error" &&
   654→   654→      b.status !== "error";
   655→   655→
   656→   656→    baseline.current = {
   657→   657→      mode: activeMode,
   658→   658→      len: messagesLength,
   659→   659→      status,
   660→   660→      fresh: false,
   661→   661→    };
   662→   662→
   663→   663→    if (b.fresh || !sameMode) {
   664→   664→      /* First sight of the channel — arriving from the observatory the
   665→   665→         transmission may ALREADY be forming (a suggestion was struck
   666→   666→         and the channel mounted mid-reception): meet it where it
   667→   667→         forms, never wait above the older thread. A channel entering
   668→   668→         with a failed transmission meets the error; otherwise it
   669→   669→         rests at its beginning. */
   670→   670→      if (status === "loading") {
   671→   671→        loadingRef.current?.scrollIntoView({ block: "start" });
   672→   672→      } else if (status === "error") {
   673→   673→        errorRef.current?.scrollIntoView({ block: "start" });
   674→   674→      } else if (!sameMode) {
   675→   675→        topRef.current?.scrollIntoView({ block: "start" });
   676→   676→      }
   677→   677→      return;
   678→   678→    }
   679→   679→    if (grew) {
   680→   680→      latestRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   681→   681→      /* the answer must own the frame — re-affirm once the entrance
   682→   682→         has settled, in case the first glide was preempted */
   683→   683→      const id = window.setTimeout(() => {
   684→   684→        latestRef.current?.scrollIntoView({
   685→   685→          behavior: "smooth",
   686→   686→          block: "start",
   687→   687→        });
   688→   688→      }, 450);
   689→   689→      return () => window.clearTimeout(id);
   690→   690→    }
   691→   691→    if (startedLoading) {
   692→   692→      loadingRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   693→   693→      return;
   694→   694→    }
   695→   695→    if (becameError) {
   696→   696→      errorRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   697→   697→    }
   698→   698→  }, [activeMode, messagesLength, status]);
   699→   699→
   700→   700→  return (
   701→   701→    <motion.section
   702→   702→      initial={{ opacity: 0, y: 12 }}
   703→   703→      animate={{ opacity: 1, y: 0 }}
   704→   704→      transition={{ duration: 0.55, ease: [0.22, 1, 0.36, 1] as const }}
   705→   705→      aria-label={`${meta.label} channel — mirror transmissions`}
   706→   706→      className={cn(
   707→   707→        "scope-" + scope,
   708→   708→        "mx-auto w-full max-w-[760px] px-1 pb-6 pt-6 sm:pt-8"
   709→   709→      )}
   710→   710→    >
   711→   711→      {/* beginning anchor — opening a channel keeps the thread here */}
   712→   712→      <div ref={topRef} aria-hidden="true" className="h-px" />
   713→   713→
   714→   714→      {/* ---------- thread ---------- */}
   715→   715→      {session.messages.length === 0 &&
   716→   716→        session.status === "idle" &&
   717→   717→        !session.error && (
   718→   718→          <div className="scope-frame-card relative mt-2 overflow-hidden rounded-2xl glass px-6 py-9 text-center">
   719→   719→            <span className="scope-corner scope-corner-tl" aria-hidden="true" />
   720→   720→            <span className="scope-corner scope-corner-tr" aria-hidden="true" />
   721→   721→            <span className="scope-corner scope-corner-bl" aria-hidden="true" />
   722→   722→            <span className="scope-corner scope-corner-br" aria-hidden="true" />
   723→   723→            <p className="text-[16.5px] font-medium text-foreground/85">
   724→   724→              {t("This channel is quiet.")}
   725→   725→            </p>
   726→   726→            <p className="mx-auto mt-2 max-w-[380px] text-[14.5px] leading-relaxed text-muted-foreground">
   727→   727→              {t(
   728→   728→                "Every scope keeps its own private channel — ask from the composer below to open the first transmission."
   729→   729→              )}
   730→   730→            </p>
   731→   731→          </div>
   732→   732→        )}
   733→   733→
   734→   734→      {session.messages.map((m, i) => (
   735→   735→        <div
   736→   736→          key={m.id}
   737→   737→          ref={
   738→   738→            i === session.messages.length - 1
   739→   739→              ? (node) => {
   740→   740→                  latestRef.current = node;
   741→   741→                }
   742→   742→              : undefined
   743→   743→          }
   744→   744→        >
   745→   745→          <Exchange
   746→   746→            message={m}
   747→   747→            index={i}
   748→   748→            scope={scope}
   749→   749→            animate={i === session.messages.length - 1 && session.status !== "loading"}
   750→   750→          />
   751→   751→        </div>
   752→   752→      ))}
   753→   753→
   754→   754→      {scope === "healing" &&
   755→   755→        session.status !== "loading" &&
   756→   756→        remedyStatus !== "crafting" &&
   757→   757→        lastSpoken && <ApothecaryStand concern={lastSpoken} />}
   758→   758→
   759→   759→      {session.status === "loading" &&
   760→   760→        session.messages.at(-1)?.visual !== "pending" && (
   761→   761→        <div
   762→   762→          ref={(node) => {
   763→   763→            loadingRef.current = node;
   764→   764→          }}
   765→   765→        >
   766→   766→          <LoadingTransmission query={session.activeQuery} />
   767→   767→        </div>
   768→   768→      )}
   769→   769→
   770→   770→      {session.status === "error" && (
   771→   771→        <div
   772→   772→          ref={errorRef}
   773→   773→          className="scope-frame-card mt-8 rounded-2xl glass p-6 text-center"
   774→   774→        >
   775→   775→          <p className="text-[16px] leading-relaxed text-foreground/85">
   776→   776→            {t(
   777→   777→              "The field received your question but could not complete the transmission."
   778→   778→            )}
   779→   779→          </p>
   780→   780→          <p className="mt-2 text-[14.5px] italic text-muted-foreground">
   781→   781→            {session.error}
   782→   782→          </p>
   783→   783→        </div>
   784→   784→      )}
   785→   785→
   786→   786→      {/* new generations settle at the TOP of this anchor — the reader
   787→   787→          begins at their first line and scrolls down for the rest */}
   788→   788→      <div aria-hidden="true" className="h-px" />
   789→   789→    </motion.section>
   790→   790→  );
   791→   791→}
