   308→/* ---------------- group dossier — deep + full reveal ---------------- */
   309→
   310→function GroupDossierBody({
   311→  kind,
   312→  id,
   313→}: {
   314→  kind: DossierKind;
   315→  id: string;
   316→}) {
   317→  const t = useT();
   318→  const entry = useMemo(() => findDossier(kind, id), [kind, id]);
   319→  const openModal = useMirror((s) => s.openModal);
   320→  const extras = useMemo(
   321→    () => (entry ? groupProfiles[entry.id] : undefined),
   322→    [entry]
   323→  );
   324→
   325→  const [visible, setVisible] = useState(BATCH);
   326→  const [filter, setFilter] = useState("");
   327→  const [prevGroupKey, setPrevGroupKey] = useState(`${kind}|${id}`);
   328→  const listRef = useRef<HTMLUListElement>(null);
   329→  const sentinelRef = useRef<HTMLDivElement>(null);
   330→
   331→  // Reset reveal state when switching groups — adjusted during render.
   332→  const groupKey = `${kind}|${id}`;
   333→  if (groupKey !== prevGroupKey) {
   334→    setPrevGroupKey(groupKey);
   335→    setVisible(BATCH);
   336→    setFilter("");
   337→  }
   338→
   339→  const filteredReps = useMemo(() => {
   340→    if (!entry) return [];
   341→    const q = filter.trim().toLowerCase();
   342→    if (!q) return entry.representatives;
   343→    return entry.representatives.filter(
   344→      (r) =>
   345→        r.name.toLowerCase().includes(q) ||
   346→        r.origin.toLowerCase().includes(q) ||
   347→        r.specialty.toLowerCase().includes(q)
   348→    );
   349→  }, [entry, filter]);
   350→
   351→  // Auto-reveal as the sentinel scrolls into view
   352→  useEffect(() => {
   353→    const el = sentinelRef.current;
   354→    if (!el) return;
   355→    const io = new IntersectionObserver(
   356→      (entries) => {
   357→        if (entries[0]?.isIntersecting) {
   358→          setVisible((v) =>
   359→            v < filteredReps.length
   360→              ? Math.min(v + BATCH, filteredReps.length)
   361→              : v
   362→          );
   363→        }
   364→      },
   365→      { rootMargin: "300px" }
   366→    );
   367→    io.observe(el);
   368→    return () => io.disconnect();
   369→  }, [filteredReps.length]);
   370→
   371→  if (!entry) return null;
   372→
   373→  const kindLabel =
   374→    kind === "civilization" ? t("Civilization dossier") : t("Interdimensional dossier");
   375→  const extras_ = extras;
   376→
   377→  return (
   378→    <>
   379→      {/* the three context images — banner impression + two context scenes */}
   380→      <div className="px-5 pt-1 sm:px-6">
   381→        <ProfileGallery
   382→          slots={[
   383→            [groupImage(kind, entry.id)],
   384→            [
   385→              sceneImageFor(entry.id, "milieu"),
   386→              groupImage(kind, entry.id),
   387→            ],
   388→            [
   389→              sceneImageForDistinct(entry.id, "homelands", [
   390→                sceneImageFor(entry.id, "milieu"),
   391→                groupImage(kind, entry.id),
   392→              ]),
   393→              sceneImageFor(entry.id, "milieu"),
   394→            ],
   395→          ]}
   396→          alts={[
   397→            t("AI-rendered scene impression of the {name}", { name: entry.name }),
   398→            t("The milieu of {name} — an impressionistic scene", { name: entry.name }),
   399→            t("The homelands of {name} — an impressionistic scene", { name: entry.name }),
   400→          ]}
   401→          testid="group-gallery"
   402→        />
   403→      </div>
   404→
   405→      <div className="flex flex-wrap items-center gap-2 px-5 pt-4 sm:px-6">
   406→        <Badge tone="cy">{kindLabel}</Badge>
   407→        <Badge tone="ok">{t("{n} named representatives", { n: entry.count })}</Badge>
   408→        <Badge tone="pk">{entry.range}</Badge>
   409→      </div>
   410→
   411→      <div className="nice-scroll max-h-[min(60vh,540px)] space-y-5 overflow-y-auto px-5 pb-4 pt-4 sm:px-6">
   412→        <ProfileSection index="01" label={t("Essence")}>
   413→          <p className="text-[14.5px] font-medium leading-relaxed text-foreground/90">{entry.essence}</p>
   414→        </ProfileSection>
   415→        <ProfileSection index="02" label={t("Role in human awakening")}>
   416→          <p className="text-[14.5px] leading-relaxed text-foreground/85">{entry.role}</p>
   417→        </ProfileSection>
   418→        <ProfileSection index="03" label={t("Signs of resonance")}>
   419→          <p className="text-[14.5px] leading-relaxed text-foreground/85">{entry.signal}</p>
   420→        </ProfileSection>
   421→
   422→        {/* deep sections */}
   423→        {extras_ && (
   424→          <>
   425→            <div className="h-px w-full bg-gradient-to-r from-[color-mix(in_srgb,var(--cy)_35%,transparent)] to-transparent" />
   426→            <ProfileSection index="04" label={t("Recorded history")}>
   427→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.history}</p>
   428→            </ProfileSection>
   429→            <ProfileSection index="05" label={t("How their society is organized")}>
   430→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.structure}</p>
   431→            </ProfileSection>
   432→            <ProfileSection index="06" label={t("Ships, temples & artifacts")}>
   433→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.artifacts}</p>
   434→            </ProfileSection>
   435→            <ProfileSection index="07" label={t("Contact protocol")}>
   436→              <p className="text-[14.5px] leading-relaxed text-foreground/85">{extras_.contactProtocol}</p>
   437→            </ProfileSection>
   438→
   439→            <ProfileSection index="08" label={t("Three core teachings")}>
   440→              <ul className="mt-1.5 space-y-1.5">
   441→                {extras_.teachings.map((teaching, i) => (
   442→                  <li key={i} className="flex items-start gap-2.5">
   443→                    <span
   444→                      className="mt-[7px] inline-block size-1.5 shrink-0 rotate-45"
   445→                      style={{ background: "var(--cy)" }}
   446→                      aria-hidden="true"
   447→                    />
   448→                    <span className="font-serif text-[14.5px] italic leading-relaxed text-foreground/85">
   449→                      {teaching}
   450→                    </span>
   451→                  </li>
   452→                ))}
   453→              </ul>
   454→            </ProfileSection>
   455→
   456→            <ProfileSection index="09" label={t("Resonant tools & practices")}>
   457→              <div className="mt-2 flex flex-wrap gap-1.5">
   458→                {extras_.resonances.map((r, i) => (
   459→                  <Badge key={i}>{r}</Badge>
   460→                ))}
   461→              </div>
   462→            </ProfileSection>
   463→
   464→            <div className="rounded-xl border hairline bg-[color-mix(in_srgb,var(--gd)_6%,transparent)] p-3.5">
   465→              <h4 className="mono-label text-[10px] text-[var(--gd)]">
   466→                {t("Discernment note")}
   467→              </h4>
   468→              <p className="mt-1.5 text-[13px] leading-relaxed text-muted-foreground">
   469→                {extras_.discernment}
   470→              </p>
   471→            </div>
   472→          </>
   473→        )}
   474→
   475→        {/* representatives — full reveal */}
   476→        <section>
   477→          <div className="flex flex-wrap items-center justify-between gap-2">
   478→            <h4 className="mono-label flex items-center gap-1.5 text-[10.5px] text-[var(--cy)]">
   479→              <Fingerprint className="size-3" aria-hidden="true" />
   480→              {t("Named representatives — the full register")}
   481→            </h4>
   482→            <Badge tone="ok">
   483→              {visible >= filteredReps.length
   484→                ? t("all {n} revealed", { n: filteredReps.length })
   485→                : t("{a} of {b} revealed", {
   486→                    a: visible,
   487→                    b: filteredReps.length,
   488→                  })}
   489→            </Badge>
   490→          </div>
   491→
   492→          {entry.representatives.length > 36 && (
   493→            <div className="relative mt-2.5">
   494→              <Search
   495→                className="pointer-events-none absolute left-2.5 top-1/2 size-3 -translate-y-1/2 text-muted-foreground/70"
   496→                aria-hidden="true"
   497→              />
   498→              <input
   499→                type="search"
   500→                value={filter}
   501→                onChange={(e) => {
   502→                  setFilter(e.target.value);
   503→                  setVisible(BATCH);
   504→                }}
   505→                placeholder={t("Filter the {n} names…", { n: entry.count })}
   506→                aria-label={t("Filter named representatives of the {name}", {
   507→                  name: entry.name,
   508→                })}
   509→                className="focus-glow h-8 w-full rounded-lg border hairline bg-transparent pl-8 pr-3 text-[13.5px] text-foreground placeholder:text-muted-foreground/60 focus:outline-none"
   510→              />
   511→            </div>
   512→          )}
   513→
   514→          <ul
   515→            ref={listRef}
   516→            className="nice-scroll mt-2 max-h-72 space-y-0.5 overflow-y-auto rounded-xl border hairline p-1.5"
   517→          >
   518→            {filteredReps.slice(0, visible).map((rep) => (
   519→              <li key={rep.id}>
   520→                <button
   521→                  type="button"
   522→                  onClick={() =>
   523→                    openModal({ type: "entity", kind, id: rep.id })
   524→                  }

   621→export function DossierModal() {
   622→  const modal = useMirror((s) => s.modal);
   623→  const closeModal = useMirror((s) => s.closeModal);
   624→  const t = useT();
   625→
   626→  const isEntity = modal?.type === "entity";
   627→  const isGroup = modal?.type === "dossier";
   628→  const kind = modal && (isEntity || isGroup) ? modal.kind : null;
   629→  const id = modal && (isEntity || isGroup) ? modal.id : null;
   630→
   631→  const groupEntry =
   632→    kind && !isEntity ? findDossier(kind, id as string) : null;
   633→  const open = (isGroup && !!groupEntry) || (isEntity && !!kind && !!id);
   634→
   635→  let title = t("Dossier");
   636→  let description: string | undefined;
   637→  if (isGroup && groupEntry) {
   638→    title = groupEntry.name;
   639→    description = groupEntry.origin;
   640→  } else if (isEntity && kind && id) {
   641→    const rep = findEntity(kind, id);
   642→    const p = rep ? getEntityProfile(rep, kind) : null;
   643→    title = rep?.name ?? t("Representative");
   644→    description = p
   645→      ? `${p.archiveNo} · ${p.rank}`
   646→      : t("A named representative of the archive");
   647→  }
   648→
   649→  return (
   650→    <ModalShell
   651→      open={open}
   652→      onOpenChange={(o) => (o ? undefined : closeModal())}
   653→      title={title}
   654→      description={description}
   655→      widthClass="sm:max-w-[560px]"
   656→    >
   657→      {isEntity && kind && id ? (
   658→        <EntityDossier kind={kind} id={id} />
   659→      ) : isGroup && kind && id ? (
   660→        <GroupDossierBody kind={kind} id={id} />
   661→      ) : null}
   662→    </ModalShell>
   663→  );
   664→}
