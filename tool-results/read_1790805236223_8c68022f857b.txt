    25→/* ---------- parsing the transmission into rich blocks ---------- */
    26→
    27→type Block =
    28→  | { kind: "opening"; text: string }
    29→  | { kind: "para"; text: string }
    30→  | { kind: "list"; items: string[] }
    31→  | { kind: "signature"; text: string };
    32→
    33→function parseBlocks(text: string): Block[] {
    34→  const raw = text
    35→    .split(/\n{2,}/)
    36→    .map((p) => p.trim())
    37→    .filter(Boolean);
    38→  const blocks: Block[] = [];
    39→
    40→  raw.forEach((p, i) => {
    41→    const isLast = i === raw.length - 1;
    42→    if (isLast && (p.startsWith("—") || p.startsWith("—"))) {
    43→      blocks.push({ kind: "signature", text: p });
    44→      return;
    45→    }
    46→    const lines = p.split("\n").map((l) => l.trim());
    47→    const isList =
    48→      lines.length > 0 &&
    49→      lines.every((l) => l.startsWith("- ") || l.startsWith("• "));
    50→    if (isList) {
    51→      blocks.push({
    52→        kind: "list",
    53→        items: lines.map((l) => l.replace(/^[-•]\s*/, "")),
    54→      });
    55→      return;
    56→    }
    57→    if (i === 0) {
    58→      blocks.push({ kind: "opening", text: p });
    59→      return;
    60→    }
    61→    blocks.push({ kind: "para", text: p });
    62→  });
    63→
    64→  return blocks;
    65→}

   298→function ExchangeVisual({ message }: { message: ChatMessage }) {
   299→  const t = useT();
   300→  const askScopeVisual = useMirror((s) => s.askScopeVisual);
   301→  const activeMode = useMirror((s) => s.activeMode);
   302→
   303→  if (message.visual === "pending") {

   306→        accent="var(--scope-a)"
   307→        testIdPrefix="scope-visual"
   308→        repaint
   309→      />

   311→  }
   312→  if (message.visual === "error") {
   313→    return (
   314→      <div
   315→        className="glass rounded-2xl border border-[color-mix(in_srgb,var(--hairline)_65%,transparent)] px-4 py-3.5"
   316→        data-testid="scope-visual-error"
   317→      >
   318→        <p className="text-[14px] italic leading-relaxed text-foreground/80">
   319→          {t(
   320→            "The atelier is quiet — the vision could not be composed. Rest a breath, then ask again."
   321→          )}
   322→        </p>
   323→        {message.visualRequest && (
   324→          <button
   325→            type="button"
   326→            onClick={() =>
   327→              void askScopeVisual(activeMode, message.visualRequest ?? "", null)
   328→            }
   329→            data-testid="scope-visual-retry"
   330→            className="focus-glow mt-2.5 flex h-9 items-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--scope-a)_35%,transparent)] px-4 text-[13px] text-foreground/90 transition-all duration-300 hover:-translate-y-px"
   331→          >
   332→            <RotateCcw className="size-3.5" aria-hidden="true" />
   333→            {t("Be still and receive")}
   334→          </button>
   335→        )}
   336→      </div>
   337→    );
   338→  }
   339→  if (!message.artifact) return null;
   340→  const artifact = message.artifact;
   341→  if (!artifact.imageUrl && artifact.slides.length === 0) {
   342→    return (
   343→      <PreparedPromptFallback
   344→        artifact={artifact}
   345→        accent="var(--scope-a)"
   346→        testIdPrefix="scope-visual"
   347→        onPaint={() =>
   348→          void askScopeVisual(
   349→            activeMode,
   350→            message.visualRequest ?? artifact.subject,
   351→            {
   352→              id: message.id,
   353→              request: message.visualRequest ?? artifact.subject,
   354→              prompt: artifact.prompt,
   355→              subject: artifact.subject,
   356→              mode: artifact.mode,
   357→            }
   358→          )
   359→        }
   360→      />
   361→    );
   362→  }
   363→  return (
   364→    <VisualizationCard
   365→      artifact={artifact}
   366→      accent="var(--scope-a)"
   367→      testIdPrefix="scope-visual"
   368→      onRegenerate={() =>
   369→        void askScopeVisual(activeMode, message.visualRequest ?? artifact.subject, {
   370→          id: message.id,
   371→          request: message.visualRequest ?? artifact.subject,
   372→          prompt: artifact.prompt,
   373→          subject: artifact.subject,
   374→          mode: artifact.mode,
   375→        })
   376→      }
   377→    />
   378→  );
   379→}

   381→/* ---------- one completed exchange ---------- */
   382→
   383→function Exchange({
   384→  message,
   385→  index,
   386→  animate,
   387→  scope,
   388→}: {
   389→  message: ChatMessage;
   390→  index: number;
   391→  animate: boolean;
   392→  scope: Scope;
   393→}) {
   394→  const scopeMeta = SCOPE_META[scope];
   395→  const [copied, setCopied] = useState(false);
   396→  const t = useT();
   397→  /* Each card draws its own light — quietly, never spoken of. */
   398→  const aura = auraFor(message.id);
   399→
   400→  const handleCopy = async () => {
   401→    try {
   402→      await navigator.clipboard.writeText(message.text);
   403→      setCopied(true);
   404→      window.setTimeout(() => setCopied(false), 2200);
   405→    } catch {
   406→      /* clipboard unavailable — quiet fail */
   407→    }
   408→  };
   409→
   410→  const item = animate ? staggerItem : stillItem;
   411→  const hasVisual = Boolean(message.artifact || message.visual);
   412→
   413→  return (
   414→    <motion.article
   415→      initial={{ opacity: animate ? 0 : 1, y: animate ? 12 : 0 }}
   416→      animate={{ opacity: 1, y: 0 }}
   417→      transition={{ duration: animate ? 0.55 : 0, ease: [0.22, 1, 0.36, 1] }}

   419→      style={{ "--scope-a": aura.a, "--scope-b": aura.b } as CSSProperties}
   420→      className="pt-7 first:pt-0"
   421→    >
   422→      {/* index ribbon */}

   584→        {t("Prepared from what has been spoken here")}
   585→      </p>
   586→    </motion.div>
   587→  );
   588→}

   590→/* ---------- main view: one independent channel per scope ---------- */
   591→
   592→export function TransmissionView() {
   593→  const activeMode = useMirror((s) => s.activeMode);
   594→  const session = useMirror((s) => s.sessions[s.activeMode]);
   595→  const remedyStatus = useMirror((s) => s.remedyStatus);
   596→  const t = useT();
   597→
   598→  const scope: Scope = activeMode;
   599→  const meta = SCOPE_META[scope];
   600→
   601→  /* The apothecary serves the concern most recently spoken — the last
   602→     exchange's question, held quietly until the thread rests. */
   603→  const lastSpoken = useMemo(
   604→    () =>
   605→      scope === "healing"
   606→        ? (session.messages.at(-1)?.query ?? null)
   607→        : null,
   608→    [scope, session.messages]
   609→  );
   610→
   611→  /* The thread begins where it begins: opening, reloading or switching
   612→     into a channel keeps the thread at its BEGINNING. New generations
   613→     load from the TOP — the newest exchange settles its first line at
   614→     the top of the view and the visitor reads downward from there.
   615→     One law above all: the moment a suggestion is struck — even after
   616→     the thread already carries conversations — the frame must meet
   617→     THAT exchange, while it forms and again when it lands. */
   618→  const topRef = useRef<HTMLDivElement | null>(null);
   619→  const latestRef = useRef<HTMLDivElement | null>(null);
   620→  const loadingRef = useRef<HTMLDivElement | null>(null);
   621→  const errorRef = useRef<HTMLDivElement | null>(null);
   622→  const messagesLength = session.messages.length;
   623→  const status = session.status;
   624→  const baseline = useRef({
   625→    mode: activeMode,
   626→    len: messagesLength,
   627→    status,
   628→    fresh: true,
   629→  });
   630→
   631→  useEffect(() => {
   632→    const b = baseline.current;
   633→    const sameMode = b.mode === activeMode;
   634→    const grew = sameMode && messagesLength !== b.len;
   635→    const startedLoading =
   636→      sameMode && !grew && status === "loading" && b.status !== "loading";
   637→    const becameError =
   638→      sameMode &&
   639→      !grew &&
   640→      !startedLoading &&
   641→      status === "error" &&
   642→      b.status !== "error";
   643→
   644→    baseline.current = {
   645→      mode: activeMode,
   646→      len: messagesLength,
   647→      status,
   648→      fresh: false,
   649→    };
   650→
   651→    if (b.fresh || !sameMode) {
   652→      /* First sight of the channel — arriving from the observatory the
   653→         transmission may ALREADY be forming (a suggestion was struck
   654→         and the channel mounted mid-reception): meet it where it
   655→         forms, never wait above the older thread. A channel entering
   656→         with a failed transmission meets the error; otherwise it
   657→         rests at its beginning. */
   658→      if (status === "loading") {
   659→        loadingRef.current?.scrollIntoView({ block: "start" });
   660→      } else if (status === "error") {
   661→        errorRef.current?.scrollIntoView({ block: "start" });
   662→      } else if (!sameMode) {
   663→        topRef.current?.scrollIntoView({ block: "start" });
   664→      }
   665→      return;
   666→    }
   667→    if (grew) {
   668→      latestRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   669→      /* the answer must own the frame — re-affirm once the entrance
   670→         has settled, in case the first glide was preempted */
   671→      const id = window.setTimeout(() => {
   672→        latestRef.current?.scrollIntoView({
   673→          behavior: "smooth",
   674→          block: "start",
   675→        });
   676→      }, 450);
   677→      return () => window.clearTimeout(id);
   678→    }
   679→    if (startedLoading) {
   680→      loadingRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   681→      return;
   682→    }
   683→    if (becameError) {
   684→      errorRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   685→    }
   686→  }, [activeMode, messagesLength, status]);
   687→
   688→  return (
   689→    <motion.section
   690→      initial={{ opacity: 0, y: 12 }}
   691→      animate={{ opacity: 1, y: 0 }}
   692→      transition={{ duration: 0.55, ease: [0.22, 1, 0.36, 1] }}
   693→      aria-label={`${meta.label} channel — mirror transmissions`}
   694→      className={cn(
   695→        "scope-" + scope,
   696→        "mx-auto w-full max-w-[760px] px-1 pb-6 pt-6 sm:pt-8"
   697→      )}
   698→    >
   699→      {/* beginning anchor — opening a channel keeps the thread here */}
   700→      <div ref={topRef} aria-hidden="true" className="h-px" />
   701→
   702→      {/* ---------- thread ---------- */}
   703→      {session.messages.length === 0 &&
   704→        session.status === "idle" &&
   705→        !session.error && (
   706→          <div className="scope-frame-card relative mt-2 overflow-hidden rounded-2xl glass px-6 py-9 text-center">
   707→            <span className="scope-corner scope-corner-tl" aria-hidden="true" />
   708→            <span className="scope-corner scope-corner-tr" aria-hidden="true" />
   709→            <span className="scope-corner scope-corner-bl" aria-hidden="true" />
   710→            <span className="scope-corner scope-corner-br" aria-hidden="true" />
   711→            <p className="text-[16.5px] font-medium text-foreground/85">
   712→              {t("This channel is quiet.")}
   713→            </p>
   714→            <p className="mx-auto mt-2 max-w-[380px] text-[14.5px] leading-relaxed text-muted-foreground">
   715→              {t(
   716→                "Every scope keeps its own private channel — ask from the composer below to open the first transmission."
   717→              )}
   718→            </p>
   719→          </div>
   720→        )}
   721→
   722→      {session.messages.map((m, i) => (
   723→        <div
   724→          key={m.id}
   725→          ref={
   726→            i === session.messages.length - 1
   727→              ? (node) => {
   728→                  latestRef.current = node;
   729→                }
   730→              : undefined
   731→          }
   732→        >
   733→          <Exchange
   734→            message={m}
   735→            index={i}
   736→            scope={scope}
   737→            animate={i === session.messages.length - 1 && session.status !== "loading"}
   738→          />
   739→        </div>
   740→      ))}
   741→
   742→      {scope === "healing" &&
   743→        session.status !== "loading" &&
   744→        remedyStatus !== "crafting" &&
   745→        lastSpoken && <ApothecaryStand concern={lastSpoken} />}
   746→
   747→      {session.status === "loading" &&
   748→        session.messages.at(-1)?.visual !== "pending" && (
   749→        <div
   750→          ref={(node) => {
   751→            loadingRef.current = node;
   752→          }}
   753→        >
   754→          <LoadingTransmission scope={scope} query={session.activeQuery} />
   755→        </div>
   756→      )}
   757→
   758→      {session.status === "error" && (
   759→        <div
   760→          ref={errorRef}
   761→          className="scope-frame-card mt-8 rounded-2xl glass p-6 text-center"
   762→        >
   763→          <p className="text-[16px] leading-relaxed text-foreground/85">
   764→            {t(
   765→              "The field received your question but could not complete the transmission."
   766→            )}
   767→          </p>
   768→          <p className="mt-2 text-[14.5px] italic text-muted-foreground">
   769→            {session.error}
   770→          </p>
   771→        </div>
   772→      )}
   773→
   774→      {/* new generations settle at the TOP of this anchor — the reader
   775→          begins at their first line and scrolls down for the rest */}
   776→      <div aria-hidden="true" className="h-px" />
   777→    </motion.section>
   778→  );
   779→}
