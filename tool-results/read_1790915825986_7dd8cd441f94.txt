    49→function hash32(str: string): number {
    50→  let h = 2166136261;
    51→  for (let i = 0; i < str.length; i++) {
    52→    h ^= str.charCodeAt(i);
    53→    h = Math.imul(h, 16777619);
    54→  }
    55→  return h >>> 0;
    56→}
    57→
    58→function mulberry32(seed: number) {
    59→  let a = seed;
    60→  return () => {
    61→    a |= 0;
    62→    a = (a + 0x6d2b79f5) | 0;
    63→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    64→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    65→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    66→  };
    67→}

    69→export function InkSigil({ text, size = 44 }: { text: string; size?: number }) {
    70→  const svg = useMemo(() => {
    71→    const rnd = mulberry32(hash32(text.trim() || "mirror"));
    72→    const c = 60;
    73→    const sides = 3 + Math.floor(rnd() * 6);
    74→    const rot = rnd() * 360;
    75→    const r2 = 34 + rnd() * 10;
    76→    const dash = `${5 + rnd() * 14} ${3 + rnd() * 9}`;
    77→    const pts: string[] = [];
    78→    for (let i = 0; i < sides; i++) {
    79→      const a = ((rot + (360 / sides) * i) * Math.PI) / 180;
    80→      pts.push(`${(c + r2 * Math.cos(a)).toFixed(1)},${(c + r2 * Math.sin(a)).toFixed(1)}`);
    81→    }
    82→    const spokes = 5 + Math.floor(rnd() * 6);
    83→    const lines: string[] = [];
    84→    for (let i = 0; i < spokes; i++) {
    85→      const a = ((rot + 12 + (360 / spokes) * i) * Math.PI) / 180;
    86→      lines.push(
    87→        `<line x1="${(c + 16 * Math.cos(a)).toFixed(1)}" y1="${(c + 16 * Math.sin(a)).toFixed(1)}" x2="${(c + 47 * Math.cos(a)).toFixed(1)}" y2="${(c + 47 * Math.sin(a)).toFixed(1)}" stroke="currentColor" stroke-opacity="0.38" stroke-width="1"/>`
    88→      );
    89→    }
    90→    return `<svg width="${size}" height="${size}" viewBox="0 0 120 120" xmlns="http://www.w3.org/2000/svg">
    91→      <circle cx="60" cy="60" r="55" fill="none" stroke="currentColor" stroke-opacity="0.5" stroke-width="1.4"/>
    92→      <circle cx="60" cy="60" r="40" fill="none" stroke="currentColor" stroke-opacity="0.3" stroke-width="1" stroke-dasharray="${dash}"/>
    93→      <polygon points="${pts.join(" ")}" fill="none" stroke="currentColor" stroke-opacity="0.6" stroke-width="1.2"/>
    94→      ${lines.join("")}
    95→      <circle cx="60" cy="60" r="3.2" fill="currentColor" fill-opacity="0.85"/>
    96→    </svg>`;
    97→  }, [text, size]);
    98→
    99→  return (
   100→    <span
   101→      aria-hidden="true"
   102→      className="inline-block shrink-0"
   103→      dangerouslySetInnerHTML={{ __html: svg }}
   104→    />
   105→  );
   106→}
