    51→    51→let communionIdSeed = 0;

    57→    57→/**

    65→    65→export function CommunionView() {
    66→    66→  const closeCommunion = useMirror((s) => s.closeCommunion);
    67→    67→  const language = useMirror((s) => s.language);
    68→    68→  const t = useT();
    69→    69→
    70→    70→  const [messages, setMessages] = useState<CommunionMessage[]>([]);
    71→    71→  const [receiving, setReceiving] = useState(false);
    72→    72→  const [error, setError] = useState(false);
    73→    73→  const [draft, setDraft] = useState("");
    74→    74→  const [attachments, setAttachments] = useState<ChatAttachment[]>([]);
    75→    75→  /* "this" in a follow-up refers to the last artifact of the meeting */
    76→    76→  const visualContextRef = useRef<{ subject: string; mode: string } | null>(
    77→    77→    null
    78→    78→  );
    79→    79→  const endRef = useRef<HTMLDivElement | null>(null);
    80→    80→  const latestRef = useRef<HTMLDivElement | null>(null);
    81→    81→  const openedRef = useRef(false);
    82→    82→  const receivingRef = useRef(false);
    83→    83→  const messagesRef = useRef<CommunionMessage[]>([]);
    84→    84→

    86→    86→  useEffect(() => {

    88→    88→  }, [messages]);
    89→    89→
    90→    90→  /* Every generation loads from the TOP: when a new word arrives, its

    96→    96→

   100→   100→  }, [receiving, error]);
   101→   101→
   102→   102→  const transmit = useCallback(

   104→   104→      visitorWords: string | null,
   105→   105→      fresh = false,
   106→   106→      carried?: ChatAttachment[]

   111→   111→      setError(false);

   118→   118→          id: nextCommunionId(),
   119→   119→          role: "visitor",
   120→   120→          text: visitorWords ?? "",
   121→   121→          ...(carried && carried.length > 0
   122→   122→            ? {
   123→   123→                attachments: {
   124→   124→                  images: carried.filter((a) => a.kind === "image").length,
   125→   125→                  docNames: carried
   126→   126→                    .filter((a) => a.kind === "document")
   127→   127→                    .map((a) => a.name),
   128→   128→                },
   129→   129→              }
   130→   130→            : {}),
   131→   131→        };
   132→   132→        messagesRef.current = [...messagesRef.current, entry];
   133→   133→        setMessages(messagesRef.current);
   134→   134→      }
   135→   135→
   136→   136→      const history = messagesRef.current
   137→   137→        .slice(-14)
   138→   138→        .map((m) => ({
   139→   139→          role: m.role,
   140→   140→          text: m.artifact
   141→   141→            ? `[a visualization was created: "${m.artifact.title}" — ${m.artifact.subject}]`
   142→   142→            : m.text,
   143→   143→        }));
   144→   144→
   145→   145→      try {
   146→   146→        const payload = carried ? attachmentsToPayload(carried) : null;
   147→   147→        const res = await fetch("/api/communion", {
   148→   148→          method: "POST",
   149→   149→          headers: { "Content-Type": "application/json" },
   150→   150→          body: JSON.stringify({
   151→   151→            language,
   152→   152→            message: visitorWords,
   153→   153→            history,
   154→   154→            ...(payload ?? {}),
   155→   155→          }),
   156→   156→        });
   157→   157→        const data = (await res.json()) as {
   158→   158→          transmission?: string;
   159→   159→          error?: string;
   160→   160→        };

   162→   162→          throw new Error(data.error ?? "the Reflection stayed still");
   163→   163→        }

   165→   165→          id: nextCommunionId(),
   166→   166→          role: "mirror",
   167→   167→          text: data.transmission,
   168→   168→        };
   169→   169→        messagesRef.current = [...messagesRef.current, entry];
   170→   170→        setMessages(messagesRef.current);
   171→   171→        setError(false);
   172→   172→      } catch {
   173→   173→        setError(true);
   174→   174→      } finally {
   175→   175→        receivingRef.current = false;
   176→   176→        setReceiving(false);
   177→   177→      }
   178→   178→    },
   179→   179→    [language]
   180→   180→  );
   181→   181→
   182→   182→  /* the Reflection opens every communion with its own first word */
   183→   183→  useEffect(() => {
   184→   184→    if (openedRef.current) return;
   185→   185→    openedRef.current = true;

   192→   192→  /* ---------------------------------------------------------------- */

   200→   200→      receivingRef.current = true;

   202→   202→      setError(false);

   210→   210→          m.id === regenerateOf.id ? { ...m, pendingVisual: true } : m

   214→   214→          id: messageId,
   215→   215→          role: "visitor",
   216→   216→          text: visitorWords,
   217→   217→          ...(carried && carried.length > 0
   218→   218→            ? {
   219→   219→                attachments: {
   220→   220→                  images: carried.filter((a) => a.kind === "image").length,
   221→   221→                  docNames: carried
   222→   222→                    .filter((a) => a.kind === "document")
   223→   223→                    .map((a) => a.name),
   224→   224→                },
   225→   225→              }
   226→   226→            : {}),

   248→   248→
   249→   249→      const prior = regenerateOf;

   256→   256→            language,
   257→   257→            message: prior ? prior.visualRequest ?? visitorWords : visitorWords,

   259→   259→            ...(visualContextRef.current && !prior
   260→   260→              ? {
   261→   261→                  contextSubject: visualContextRef.current.subject,

   264→   264→              : {}),
   265→   265→            ...(prior

   268→   268→                  previousPrompt: prior.artifact?.prompt,
   269→   269→                  contextSubject: prior.artifact?.subject,
   270→   270→                  previousMode: prior.artifact?.mode,
   271→   271→                }
   272→   272→              : {}),
   273→   273→            ...(payload ?? {}),
   274→   274→          }),
   275→   275→        });

   277→   277→          artifact?: VisualizationArtifact;
   278→   278→          painted?: boolean;
   279→   279→          error?: string;
   280→   280→        };

   282→   282→          throw new Error(data.error ?? "the atelier stayed quiet");
   283→   283→        }

   285→   285→        visualContextRef.current = { subject: artifact.subject, mode: artifact.mode };

   287→   287→        messagesRef.current = messagesRef.current.map((m) =>
   288→   288→          m.id === targetId
   289→   289→            ? {
   290→   290→                ...m,
   291→   291→                pendingVisual: false,
   292→   292→                artifact,
   293→   293→                visualError: false,
   294→   294→                visualRequest: prior ? prior.visualRequest : visitorWords,
   295→   295→              }
   296→   296→            : m
   297→   297→        );
   298→   298→        setMessages(messagesRef.current);
   299→   299→        setError(false);

   301→   301→        const failId = prior ? prior.id : pendingId;
   302→   302→        messagesRef.current = messagesRef.current.map((m) =>
   303→   303→          m.id === failId
   304→   304→            ? { ...m, pendingVisual: false, visualError: true }
   305→   305→            : m
   306→   306→        );
   307→   307→        setMessages(messagesRef.current);
   308→   308→        setError(false);
   309→   309→      } finally {
   310→   310→        receivingRef.current = false;
   311→   311→        setReceiving(false);
   312→   312→      }
   313→   313→    },
   314→   314→    [language]
   315→   315→  );

   319→   319→    (message: CommunionMessage) => {

   321→   321→        message.visualRequest ?? message.artifact?.subject ?? "",
   322→   322→        undefined,
   323→   323→        message

   325→   325→    },
   326→   326→    [requestVisualization]
   327→   327→  );
   328→   328→
   329→   329→  const sendText = (v: string) => {

   332→   332→    const carried = attachments.length > 0 ? attachments : undefined;
   333→   333→    /* No button, no wand — a request to see simply is one. Ordinary

   336→   336→    const wantsVisual =
   337→   337→      val &&
   338→   338→      (intent.direct ||
   339→   339→        (intent.followUp && visualContextRef.current !== null));
   340→   340→    setDraft("");
   341→   341→    setAttachments([]);
   342→   342→    if (wantsVisual && val) {
   343→   343→      void requestVisualization(val, carried);
   344→   344→    } else {
   345→   345→      void transmit(val || null, false, carried);
   346→   346→    }
   347→   347→  };
   348→   348→
   349→   349→  const send = (e: FormEvent) => {
   350→   350→    e.preventDefault();
   351→   351→    sendText(draft);
   352→   352→  };
   353→   353→
   354→   354→  /* the voice becomes words in the field — visible for a breath — then
   355→   355→     the words travel to the Reflection on their own */
   356→   356→  const sendTextRef = useRef(sendText);
   357→   357→  sendTextRef.current = sendText;
   358→   358→  const draftRef = useRef(draft);
   359→   359→  draftRef.current = draft;
   360→   360→  const handleVoiceSubmit = (text: string) => {
   361→   361→    const prev = draftRef.current.trim();
   362→   362→    const finalText = prev ? `${prev} ${text}` : text;
   363→   363→    setDraft(finalText);
   364→   364→    window.setTimeout(() => sendTextRef.current(finalText), 650);
   365→   365→  };
   366→   366→

   368→   368→    <div
   369→   369→      className="communion-deep relative flex h-full flex-col overflow-x-clip overflow-y-clip"
   370→   370→      data-testid="communion-view"
   371→   371→    >
   372→   372→      {/* ---------- drifting deep-field glows ---------- */}
   373→   373→      <div
   374→   374→        aria-hidden="true"
   375→   375→        className="animate-drift-a pointer-events-none absolute -left-32 top-[-10%] size-[420px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,var(--sp-a)_16%,transparent),transparent_65%)] blur-3xl"
   376→   376→      />
   377→   377→      <div
   378→   378→        aria-hidden="true"
   379→   379→        className="animate-drift-c pointer-events-none absolute -right-24 bottom-[-12%] size-[380px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,var(--sp-b)_13%,transparent),transparent_65%)] blur-3xl"
   380→   380→      />
   381→   381→
   382→   382→      {/* ---------- the threshold: return · name · new communion ---------- */}
   383→   383→      <header className="relative z-20 flex items-center gap-2 border-b border-[color-mix(in_srgb,var(--sp-b)_14%,transparent)] bg-[color-mix(in_srgb,#0a0616_62%,transparent)] px-3 py-2.5 backdrop-blur-md sm:px-5">

   390→   390→        >
   391→   391→          <ArrowLeft

   398→   398→        {/* the living name of the doorway */}

   400→   400→          <span className="communion-halo relative size-9 shrink-0 overflow-hidden rounded-full border border-[color-mix(in_srgb,var(--sp-b)_38%,transparent)] shadow-[0_0_24px_-6px_color-mix(in_srgb,var(--sp-b)_70%,transparent)]">
   401→   401→            <img

   405→   405→            />
   406→   406→          </span>

   408→   408→            {t("Meet with the Reflection of the Absolute")}
   409→   409→          </span>
