   547→    if (raw && raw !== ledgerSnapshot) {
   548→      ledgerSnapshot = raw;
   549→      ledgerListeners.forEach((l) => l());
   550→    }
   551→  } catch {
   552→    /* storage unavailable */
   553→  }
   554→}

   556→function parseLedger(json: string): LedgerEntry[] {
   557→  try {
   558→    const parsed = JSON.parse(json) as LedgerEntry[];
   559→    return Array.isArray(parsed) ? parsed : [];
   560→  } catch {
   561→    return [];
   562→  }
   563→}
