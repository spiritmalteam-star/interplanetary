    34→
    35→interface AkashicRecord {
    36→  title: string;
    37→  era: string;
    38→  record: string;
    39→  seal: string;
    40→}
    41→
    42→interface AkashicResponse extends AkashicRecord {
    43→  entrance?: string;
    44→  error?: string;
    45→}
    46→
    47→/**
    48→ * AkashicView — the ancient one's reading room. A wall of golden
    49→ * light codes, and one sheet of papyrus on which the Librarian sets
    50→ * down a record retrieved by the visitor's resonance. The hand is
    51→ * old but readable; the voice, an old man behind a great desk.
    52→ */
    53→export function AkashicView() {
    54→  const exitAkashic = useMirror((s) => s.exitAkashic);
    55→  const language = useMirror((s) => s.language);
    56→  const t = useT();
    57→
    58→  const [record, setRecord] = useState<AkashicRecord | null>(null);
    59→  const [seeking, setSeeking] = useState(false);
    60→  const [error, setError] = useState(false);
    61→  const [draft, setDraft] = useState("");
    62→  const [copied, setCopied] = useState(false);
    63→  const [attachments, setAttachments] = useState<ChatAttachment[]>([]);
    64→
    65→  /* the Universal Visualization Engine — the Library also answers in
    66→     images when the visitor asks to see */
    67→  const [visual, setVisual] = useState<{
    68→    state: "pending" | "error" | "ready";
    69→    request: string;
    70→    artifact?: VisualizationArtifact;
    71→  } | null>(null);
    72→  const visualContextRef = useRef<{ subject: string; mode: string } | null>(
    73→    null
    74→  );
    75→  const visualBusyRef = useRef(false);
    76→
    77→  const seekingRef = useRef(false);
    78→  /* the last entrance kinds the Librarian used — never the same door twice */
    79→  const recentEntrancesRef = useRef<string[]>([]);
    80→  /* whether the attempt in flight (or the one that failed) was a reply */
    81→  const [replyAttempt, setReplyAttempt] = useState(false);
    82→
    83→  const seek = useCallback(
    84→    async (
    85→      resonance: string | null,
    86→      asReply: boolean,
    87→      carried?: ChatAttachment[]
    88→    ) => {
    89→      if (seekingRef.current) return;
    90→      seekingRef.current = true;
    91→      setReplyAttempt(asReply);
    92→      setSeeking(true);
    93→      setError(false);
    94→      setVisual(null);
    95→
    96→      try {
    97→        const payload = carried ? attachmentsToPayload(carried) : null;
    98→        const res = await fetch("/api/akashic", {
    99→          method: "POST",
   100→          headers: { "Content-Type": "application/json" },
   101→          body: JSON.stringify({
   102→            language,
   103→            resonance,
   104→            thread:
   105→              asReply && record
   106→                ? {
   107→                    title: record.title,
   108→                    era: record.era,
   109→                    record: record.record,
   110→                    seal: record.seal,
   111→                  }
   112→                : undefined,
   113→            recentEntrances: recentEntrancesRef.current,
   114→            ...(payload ?? {}),
   115→          }),
   116→        });
   117→        const data = (await res.json()) as AkashicResponse;
   118→        if (!res.ok || !data.record) {
   119→          throw new Error(data.error ?? "the shelf stayed quiet");
   120→        }
   121→        if (data.entrance) {
   122→          recentEntrancesRef.current = [
   123→            data.entrance,
   124→            ...recentEntrancesRef.current.filter((k) => k !== data.entrance),
   125→          ].slice(0, 4);
   126→        }
   127→        setRecord({
   128→          title: data.title ?? "A Record Set Aside",
   129→          era: data.era ?? "inscribed in an age the shelves remember",
   130→          record: data.record,
   131→          seal: data.seal ?? "— the Keeper of Records",
   132→        });
   133→      } catch {
   134→        setError(true);
   135→      } finally {
   136→        seekingRef.current = false;
   137→        setSeeking(false);
   138→      }
   139→    },
   140→    [language, record]
   141→  );
   142→
   143→  /* ---------- copy · share · the Librarian's own voice ---------- */
   144→
   145→  const recordText = record
   146→    ? `${record.title}\n${record.era}\n\n${record.record}\n\n${record.seal}`
   147→    : "";
   148→
   149→  const handleCopy = async () => {
   150→    if (!recordText) return;
   151→    try {
   152→      await navigator.clipboard.writeText(recordText);
   153→      setCopied(true);
   154→      toast.success(t("The record is copied."));
   155→      window.setTimeout(() => setCopied(false), 2200);
   156→    } catch {
   157→      toast.error(t("The copying did not take — select and copy by hand."));
   158→    }
   159→  };
   160→
   161→  const handleShare = async () => {
   162→    if (!recordText) return;
   163→    const shareTitle = record ? `Akashic Library — ${record.title}` : "Akashic Library";
   164→    try {
   165→      if (typeof navigator.share === "function") {
   166→        await navigator.share({ title: shareTitle, text: recordText });
   167→        return;
   168→      }
   169→      throw new Error("share unavailable");
   170→    } catch (err) {
   171→      if (err instanceof DOMException && err.name === "AbortError") return;
   172→      try {
   173→        await navigator.clipboard.writeText(recordText);
   174→        toast.success(t("Sharing is quiet here — the record is copied instead."));
   175→      } catch {
   176→        toast.error(t("The copying did not take — select and copy by hand."));
   177→      }
   178→    }
   179→  };
   180→
   181→  const [voiceState, setVoiceState] = useState<
   182→    "idle" | "loading" | "playing"
   183→  >("idle");
   184→  const audioRef = useRef<HTMLAudioElement | null>(null);
   185→
   186→  const stopVoice = useCallback(() => {
   187→    try {
   188→      audioRef.current?.pause();
   189→    } catch {
   190→      /* already stopped */
   191→    }
   192→    audioRef.current = null;
   193→    setVoiceState("idle");
   194→  }, []);
   195→
   196→  useEffect(() => stopVoice, [stopVoice]);
   197→
   198→  const handleListen = useCallback(async () => {
   199→    if (voiceState === "loading") return;
   200→    if (voiceState === "playing") {
   201→      stopVoice();
   202→      return;
   203→    }
   204→    if (!record) return;
   205→    /* The Librarian reads aloud himself — the old man's own voice,
   206→       slower than the world, from behind the great desk. */
   207→    setVoiceState("loading");
   208→    try {
   209→      const res = await fetch("/api/tts", {
   210→        method: "POST",
   211→        headers: { "Content-Type": "application/json" },
   212→        body: JSON.stringify({
   213→          text: `${record.title}. ${record.era}. ${record.record.replaceAll(
   214→            /\s*\n\s*/g,
   215→            " "
   216→          )} ${record.seal}`,
   217→          voice: "regent",
   218→          pace: 0.82,
   219→        }),
   220→      });
   221→      if (!res.ok) throw new Error("voice quiet");
   222→      const blob = await res.blob();
   223→      const audio = new Audio(URL.createObjectURL(blob));
   224→      audioRef.current = audio;
   225→      audio.onended = () => {
   226→        if (audioRef.current === audio) audioRef.current = null;
   227→        setVoiceState("idle");
   228→      };
   229→      audio.onerror = () => {
   230→        if (audioRef.current === audio) audioRef.current = null;
   231→        setVoiceState("idle");
   232→      };
   233→      await audio.play();
   234→      setVoiceState("playing");
   235→    } catch {
   236→      setVoiceState("idle");
   237→      toast.error(t("The Librarian is quiet. Rest, then listen again."));
   238→    }
   239→  }, [record, voiceState, stopVoice, t]);
   240→
   241→  /* ---------------------------------------------------------------- */
   242→  /*  The Universal Visualization Engine — a request to SEE becomes    */
   243→  /*  a vision the Library sets beside the open record.                */
   244→  /* ---------------------------------------------------------------- */
   245→  const requestVisualization = useCallback(
   246→    async (request: string) => {
   247→      if (visualBusyRef.current) return;
   248→      visualBusyRef.current = true;
   249→      setVisual({ state: "pending", request });
   250→
   251→      try {
   252→        const res = await fetch("/api/visualize", {
   253→          method: "POST",
   254→          headers: { "Content-Type": "application/json" },
   255→          body: JSON.stringify({
   256→            language,
   257→            message: request,
   258→            history: record
   259→              ? [
   260→                  {
   261→                    role: "mirror",
   262→                    text: `${record.title}. ${record.era}. ${record.record}`,
   263→                  },
   264→                ]
   265→              : [],
   266→            ...(visualContextRef.current
   267→              ? {
   268→                  contextSubject: visualContextRef.current.subject,
   269→                  previousMode: visualContextRef.current.mode,
   270→                }
   271→              : {}),
   272→          }),
   273→        });
   274→        const data = (await res.json().catch(() => null)) as {
   275→          artifact?: VisualizationArtifact;
   276→          error?: string;
   277→        } | null;
   278→        if (!res.ok || !data?.artifact) {
   279→          throw new Error(data?.error ?? "the atelier stayed quiet");
   280→        }
   281→        const artifact = data.artifact;
   282→        visualContextRef.current = {
   283→          subject: artifact.subject,
   284→          mode: artifact.mode,
   285→        };
   286→        setVisual({ state: "ready", request, artifact });
   287→      } catch {
   288→        setVisual((prev) =>
   289→          prev ? { ...prev, state: "error" } : { state: "error", request }
   290→        );
   291→      } finally {
   292→        visualBusyRef.current = false;
   293→      }
   294→    },
   295→    [language, record]
   296→  );
   297→
   298→  /* ---------- composer ---------- */
   299→
   300→  const sendText = (v: string) => {
   301→    if (seeking) return;
   302→    const val = v.trim();
   303→    if (!val && attachments.length === 0) return;
   304→    const carried = attachments.length > 0 ? attachments : undefined;
   305→    /* No button — a request to see simply is one. Ordinary words
   306→       never wake the atelier. */
   307→    const intent = detectVisualIntent(val);
   308→    const wantsVisual =
   309→      val &&
   310→      (intent.direct ||
   311→        (intent.followUp && visualContextRef.current !== null));
   312→    setDraft("");
   313→    setAttachments([]);
   314→    if (wantsVisual) {
   315→      void requestVisualization(val);
   316→    } else {
   317→      void seek(val || null, Boolean(record), carried);
   318→    }
   319→  };
   320→
   321→  const send = (e: FormEvent) => {
   322→    e.preventDefault();
   323→    sendText(draft);
   324→  };
   325→
   326→  /* the voice becomes words on the desk — visible for a breath — then
   327→     the resonance is sought on its own */
   328→  const sendTextRef = useRef(sendText);
   329→  sendTextRef.current = sendText;
   330→  const draftRef = useRef(draft);
   331→  draftRef.current = draft;
   332→  const handleVoiceSubmit = (text: string) => {
   333→    const prev = draftRef.current.trim();
   334→    const finalText = prev ? `${prev} ${text}` : text;
   335→    setDraft(finalText);
   336→    window.setTimeout(() => sendTextRef.current(finalText), 650);
   337→  };
   338→
   339→  const listening = voiceState === "playing";
   340→  /* once a record lies open on the desk, everything written becomes a reply */
   341→  const replying = Boolean(record) && !seeking;
   342→  /* the reveal: while a record's text lies open before the visitor, the
   343→     room recedes — the wall of light codes dims so the parchment may
   344→     draw near (the sheet itself zooms gently toward the reader). */
   345→  const revealed = replying;
   346→
   347→  return (
   348→    <div
   349→      className="akashic-world relative flex h-full flex-col overflow-hidden"
   350→      data-testid="akashic-view"
   351→    >
   352→      {/* ---------- the wall of light codes ---------- */}
   353→      <div aria-hidden="true" className="akashic-veil absolute inset-0" />
   354→      <div
   355→        aria-hidden="true"
   356→        className="animate-drift-a pointer-events-none absolute -left-28 top-[-8%] size-[380px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,var(--gd)_14%,transparent),transparent_65%)] blur-3xl"
   357→      />
   358→      <div
   359→        aria-hidden="true"
   360→        className="animate-drift-c pointer-events-none absolute -right-24 bottom-[-10%] size-[340px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,#8a8a89_12%,transparent),transparent_65%)] blur-3xl"
   361→      />
   362→
   363→      {/* the reveal veil — rises only when a record's text is open, deepening
   364→          the room so the parchment stands out and reads easier */}
   365→      <div
   366→        aria-hidden="true"
   367→        data-testid="akashic-dimveil"
   368→        className={cn(
   369→          "pointer-events-none absolute inset-0 z-[5] bg-[#0a0a0b] transition-opacity duration-[1200ms] ease-out",
   370→          revealed ? "opacity-[0.74]" : "opacity-0"
   371→        )}
   372→      />
   373→
   374→      {/* ---------- threshold: return · the name · a new record ---------- */}
   375→      <header className="relative z-20 flex items-center gap-2 border-b border-[color-mix(in_srgb,var(--gd)_20%,transparent)] bg-[color-mix(in_srgb,#0d0d0e_66%,transparent)] px-3 py-2.5 backdrop-blur-md sm:px-5">
   376→        <button
   377→          type="button"
   378→          onClick={exitAkashic}
   379→          data-testid="akashic-return"
   380→          aria-label={t("Return from the Library")}
   381→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--gd)_30%,transparent)] text-[12px] text-foreground/85 transition-all duration-300 hover:-translate-x-px hover:border-[var(--hairline-hover)] hover:text-foreground sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   382→        >
   383→          <ArrowLeft
   384→            className="size-3.5 transition-transform duration-300 group-hover:-translate-x-0.5"
   385→            aria-hidden="true"
   386→          />
   387→          <span className="hidden sm:inline">{t("Return")}</span>
   388→        </button>
   389→
   390→        <div className="flex min-w-0 flex-1 flex-col items-center justify-center gap-0.5">
   391→          <span className="flex items-center gap-2">
   392→            <span className="akashic-halo flex size-6 shrink-0 items-center justify-center rounded-full border border-[color-mix(in_srgb,var(--gd)_45%,transparent)] bg-[color-mix(in_srgb,var(--gd)_10%,transparent)]">
   393→              <Scroll className="size-3 text-[var(--gd)]" aria-hidden="true" />
   394→            </span>
   395→            <h1 className="font-ancient min-w-0 truncate text-[15px] font-semibold tracking-[0.14em] text-[var(--gd)] sm:text-[17px]">
   396→              {t("The Akashic Library")}
   397→            </h1>
   398→          </span>
   399→          <span className="mono-label hidden text-[9px] uppercase tracking-[0.26em] text-muted-foreground/70 sm:block">
   400→            {t("resonance · records · the ancient one")}
   401→          </span>
   402→        </div>
   403→
   404→        <button
   405→          type="button"
   406→          onClick={() => void seek(null, false)}
   407→          disabled={seeking}
   408→          data-testid="akashic-new"
   409→          aria-label={t("New record")}
   410→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--gd)_30%,transparent)] text-[12px] text-foreground/85 transition-all duration-300 hover:-translate-y-px hover:border-[var(--hairline-hover)] hover:text-foreground disabled:cursor-not-allowed disabled:opacity-40 sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   411→        >
   412→          <RotateCcw
   413→            className="size-3.5 transition-transform duration-500 group-hover:-rotate-180"
   414→            aria-hidden="true"
   415→          />
   416→          <span className="hidden sm:inline">{t("New record")}</span>
   417→        </button>
   418→      </header>
   419→
   420→      {/* ---------- the reading room: one papyrus, one record ---------- */}
   421→      <div className="nice-scroll relative z-10 flex-1 overflow-y-auto">
   422→        <div className="mx-auto flex w-full max-w-[780px] flex-col px-3 pb-10 pt-5 sm:px-6 sm:pt-7">
   423→          <AnimatePresence mode="wait">
   424→            {/* the unopened shelf — an invitation, before the first record */}
   425→            {!record && !seeking && !error && (
   426→              <motion.div
   427→                key="invitation"
   428→                initial={{ opacity: 0, y: 14 }}
   429→                animate={{ opacity: 1, y: 0 }}
   430→                exit={{ opacity: 0, y: -8 }}
   431→                transition={{ duration: 0.7, ease: [0.22, 1, 0.36, 1] }}
   432→                className="papyrus papyrus-frame relative mx-auto w-full rounded-2xl px-6 py-12 text-center sm:px-12 sm:py-16"
   433→                data-testid="akashic-invitation"
   434→              >
   435→                <span
   436→                  aria-hidden="true"
   437→                  className="ink-gold mb-5 block text-[22px] leading-none"
   438→                >
   439→                  ❧
   440→                </span>
   441→                <p className="ink-hand mx-auto max-w-[460px] text-[18px] leading-[1.9] sm:text-[20px]">
   442→                  {t(
   443→                    "Set down a resonance — a name, a question, a feeling — and the Librarian will draw out the record it belongs to. Or leave the desk silent, and the Library will choose for you."
   444→                  )}
   445→                </p>
   446→                <button
   447→                  type="button"
   448→                  onClick={() => void seek(null, false)}
   449→                  data-testid="akashic-receive"
   450→                  className="papyrus-btn focus-glow mt-8 inline-flex h-12 items-center gap-2.5 rounded-full px-7 text-[13px] font-semibold tracking-[0.1em]"
   451→                >
   452→                  <Feather className="size-4" aria-hidden="true" />
   453→                  {t("Receive a record")}
   454→                </button>
   455→              </motion.div>
   456→            )}
   457→
   458→            {/* the ink forming — the Librarian walks the shelves */}
   459→            {seeking && (
   460→              <motion.div
   461→                key="seeking"
   462→                initial={{ opacity: 0 }}
   463→                animate={{ opacity: 1 }}
   464→                exit={{ opacity: 0 }}
   465→                transition={{ duration: 0.5 }}
   466→                className="papyrus papyrus-frame relative mx-auto w-full rounded-2xl px-6 py-12 sm:px-12"
   467→                aria-live="polite"
   468→                aria-busy="true"
   469→                data-testid="akashic-seeking"
   470→              >
   471→                <span
   472→                  aria-hidden="true"
   473→                  className="ink-gold mb-6 block text-center text-[22px] leading-none"
   474→                >
   475→                  ❧
   476→                </span>
   477→                <div className="mx-auto flex max-w-[480px] flex-col gap-3.5">
   478→                  {[86, 70, 92, 60, 78].map((w, i) => (
   479→                    <div
   480→                      key={i}
   481→                      className="ink-shimmer h-[11px] rounded-full"
   482→                      style={{ width: `${w}%`, animationDelay: `${i * 0.16}s` }}
   483→                    />
   484→                  ))}
   485→                </div>
   486→                <p className="ink-hand ink-faint mt-8 text-center text-[16px] italic">
   487→                  {replyAttempt
   488→                    ? t("the Librarian weighs your reply...")
   489→                    : t("the Librarian is fetching your record...")}
   490→                </p>
   491→              </motion.div>
   492→            )}
   493→
   494→            {/* when the shelves keep their silence */}
   495→            {error && !seeking && (
   496→              <motion.div
   497→                key="error"
   498→                initial={{ opacity: 0 }}
   499→                animate={{ opacity: 1 }}
   500→                className="papyrus papyrus-frame relative mx-auto w-full rounded-2xl px-6 py-12 text-center sm:px-12"
   501→                data-testid="akashic-error"
   502→              >
   503→                <p className="ink-hand mx-auto max-w-[420px] text-[17px] leading-[1.85]">
   504→                  {t(
   505→                    "The shelf is momentarily quiet. Rest a breath, then ask the Library again."
   506→                  )}
   507→                </p>
   508→                <button
   509→                  type="button"
   510→                  onClick={() =>
   511→                    void seek(draft.trim() || null, Boolean(record))
   512→                  }
   513→                  data-testid="akashic-retry"
   514→                  className="papyrus-btn focus-glow mt-7 inline-flex h-11 items-center gap-2 rounded-full px-6 text-[12.5px] font-semibold tracking-[0.08em]"
   515→                >
   516→                  <Feather className="size-4" aria-hidden="true" />
   517→                  {replyAttempt
   518→                    ? t("Set the reply down again")
   519→                    : t("Ask again, softly")}
   520→                </button>
   521→              </motion.div>
   522→            )}
   523→
   524→            {/* the record itself — ink on papyrus. As the text is revealed
   525→                the sheet zooms gently toward the reader (anchored at the
   526→                title edge) so the old hand reads easier. */}
   527→            {record && !seeking && (
   528→              <motion.article
   529→                key="record"
   530→                initial={{ opacity: 0, y: 16, scale: 1 }}
   531→                animate={{ opacity: 1, y: 0, scale: 1.035 }}
   532→                transition={{
   533→                  duration: 0.8,
   534→                  ease: [0.22, 1, 0.36, 1],
   535→                  scale: { duration: 1.6, ease: [0.22, 1, 0.36, 1], delay: 0.15 },
   536→                }}
   537→                style={{ transformOrigin: "50% 0%" }}
   538→                className="papyrus papyrus-frame relative mx-auto w-full rounded-2xl px-6 py-9 sm:px-12 sm:py-12"
   539→                data-testid="akashic-record"
   540→              >
   541→                {/* corner light codes */}
   542→                <span
   543→                  aria-hidden="true"
   544→                  className="ink-gold pointer-events-none absolute left-3 top-2.5 text-[11px] opacity-70"
   545→                >
   546→                  ◆
   547→                </span>
