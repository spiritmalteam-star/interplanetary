   218→function OuterRealmsSection() {
   219→  const openRegister = useMirror((s) => s.openRegister);
   220→  const search = useMirror((s) => s.search);
   221→  const setSearch = useMirror((s) => s.setSearch);
   222→  const t = useT();
   223→  const [open, setOpen] = useState(false);
   224→  const [book, setBook] = useState<BookId | null>(null);
   225→
   226→  const listRef = useRef<HTMLUListElement | null>(null);
   227→  const [scrollState, setScrollState] = useState({
   228→    scrollable: false,
   229→    atEnd: true,
   230→  });
   231→
   232→  const measure = useCallback(() => {
   233→    const el = listRef.current;
   234→    if (!el) return;
   235→    const scrollable = el.scrollHeight > el.clientHeight + 8;
   236→    const atEnd =
   237→      el.scrollTop + el.clientHeight >= el.scrollHeight - 14 || !scrollable;
   238→    setScrollState((prev) =>
   239→      prev.scrollable === scrollable && prev.atEnd === atEnd
   240→        ? prev
   241→        : { scrollable, atEnd }
   242→    );
   243→  }, []);
   244→
   245→  useEffect(() => {
   246→    if (!open || !book) return;
   247→    measure();
   248→    const el = listRef.current;
   249→    if (!el || typeof ResizeObserver === "undefined") return;
   250→    const ro = new ResizeObserver(measure);
   251→    ro.observe(el);
   252→    for (const child of Array.from(el.children)) ro.observe(child);
   253→    return () => ro.disconnect();
   254→  }, [measure, open, book]);
   255→
   256→  useEffect(() => {
   257→    if (!open || !book) return;
   258→    const id = window.setTimeout(measure, 60);
   259→    return () => window.clearTimeout(id);
   260→  }, [measure, open, book, search]);
   261→
   262→  const source =
   263→    book === "civilizations"
   264→      ? civilizations
   265→      : book === "interdim"
   266→        ? interdimensional
   267→        : null;
   268→
   269→  const kind = book === "civilizations" ? "civilization" : "interdim";
   270→
   271→  const filtered = useMemo(() => {
   272→    if (!source) return [];
   273→    const q = search.trim().toLowerCase();
   274→    if (!q) return source;
   275→    return source.filter(
   276→      (entry) =>
   277→        entry.name.toLowerCase().includes(q) ||
   278→        entry.origin.toLowerCase().includes(q)
   279→    );
   280→  }, [search, source]);
   281→
   282→  const filteredSpecies = useMemo(() => {
   283→    if (book !== "innerearth") return [];
   284→    const q = search.trim().toLowerCase();
   285→    if (!q) return innerEarth;
   286→    return innerEarth.filter(
   287→      (s) =>
   288→        s.name.toLowerCase().includes(q) ||
   289→        s.hall.toLowerCase().includes(q)
   290→    );
   291→  }, [search, book]);
   292→
   293→  const entityMatches = useMemo(() => {
   294→    if (!search.trim() || !book || book === "innerearth") return [];
   295→    return searchEntities(kind, search, 20);
   296→  }, [search, kind, book]);
   297→
   298→  const emptyLabel =
   299→    book === "civilizations"
   300→      ? t("No matching civilizations found.")
   301→      : book === "interdim"
   302→        ? t("No matching interdimensional categories found.")
   303→        : t("No matching inner earth peoples found.");
   304→
   305→  const books: {
   306→    id: BookId;
   307→    icon: typeof Users;
   308→    label: string;
   309→    count: number;
   310→  }[] = [
   311→    {
   312→      id: "civilizations",
   313→      icon: Users,
   314→      label: t("Civilizations"),
   315→      count: archiveTotals.civilizations,
   316→    },
   317→    {
   318→      id: "interdim",
   319→      icon: Sparkles,
   320→      label: t("Interdimensional"),
   321→      count: archiveTotals.interdim,
   322→    },
   323→    {
   324→      id: "innerearth",
   325→      icon: Mountain,
   326→      label: t("Inner Earth"),
   327→      count: archiveTotals.innerearth,
   328→    },
   329→  ];
   330→
   331→  return (
   332→    <div className="min-h-0 shrink-0 border-t hairline">
   333→      {/* the one quiet door of the library */}
   334→      <button
   335→        type="button"
   336→        onClick={() => setOpen((o) => !o)}
   337→        aria-expanded={open}
   338→        data-testid="encyclopedia-toggle"
   339→        className="focus-glow group flex w-full items-center gap-2.5 px-4 py-3 text-left transition-colors duration-200 hover:text-foreground"
   340→      >
   341→        <BookOpenText
   342→          className="size-4 shrink-0 text-muted-foreground transition-colors duration-200 group-hover:text-foreground"
   343→          aria-hidden="true"
   344→        />
   345→        <span className="mono-label min-w-0 flex-1 truncate text-[12.5px] font-semibold text-foreground/85">
   346→          {t("Outer Realms")}
   347→        </span>
   348→        <ChevronDown
   349→          className={cn(
   350→            "size-3.5 shrink-0 text-muted-foreground/70 transition-transform duration-300",
   351→            open && "rotate-180"
   352→          )}
   353→          aria-hidden="true"
   354→        />
   355→      </button>
   356→
   357→      {open && (
   358→        <div className="min-h-0 pb-2">
   359→          {/* one shared search for every book */}
   360→          <div className="px-3.5 pb-2">
   361→            <div className="relative">
   362→              <Search
   363→                className="pointer-events-none absolute left-3 top-1/2 size-3.5 -translate-y-1/2 text-muted-foreground/70"
   364→                aria-hidden="true"
   365→              />
   366→              <input
   367→                type="search"
   368→                value={search}
   369→                onChange={(e) => setSearch(e.target.value)}
   370→                placeholder={t("Search civilizations, beings, origins...")}
   371→                aria-label={t("Search civilizations, beings, origins")}
   372→                className="focus-glow h-9 w-full rounded-lg border hairline bg-transparent pl-9 pr-8 text-[13.5px] text-foreground placeholder:text-muted-foreground/60 focus:outline-none"
   373→              />
   374→              {search && (
   375→                <button
   376→                  type="button"
   377→                  onClick={() => setSearch("")}
   378→                  aria-label={t("Clear search")}
   379→                  className="absolute right-2.5 top-1/2 -translate-y-1/2 text-muted-foreground/70 transition-colors hover:text-foreground"
   380→                >
   381→                  <X className="size-3.5" aria-hidden="true" />
   382→                </button>
   383→              )}
   384→            </div>
   385→          </div>
   386→
   387→          {/* the books — dropdown buttons, one open at a time */}
   388→          <div className="flex flex-col gap-1 px-3">
   389→            {books.map((b) => {
   390→              const isOpen = book === b.id;
   391→              const Icon = b.icon;
   392→              return (
   393→                <div
   394→                  key={b.id}
   395→                  className="overflow-hidden rounded-xl border hairline bg-[var(--glass-bg-soft)]"
   396→                >
   397→                  <button
   398→                    type="button"
   399→                    onClick={() => setBook(isOpen ? null : b.id)}
   400→                    aria-expanded={isOpen}
   401→                    data-testid={`book-${b.id}`}
   402→                    className="focus-glow flex w-full items-center gap-2 px-2.5 py-2 text-left transition-colors duration-200 hover:bg-[color-mix(in_srgb,var(--cy)_7%,transparent)]"
   403→                  >
   404→                    <Icon
   405→                      className="size-3.5 shrink-0 text-muted-foreground"
   406→                      aria-hidden="true"
   407→                    />
   408→                    <span className="min-w-0 flex-1 truncate text-[13px] font-medium text-foreground/80">
   409→                      {b.label}
   410→                    </span>
   411→                    <span className="shrink-0 font-mono text-[11px] tabular-nums text-muted-foreground/70">
   412→                      {b.count}
   413→                    </span>
   414→                    <ChevronDown
   415→                      className={cn(
   416→                        "size-3.5 shrink-0 text-muted-foreground/60 transition-transform duration-300",
   417→                        isOpen && "rotate-180"
   418→                      )}
   419→                      aria-hidden="true"
   420→                    />
   421→                  </button>
   422→
   423→                  {isOpen && (
   424→                    <div className="relative border-t hairline">
   425→                      <ul
   426→                        ref={listRef}
   427→                        onScroll={measure}
   428→                        className={cn(
   429→                          "nice-scroll archive-scroll max-h-[30vh] space-y-0.5 overflow-y-auto px-1.5 py-1.5",
   430→                          scrollState.scrollable &&
   431→                            !scrollState.atEnd &&
   432→                            "is-scrollable"
   433→                        )}
   434→                        aria-live="polite"
   435→                        data-testid="archive-list"
   436→                      >
   437→                        {entityMatches.length > 0 && (
   438→                          <li className="px-2.5 pb-1 pt-1">
   439→                            <span className="mono-label text-[10.5px] text-muted-foreground/70">
   440→                              {t("Named representatives")}
   441→                            </span>
   442→                          </li>
   443→                        )}
   444→                        {entityMatches.map((e) => (
   445→                          <EntityResultRow
   446→                            key={e.id}
   447→                            entityId={e.id}
   448→                            name={e.name}
   449→                            kind={kind}
   450→                          />
   451→                        ))}
   452→                        {b.id === "innerearth" &&
   453→                          filteredSpecies.map((s) => (
   454→                            <SpeciesRow key={s.id} species={s} />
   455→                          ))}
   456→
   457→                        {b.id !== "innerearth" &&
   458→                          filtered.map((entry) => (
   459→                            <Row key={entry.id} entry={entry} />
   460→                          ))}
   461→
   462→                        {b.id === "innerearth"
   463→                          ? filteredSpecies.length === 0 && (
   464→                              <li className="px-2.5 py-3 text-[13.5px] italic leading-relaxed text-muted-foreground/80">
   465→                                {emptyLabel}
   466→                              </li>
   467→                            )
   468→                          : filtered.length === 0 &&
   469→                            entityMatches.length === 0 && (
   470→                              <li className="px-2.5 py-3 text-[13.5px] italic leading-relaxed text-muted-foreground/80">
   471→                                {emptyLabel}
   472→                              </li>
   473→                            )}
   474→
   475→                        {entityMatches.length >= 20 && (
   476→                          <li className="px-2.5 pb-2 pt-1">
   477→                            <button
   478→                              type="button"
   479→                              onClick={() => openRegister(kind)}
   480→                              className="focus-glow mono-label text-[11px] text-muted-foreground transition-colors hover:text-foreground"
   481→                            >
   482→                              {t(
   483→                                "Showing first {n} matches — open the full register to search every name →",
   484→                                { n: 20 }
   485→                              )}
   486→                            </button>
   487→                          </li>
   488→                        )}
   489→                      </ul>
   490→
   491→                      {scrollState.scrollable && !scrollState.atEnd && (
   492→                        <div
   493→                          aria-hidden="true"
   494→                          className="pointer-events-none absolute inset-x-0 bottom-0 h-10 bg-gradient-to-t from-[color-mix(in_srgb,var(--background)_88%,transparent)] to-transparent"
   495→                        />
   496→                      )}
   497→                    </div>
   498→                  )}
   499→                </div>
   500→              );
   501→            })}
   502→          </div>
   503→        </div>
   504→      )}
   505→    </div>
   506→  );
   507→}
