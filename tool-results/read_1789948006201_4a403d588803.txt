    84→/* ---------- parsing the transmission into rich blocks ---------- */
    85→
    86→type Block =
    87→  | { kind: "opening"; text: string }
    88→  | { kind: "para"; text: string }
    89→  | { kind: "list"; items: string[] }
    90→  | { kind: "signature"; text: string };
    91→
    92→function parseBlocks(text: string): Block[] {
    93→  const raw = text
    94→    .split(/\n{2,}/)
    95→    .map((p) => p.trim())
    96→    .filter(Boolean);
    97→  const blocks: Block[] = [];
    98→
    99→  raw.forEach((p, i) => {
   100→    const isLast = i === raw.length - 1;
   101→    if (isLast && (p.startsWith("—") || p.startsWith("—"))) {
   102→      blocks.push({ kind: "signature", text: p });
   103→      return;
   104→    }
   105→    const lines = p.split("\n").map((l) => l.trim());
   106→    const isList =
   107→      lines.length > 0 &&
   108→      lines.every((l) => l.startsWith("- ") || l.startsWith("• "));
   109→    if (isList) {
   110→      blocks.push({
   111→        kind: "list",
   112→        items: lines.map((l) => l.replace(/^[-•]\s*/, "")),
   113→      });
   114→      return;
   115→    }
   116→    if (i === 0) {
   117→      blocks.push({ kind: "opening", text: p });
   118→      return;
   119→    }
   120→    blocks.push({ kind: "para", text: p });
   121→  });
   122→
   123→  return blocks;
   124→}

   396→function Exchange({
   397→  message,
   398→  index,
   399→  animate,
   400→  scope,
   401→}: {
   402→  message: ChatMessage;
   403→  index: number;
   404→  animate: boolean;
   405→  scope: Scope;
   406→}) {
   407→  const scopeMeta = SCOPE_META[scope];
   408→  const [copied, setCopied] = useState(false);
   409→  const t = useT();
   410→  const language = useMirror((s) => s.language);
   411→
   412→  const classification =
   413→    CLASSIFICATION_META[message.classification] ?? {
   414→      label: "Archive reflection",
   415→      tone: "a" as const,
   416→      note: "Held gently by the archive — verify inwardly what resonates.",
   417→    };
   418→
   419→  const stamp = useMemo(() => {
   420→    const d = new Date(message.createdAt);
   421→    if (Number.isNaN(d.getTime())) return null;
   422→    return d.toLocaleString(LOCALES[language] ?? "en-US", {
   423→      hour: "2-digit",
   424→      minute: "2-digit",
   425→      month: "short",
   426→      day: "numeric",
   427→    });
   428→  }, [message.createdAt, language]);
   429→
   430→  const txId = useMemo(() => {
   431→    let h = 0;
   432→    for (const ch of message.createdAt + message.query) {
   433→      h = (h * 31 + ch.charCodeAt(0)) >>> 0;
   434→    }
   435→    return `TX-${h.toString(16).toUpperCase().padStart(6, "0").slice(0, 6)}`;
   436→  }, [message]);
   437→
   438→  const handleCopy = async () => {
   439→    try {
   440→      await navigator.clipboard.writeText(message.text);
   441→      setCopied(true);
   442→      window.setTimeout(() => setCopied(false), 2200);
   443→    } catch {
   444→      /* clipboard unavailable — quiet fail */
   445→    }
   446→  };
   447→
   448→  const item = animate ? staggerItem : stillItem;
   449→
   450→  return (
   451→    <motion.article
   452→      initial={{ opacity: animate ? 0 : 1, y: animate ? 12 : 0 }}
   453→      animate={{ opacity: 1, y: 0 }}
   454→      transition={{ duration: animate ? 0.55 : 0, ease: [0.22, 1, 0.36, 1] }}
   455→      aria-label={t("Transmission {n}", { n: index + 1 })}
   456→      className="pt-8 first:pt-0"
   457→    >
   458→      {/* index ribbon */}
   459→      <div className="mb-4 flex items-center gap-3">
   460→        <span
   461→          className="mono-label rounded-full border px-2 py-0.5 text-[9px]"
   462→          style={{
   463→            borderColor: "color-mix(in srgb, var(--scope-a) 30%, transparent)",
   464→            color: "color-mix(in srgb, var(--scope-a) 85%, white)",
   465→          }}
   466→        >
   467→          {t("{scope} · exchange {n}", {
   468→            scope: t(scopeMeta.label),
   469→            n: String(index + 1).padStart(2, "0"),
   470→          })}
   471→        </span>
   472→        <span
   473→          className="h-px flex-1"
   474→          style={{
   475→            background:
   476→              "linear-gradient(90deg, color-mix(in srgb, var(--scope-a) 24%, transparent), transparent)",

   478→          aria-hidden="true"
   479→        />

   482→      {/* query echo */}

   486→          style={{ color: "var(--scope-a)" }}
   487→          aria-hidden="true"

   591→  const activeMode = useMirror((s) => s.activeMode);
   592→  const session = useMirror((s) => s.sessions[s.activeMode]);
   593→  const returnToObservatory = useMirror((s) => s.returnToObservatory);
   594→  const clearChannel = useMirror((s) => s.clearChannel);
   595→  const t = useT();
   596→
   597→  const scope: Scope = activeMode;
   598→  const meta = SCOPE_META[scope];
   599→
   600→  /* Auto-focus: the view follows every new generation so the visitor
   601→     never has to scroll down manually. */
   602→  const bottomRef = useRef<HTMLDivElement | null>(null);
   603→  const messagesLength = session.messages.length;
   604→  const status = session.status;
   605→
   606→  useEffect(() => {
   607→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   608→  }, [messagesLength, status]);
   609→
   610→  const handleClear = () => {
   611→    clearChannel();
   612→    toast.success(
   613→      t("{scope} channel cleared — this scope is now quiet.", {
   614→        scope: t(meta.label),
   615→      })
   616→    );
   617→  };
   618→
   619→  return (
   620→    <motion.section
   621→      initial={{ opacity: 0, y: 12 }}
   622→      animate={{ opacity: 1, y: 0 }}
   623→      transition={{ duration: 0.55, ease: [0.22, 1, 0.36, 1] }}
   624→      aria-label={`${meta.label} channel — mirror transmissions`}
   625→      className={cn(
   626→        "scope-" + scope,
   627→        "mx-auto w-full max-w-[760px] px-1 pb-6 pt-8 sm:pt-10"
   628→      )}
   629→    >
   630→      {/* ---------- channel header ---------- */}
   631→      <div className="text-center">
   632→        <div className="flex items-center justify-center gap-4">
   633→          <div
   634→            className="hidden h-px w-16 sm:block"
   635→            style={{
   636→              background:
   637→                "linear-gradient(90deg, transparent, color-mix(in srgb, var(--scope-a) 45%, transparent))",
   638→            }}
   639→            aria-hidden="true"
   640→          />
   641→          <Medallion scope={scope} />
   642→          <div
   643→            className="hidden h-px w-16 sm:block"
   644→            style={{
   645→              background:
   646→                "linear-gradient(90deg, color-mix(in srgb, var(--scope-a) 45%, transparent), transparent)",
   647→            }}
   648→            aria-hidden="true"
   649→          />
   650→        </div>
   651→        <div className="mt-3 flex flex-wrap items-center justify-center gap-2">
   652→          <span
   653→            className="mono-label text-[10.5px]"
   654→            style={{ color: "var(--scope-a)" }}
   655→          >
   656→            {t("Mirror Transmission")}
   657→          </span>
   658→          <span
   659→            className="mono-label rounded-full border px-2 py-0.5 text-[9px] text-muted-foreground"
   660→            style={{ borderColor: "color-mix(in srgb, var(--scope-a) 26%, transparent)" }}
   661→          >
   662→            {t("{scope} scope", { scope: t(meta.label) })}
   663→          </span>
   664→          <span className="mono-label inline-flex items-center gap-1 rounded-full border hairline px-2 py-0.5 text-[9px] text-muted-foreground">
   665→            <Radio className="size-2.5" aria-hidden="true" />
   666→            {t("independent channel")}
   667→          </span>
   668→        </div>
   669→        <p className="mono-label mt-1.5 text-[9.5px] text-muted-foreground/70">
   670→          {t(meta.tagline)}
   671→        </p>
   672→        <p className="mx-auto mt-2 max-w-[480px] text-[12px] leading-relaxed text-muted-foreground/70">
   673→          {t(CHANNEL_PROMISE[scope])}
   674→        </p>
   675→      </div>
   676→
   677→      {/* ---------- thread ---------- */}
   678→      {session.messages.length === 0 &&
   679→        session.status === "idle" &&
   680→        !session.error && (
   681→          <div className="scope-frame-card relative mt-8 overflow-hidden rounded-2xl glass px-6 py-10 text-center">
   682→            <span className="scope-corner scope-corner-tl" aria-hidden="true" />
   683→            <span className="scope-corner scope-corner-tr" aria-hidden="true" />
   684→            <span className="scope-corner scope-corner-bl" aria-hidden="true" />
   685→            <span className="scope-corner scope-corner-br" aria-hidden="true" />
   686→            <p
   687→              className="mono-label text-[9.5px]"
   688→              style={{ color: "var(--scope-a)" }}
   689→            >
   690→              {t("{scope} channel", { scope: t(meta.label) })}
   691→            </p>
   692→            <p className="mt-3 text-[15px] font-medium text-foreground/85">
   693→              {t("This channel is quiet.")}
   694→            </p>
   695→            <p className="mx-auto mt-2 max-w-[380px] text-[12.5px] leading-relaxed text-muted-foreground">
   696→              {t(
   697→                "Every scope keeps its own private channel with its own history — nothing carries over from other scopes. Ask from the composer below to open the first transmission of this channel."
   698→              )}
   699→            </p>
   700→          </div>
   701→        )}
   702→
   703→      {session.messages.map((m, i) => (
   704→        <Exchange
   705→          key={m.id}
   706→          message={m}
   707→          index={i}
   708→          scope={scope}
   709→          animate={i === session.messages.length - 1 && session.status !== "loading"}
   710→        />
   711→      ))}
   712→
   713→      {session.status === "loading" && (
   714→        <LoadingTransmission scope={scope} query={session.activeQuery} />
   715→      )}
   716→
   717→      {session.status === "error" && (
   718→        <div className="scope-frame-card mt-8 rounded-2xl glass p-6 text-center">
   719→          <p className="text-[14.5px] leading-relaxed text-foreground/85">
   720→            {t(
   721→              "The field received your question but could not complete the transmission."
   722→            )}
   723→          </p>
   724→          <p className="mt-2 text-[13px] italic text-muted-foreground">
   725→            {session.error}
   726→          </p>
   727→        </div>
   728→      )}
   729→
   730→      {/* ---------- footer actions ---------- */}
   731→      {session.messages.length > 0 && (
   732→        <>
   733→          <p className="mono-label mt-8 text-center text-[9px] text-muted-foreground/60">
   734→            {session.messages.length}{" "}
   735→            {session.messages.length === 1
   736→              ? t("exchange held in this channel")
   737→              : t("exchanges held in this channel")}{" "}
   738→            · {t("Free will honored always · Transmitted with love ❤️")}
   739→          </p>
   740→          <div className="mt-4 flex flex-wrap items-center justify-center gap-2.5">
   741→            <button
   742→              type="button"
   743→              onClick={returnToObservatory}
   744→              className="focus-glow group flex items-center gap-2 rounded-full border hairline px-4 py-2 text-[12.5px] font-medium text-muted-foreground transition-all duration-300 hover:border-[var(--hairline-hover)] hover:text-foreground"
   745→            >
   746→              <ArrowLeft
   747→                className="size-3.5 transition-transform duration-300 group-hover:-translate-x-0.5"
   748→                aria-hidden="true"
   749→              />
   750→              {t("Return to the Observatory")}
   751→            </button>
   752→            <button
   753→              type="button"
   754→              onClick={handleClear}
   755→              className="focus-glow group flex items-center gap-2 rounded-full border hairline px-4 py-2 text-[12.5px] font-medium text-muted-foreground transition-all duration-300 hover:border-[var(--hairline-hover)] hover:text-foreground"
   756→            >
   757→              <Eraser
   758→                className="size-3.5 transition-transform duration-300 group-hover:rotate-12"
   759→                aria-hidden="true"
   760→              />
   761→              {t("Clear the {scope} channel", { scope: t(meta.label) })}
   762→            </button>
   763→          </div>
   764→        </>
   765→      )}
   766→
   767→      {/* auto-scroll anchor — the view follows each new generation */}
   768→      <div ref={bottomRef} aria-hidden="true" className="h-px" />
   769→    </motion.section>
   770→  );
   771→}
