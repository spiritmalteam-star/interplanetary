   392→   392→export function SidebarContent() {
   393→   393→  const sidebarTab = useMirror((s) => s.sidebarTab);
   394→   394→  const setSidebarTab = useMirror((s) => s.setSidebarTab);
   395→   395→  const openRegister = useMirror((s) => s.openRegister);
   396→   396→  const search = useMirror((s) => s.search);
   397→   397→  const setSearch = useMirror((s) => s.setSearch);
   398→   398→  const setMobileNavOpen = useMirror((s) => s.setMobileNavOpen);
   399→   399→  const t = useT();
   400→   400→
   401→   401→  /* ---------- scroll indicator state for the archive list ---------- */
   402→   402→  const listRef = useRef<HTMLUListElement | null>(null);
   403→   403→  const [scrollState, setScrollState] = useState({
   404→   404→    scrollable: false,
   405→   405→    atEnd: true,
   406→   406→  });
   407→   407→
   408→   408→  const measure = useCallback(() => {
   409→   409→    const el = listRef.current;
   410→   410→    if (!el) return;
   411→   411→    const scrollable = el.scrollHeight > el.clientHeight + 8;
   412→   412→    const atEnd =
   413→   413→      el.scrollTop + el.clientHeight >= el.scrollHeight - 14 || !scrollable;
   414→   414→    setScrollState((prev) =>
   415→   415→      prev.scrollable === scrollable && prev.atEnd === atEnd
   416→   416→        ? prev
   417→   417→        : { scrollable, atEnd }
   418→   418→    );
   419→   419→  }, []);
   420→   420→
   421→   421→  useEffect(() => {
   422→   422→    measure();
   423→   423→    const el = listRef.current;
   424→   424→    if (!el || typeof ResizeObserver === "undefined") return;
   425→   425→    const ro = new ResizeObserver(measure);
   426→   426→    ro.observe(el);
   427→   427→    for (const child of Array.from(el.children)) ro.observe(child);
   428→   428→    return () => ro.disconnect();
   429→   429→  }, [measure]);
   430→   430→
   431→   431→  const onScroll = useCallback(() => measure(), [measure]);
   432→   432→
   433→   433→  /* re-measure whenever the archive tab or the query reshapes the list */
   434→   434→  useEffect(() => {
   435→   435→    const id = window.setTimeout(measure, 60);
