    26→/* ---------- parsing the transmission into rich blocks ---------- */
    27→
    28→type Block =
    29→  | { kind: "opening"; text: string }
    30→  | { kind: "para"; text: string }
    31→  | { kind: "list"; items: string[] }
    32→  | { kind: "signature"; text: string };
    33→
    34→function parseBlocks(text: string): Block[] {
    35→  const raw = text
    36→    .split(/\n{2,}/)
    37→    .map((p) => p.trim())
    38→    .filter(Boolean);
    39→  const blocks: Block[] = [];
    40→
    41→  raw.forEach((p, i) => {
    42→    const isLast = i === raw.length - 1;
    43→    if (isLast && (p.startsWith("—") || p.startsWith("—"))) {
    44→      blocks.push({ kind: "signature", text: p });
    45→      return;
    46→    }
    47→    const lines = p.split("\n").map((l) => l.trim());
    48→    const isList =
    49→      lines.length > 0 &&
    50→      lines.every((l) => l.startsWith("- ") || l.startsWith("• "));
    51→    if (isList) {
    52→      blocks.push({
    53→        kind: "list",
    54→        items: lines.map((l) => l.replace(/^[-•]\s*/, "")),
    55→      });
    56→      return;
    57→    }
    58→    if (i === 0) {
    59→      blocks.push({ kind: "opening", text: p });
    60→      return;
    61→    }
    62→    blocks.push({ kind: "para", text: p });
    63→  });
    64→
    65→  return blocks;
    66→}

   330→        </p>
   331→        {message.visualRequest && (
   332→          <button
   333→            type="button"
   334→            onClick={() =>
   335→              void askScopeVisual(activeMode, message.visualRequest ?? "", null)

   338→            className="focus-glow mt-2.5 flex h-9 items-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--scope-a)_35%,transparent)] px-4 text-[13px] text-foreground/90 transition-all duration-300 hover:-translate-y-px"
   339→          >

   342→          </button>
   343→        )}
   344→      </div>
   345→    );
   346→  }
   347→  if (!message.artifact) return null;
   348→  const artifact = message.artifact;
   349→  if (!artifact.imageUrl && artifact.slides.length === 0) {
   350→    return (
   351→      <PreparedPromptFallback
   352→        artifact={artifact}

   354→        testIdPrefix="scope-visual"
   355→        onPaint={() =>
   356→          void askScopeVisual(
   357→            activeMode,
   358→            message.visualRequest ?? artifact.subject,
   359→            {
   360→              id: message.id,
   361→              request: message.visualRequest ?? artifact.subject,
   362→              prompt: artifact.prompt,
   363→              subject: artifact.subject,
   364→              mode: artifact.mode,
   365→            }
   366→          )
   367→        }
   368→      />
   369→    );
   370→  }
   371→  return (
   372→    <VisualizationCard
   373→      artifact={artifact}
   374→      accent="var(--scope-a)"
   375→      testIdPrefix="scope-visual"
   376→      onRegenerate={() =>
   377→        void askScopeVisual(activeMode, message.visualRequest ?? artifact.subject, {
   378→          id: message.id,
   379→          request: message.visualRequest ?? artifact.subject,
   380→          prompt: artifact.prompt,
   381→          subject: artifact.subject,
   382→          mode: artifact.mode,
   383→        })
   384→      }
   385→    />
   386→  );
   387→}

   391→function Exchange({
   392→  message,
   393→  index,
   394→  animate,
   395→  scope,
   396→}: {
   397→  message: ChatMessage;
   398→  index: number;
   399→  animate: boolean;
   400→  scope: Scope;
   401→}) {
   402→  const scopeMeta = SCOPE_META[scope];
   403→  const [copied, setCopied] = useState(false);
   404→  const t = useT();
   405→  /* Each card draws its own light — quietly, never spoken of. */
   406→  const aura = useAura(message.id);
   407→
   408→  const handleCopy = async () => {
   409→    try {
   410→      await navigator.clipboard.writeText(message.text);
   411→      setCopied(true);
   412→      window.setTimeout(() => setCopied(false), 2200);
   413→    } catch {
   414→      /* clipboard unavailable — quiet fail */
   415→    }
   416→  };
   417→
   418→  const item = animate ? staggerItem : stillItem;
   419→  const hasVisual = Boolean(message.artifact || message.visual);
   420→
   421→  return (
   422→    <motion.article
   423→      initial={{ opacity: animate ? 0 : 1, y: animate ? 12 : 0 }}
   424→      animate={{ opacity: 1, y: 0 }}
   425→      transition={{ duration: animate ? 0.55 : 0, ease: [0.22, 1, 0.36, 1] as const }}

   592→        {t("Prepared from what has been spoken here")}

   594→    </motion.div>
   595→  );

   598→/* ---------- main view: one independent channel per scope ---------- */
   599→
   600→export function TransmissionView() {
   601→  const activeMode = useMirror((s) => s.activeMode);
   602→  const session = useMirror((s) => s.sessions[s.activeMode]);
   603→  const remedyStatus = useMirror((s) => s.remedyStatus);
   604→  const t = useT();
   605→
   606→  const scope: Scope = activeMode;
   607→  const meta = SCOPE_META[scope];
   608→
   609→  /* The apothecary serves the concern most recently spoken — the last
   610→     exchange's question, held quietly until the thread rests. */
   611→  const lastSpoken = useMemo(
   612→    () =>
   613→      scope === "healing"
   614→        ? (session.messages.at(-1)?.query ?? null)
   615→        : null,
   616→    [scope, session.messages]
   617→  );
   618→
   619→  /* The thread begins where it begins: opening, reloading or switching
   620→     into a channel keeps the thread at its BEGINNING. New generations
   621→     load from the TOP — the newest exchange settles its first line at
   622→     the top of the view and the visitor reads downward from there.
   623→     One law above all: the moment a suggestion is struck — even after
   624→     the thread already carries conversations — the frame must meet
   625→     THAT exchange, while it forms and again when it lands. */
   626→  const topRef = useRef<HTMLDivElement | null>(null);
   627→  const latestRef = useRef<HTMLDivElement | null>(null);
   628→  const loadingRef = useRef<HTMLDivElement | null>(null);
   629→  const errorRef = useRef<HTMLDivElement | null>(null);
   630→  const messagesLength = session.messages.length;
   631→  const status = session.status;
   632→  const baseline = useRef({
   633→    mode: activeMode,
   634→    len: messagesLength,
   635→    status,
   636→    fresh: true,
   637→  });
   638→
   639→  useEffect(() => {
   640→    const b = baseline.current;
   641→    const sameMode = b.mode === activeMode;
   642→    const grew = sameMode && messagesLength !== b.len;
   643→    const startedLoading =
   644→      sameMode && !grew && status === "loading" && b.status !== "loading";
   645→    const becameError =
   646→      sameMode &&
   647→      !grew &&
   648→      !startedLoading &&
   649→      status === "error" &&
   650→      b.status !== "error";
   651→
   652→    baseline.current = {
   653→      mode: activeMode,
   654→      len: messagesLength,
   655→      status,
   656→      fresh: false,
   657→    };
   658→
   659→    if (b.fresh || !sameMode) {
   660→      /* First sight of the channel — arriving from the observatory the
   661→         transmission may ALREADY be forming (a suggestion was struck
   662→         and the channel mounted mid-reception): meet it where it
   663→         forms, never wait above the older thread. A channel entering
   664→         with a failed transmission meets the error; otherwise it
   665→         rests at its beginning. */
   666→      if (status === "loading") {
   667→        loadingRef.current?.scrollIntoView({ block: "start" });
   668→      } else if (status === "error") {
   669→        errorRef.current?.scrollIntoView({ block: "start" });
   670→      } else if (!sameMode) {
   671→        topRef.current?.scrollIntoView({ block: "start" });
   672→      }
   673→      return;
   674→    }
   675→    if (grew) {
   676→      latestRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   677→      /* the answer must own the frame — re-affirm once the entrance
   678→         has settled, in case the first glide was preempted */
   679→      const id = window.setTimeout(() => {
   680→        latestRef.current?.scrollIntoView({
   681→          behavior: "smooth",
   682→          block: "start",
   683→        });
   684→      }, 450);
   685→      return () => window.clearTimeout(id);
   686→    }
   687→    if (startedLoading) {
   688→      loadingRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   689→      return;
   690→    }
   691→    if (becameError) {
   692→      errorRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
   693→    }
   694→  }, [activeMode, messagesLength, status]);
   695→
   696→  return (
   697→    <motion.section
   698→      initial={{ opacity: 0, y: 12 }}
   699→      animate={{ opacity: 1, y: 0 }}
   700→      transition={{ duration: 0.55, ease: [0.22, 1, 0.36, 1] as const }}
   701→      aria-label={`${meta.label} channel — mirror transmissions`}
   702→      className={cn(
   703→        "scope-" + scope,
   704→        "mx-auto w-full max-w-[760px] px-1 pb-6 pt-6 sm:pt-8"
   705→      )}
   706→    >
   707→      {/* beginning anchor — opening a channel keeps the thread here */}
   708→      <div ref={topRef} aria-hidden="true" className="h-px" />
   709→
   710→      {/* ---------- thread ---------- */}
   711→      {session.messages.length === 0 &&
   712→        session.status === "idle" &&
   713→        !session.error && (
   714→          <div className="scope-frame-card relative mt-2 overflow-hidden rounded-2xl glass px-6 py-9 text-center">
   715→            <span className="scope-corner scope-corner-tl" aria-hidden="true" />
   716→            <span className="scope-corner scope-corner-tr" aria-hidden="true" />
   717→            <span className="scope-corner scope-corner-bl" aria-hidden="true" />
   718→            <span className="scope-corner scope-corner-br" aria-hidden="true" />
   719→            <p className="text-[16.5px] font-medium text-foreground/85">
   720→              {t("This channel is quiet.")}
   721→            </p>
   722→            <p className="mx-auto mt-2 max-w-[380px] text-[14.5px] leading-relaxed text-muted-foreground">
   723→              {t(
   724→                "Every scope keeps its own private channel — ask from the composer below to open the first transmission."
   725→              )}
   726→            </p>
   727→          </div>
   728→        )}
   729→
   730→      {session.messages.map((m, i) => (
   731→        <div
   732→          key={m.id}
   733→          ref={
   734→            i === session.messages.length - 1
   735→              ? (node) => {
   736→                  latestRef.current = node;
   737→                }
   738→              : undefined
   739→          }
   740→        >
   741→          <Exchange
   742→            message={m}
   743→            index={i}
   744→            scope={scope}
   745→            animate={i === session.messages.length - 1 && session.status !== "loading"}
   746→          />
   747→        </div>
   748→      ))}
   749→
   750→      {scope === "healing" &&
   751→        session.status !== "loading" &&
   752→        remedyStatus !== "crafting" &&
   753→        lastSpoken && <ApothecaryStand concern={lastSpoken} />}
   754→
   755→      {session.status === "loading" &&
   756→        session.messages.at(-1)?.visual !== "pending" && (
   757→        <div
   758→          ref={(node) => {
   759→            loadingRef.current = node;
   760→          }}
   761→        >
   762→          <LoadingTransmission query={session.activeQuery} />
   763→        </div>
   764→      )}
   765→
   766→      {session.status === "error" && (
   767→        <div
   768→          ref={errorRef}
   769→          className="scope-frame-card mt-8 rounded-2xl glass p-6 text-center"
   770→        >
   771→          <p className="text-[16px] leading-relaxed text-foreground/85">
   772→            {t(
   773→              "The field received your question but could not complete the transmission."
   774→            )}
   775→          </p>
   776→          <p className="mt-2 text-[14.5px] italic text-muted-foreground">
   777→            {session.error}
   778→          </p>
   779→        </div>
   780→      )}
   781→
   782→      {/* new generations settle at the TOP of this anchor — the reader
   783→          begins at their first line and scrolls down for the rest */}
   784→      <div aria-hidden="true" className="h-px" />
   785→    </motion.section>
   786→  );
   787→}
