    43→/* ------------------------------------------------------------------ */
    44→/*  Per-scope chat channels                                            */
    45→/*  Every scope owns a fully independent channel: its own history,     */
    46→/*  status, in-flight query, error and composer draft. Switching       */
    47→/*  scopes never carries content from one channel into another.        */

    49→
    50→export interface ChatMessage {
    51→  id: string;
    52→  query: string;
    53→  text: string;
    54→  classification: string;
    55→  createdAt: string;
    56→  /** What traveled with the question — one image, up to three documents. */
    57→  attachments?: { images: number; docNames: string[] };
    58→  /* the Universal Visualization Engine — every channel also answers in images */
    59→  artifact?: VisualizationArtifact;
    60→  visual?: "pending" | "error";
    61→  visualRequest?: string;
    62→}
    63→
    64→export interface ScopeSession {
    65→  /** Completed Q/A pairs — the permanent, private history of this channel. */
    66→  messages: ChatMessage[];
    67→  status: TransmissionStatus;
    68→  error: string | null;
    69→  /** Query currently being transmitted (loading state). */
    70→  activeQuery: string;
    71→  /** Composer draft — kept per channel so drafts never leak across scopes. */
    72→  draft: string;
    73→}

   422→  askMirror: async (question, attachments) => {
   423→    const query = question.trim();
   424→    const mode = get().activeMode;
   425→    const session = get().sessions[mode];
   426→    if (!query || session.status === "loading") return;
   427→
   428→    set((s) => ({
   429→      view: "transmission",
   430→      mobileNavOpen: false,
   431→      sessions: {
   432→        ...s.sessions,
   433→        [mode]: {
   434→          ...s.sessions[mode],
   435→          status: "loading",
   436→          activeQuery: query,
   437→          draft: "",
   438→          error: null,
   439→        },
   440→      },
   441→    }));
   442→
   443→    try {
   444→      /* Context memory: the mirror remembers this channel's earlier
   445→         exchanges, so conversations deepen instead of restarting. */
   446→      const history = session.messages.slice(-6).map((m) => ({
   447→        q: m.query,
   448→        a: m.text,
   449→      }));

   452→        method: "POST",

   455→          query,
   456→          mode,
   457→          scienceField: get().activeScienceField,
   458→          direction: get().activeDirection,
   459→          language: get().language,
   460→          history,
   461→          ...(payload ?? {}),
   462→        }),
   463→      });
   464→      const data = await res.json().catch(() => null);
   465→
   466→      if (!res.ok) {
   467→        throw new Error(
   468→          (data && data.error) ||
   469→            "The field is momentarily quiet. Rest, then try again."
   470→        );
   471→      }
   472→

   474→        sessions: {
   475→          ...s.sessions,
   476→          [mode]: {
   477→            ...s.sessions[mode],
   478→            status: "ready",
   479→            activeQuery: "",
   480→            messages: [
   481→              ...s.sessions[mode].messages,
   482→              {
   483→                id: nextMessageId(),
   484→                query,
   485→                text: data.transmission,
   486→                classification: data.classification,
   487→                createdAt: data.createdAt ?? new Date().toISOString(),
   488→                ...(attachments
   489→                  ? {
   490→                      attachments: {
   491→                        images: attachments.filter((a) => a.kind === "image")
   492→                          .length,
   493→                        docNames: attachments
   494→                          .filter((a) => a.kind === "document")
   495→                          .map((a) => a.name),
   496→                      },
   497→                    }
   498→                  : {}),
   499→              },
   500→            ],
   501→          },
   502→        },
   503→      }));
   504→    } catch (err) {
   505→      set((s) => ({
   506→        sessions: {
   507→          ...s.sessions,
   508→          [mode]: {
   509→            ...s.sessions[mode],
   510→            status: "error",
   511→            activeQuery: "",
   512→            error:
   513→              err instanceof Error
   514→                ? err.message
   515→                : "The field is momentarily quiet. Rest, then try again.",
   516→          },
   517→        },
   518→      }));
   519→    }
   520→  },

   522→  /* ------- the scope channels — the visualization engine ------- */

   524→  askScopeVisual: async (mode, question, regenerateOf) => {
   525→    if (get().sessions[mode].status === "loading") return;
   526→    const visualId = regenerateOf ? regenerateOf.id : nextMessageId();
   527→
   528→    set((s) => ({
   529→      view: "transmission",
   530→      mobileNavOpen: false,
   531→      sessions: {
   532→        ...s.sessions,
   533→        [mode]: {
   534→          ...s.sessions[mode],
   535→          status: "loading",
   536→          activeQuery: question,
   537→          draft: "",
   538→          error: null,
   539→          messages: regenerateOf
   540→            ? s.sessions[mode].messages.map((m) =>
   541→                m.id === regenerateOf.id
   542→                  ? { ...m, visual: "pending" as const }
   543→                  : m
   544→              )
   545→            : [
   546→                ...s.sessions[mode].messages,
   547→                {
   548→                  id: visualId,
   549→                  query: question,
   550→                  text: "",
   551→                  classification: "WORLD_BUILDING",
   552→                  createdAt: new Date().toISOString(),
   553→                  visual: "pending" as const,
   554→                  visualRequest: question,
   555→                },
   556→              ],
   557→        },
   558→      },
   559→    }));
   560→
   561→    try {
   562→      const current = get().sessions[mode];
   563→      const history = current.messages
   564→        .filter((m) => m.id !== visualId && !m.visual && !m.artifact && m.text)
   565→        .slice(-6)
   566→        .map((m) => ({ q: m.query, a: m.text }));
   567→      const lastArtifact = [...current.messages]
   568→        .reverse()
   569→        .find((m) => m.artifact)?.artifact;
   570→
   571→      const res = await fetch("/api/visualize", {
   572→        method: "POST",
   573→        headers: { "Content-Type": "application/json" },
   574→        body: JSON.stringify({
   575→          language: get().language,
   576→          message: regenerateOf ? regenerateOf.request ?? question : question,
   577→          history,
   578→          ...(regenerateOf
   579→            ? {
   580→                regenerate: true,
   581→                previousPrompt: regenerateOf.prompt,
   582→                contextSubject: regenerateOf.subject,
   583→                previousMode: regenerateOf.mode,
   584→              }
   585→            : lastArtifact
   586→              ? {
   587→                  contextSubject: lastArtifact.subject,
   588→                  previousMode: lastArtifact.mode,
   589→                }
   590→              : {}),
   591→        }),
   592→      });
   593→      const data = (await res.json().catch(() => null)) as {
   594→        artifact?: VisualizationArtifact;
   595→        painted?: boolean;
   596→        error?: string;
   597→      } | null;
   598→      if (!res.ok || !data?.artifact) {
   599→        throw new Error(
   600→          (data && data.error) ||
   601→            "The atelier is quiet — the vision could not be composed."
   602→        );
   603→      }
   604→
   605→      const artifact = data.artifact;
   606→      set((s) => ({
   607→        sessions: {
   608→          ...s.sessions,
   609→          [mode]: {
   610→            ...s.sessions[mode],
   611→            status: "ready",
   612→            activeQuery: "",
   613→            messages: s.sessions[mode].messages.map((m) =>
   614→              m.id === visualId
   615→                ? { ...m, visual: undefined, artifact }
   616→                : m
   617→            ),
   618→          },
   619→        },
   620→      }));
   621→
   622→      /* the same one silent repaint for the scope channels */
   623→      if (!regenerateOf && !artifact.imageUrl && artifact.slides.length === 0) {
   624→        window.setTimeout(() => {
   625→          const msg = get().sessions[mode].messages.find(
   626→            (m) => m.id === visualId
   627→          );
   628→          if (
   629→            msg?.artifact &&
   630→            !msg.artifact.imageUrl &&
   631→            msg.artifact.slides.length === 0 &&
   632→            !msg.visual
   633→          ) {
   634→            void get().askScopeVisual(mode, msg.visualRequest ?? artifact.subject, {
   635→              id: visualId,
   636→              request: msg.visualRequest ?? artifact.subject,
   637→              prompt: artifact.prompt,
   638→              subject: artifact.subject,
   639→              mode: artifact.mode,
   640→            });
   641→          }

   771→    if (get().osStatus === "loading") return;

   773→
   774→    set((s) => ({

   777→      osDraft: "",

   788→            },

   795→            },

   797→    }));
   798→
   799→    try {
   800→      const history = get()
   801→        .osMessages.filter((m) => !m.visual && m.id !== visualId)
   802→        .slice(-10)

   804→          role: m.role,
   805→          text: m.artifact
   806→            ? `[a visualization was created: "${m.artifact.title}" — ${m.artifact.subject}]`
   807→            : m.text,
   808→        }));
   809→
   810→      const res = await fetch("/api/visualize", {
   811→        method: "POST",
   812→        headers: { "Content-Type": "application/json" },
   813→        body: JSON.stringify({
   814→          language: get().language,
   815→          message: regenerateOf ? regenerateOf.request ?? question : question,
   816→          history,

   819→                regenerate: true,
   820→                previousPrompt: regenerateOf.prompt,
   821→                contextSubject: regenerateOf.subject,
   822→                previousMode: regenerateOf.mode,

   826→                  contextSubject: context.subject,
   827→                  previousMode: context.mode,
   828→                }
   829→              : {}),
   830→        }),
   831→      });
   832→      const data = (await res.json().catch(() => null)) as {
   833→        artifact?: VisualizationArtifact;
   834→        painted?: boolean;
   835→        error?: string;
   836→      } | null;

   838→        throw new Error(
   839→          (data && data.error) ||
   840→            "The atelier is quiet — the vision could not be composed."
   841→        );
   842→      }

   847→        osMessages: s.osMessages.map((m) =>
   848→          m.id === visualId

   860→          if (
   861→            msg?.artifact &&
   862→            !msg.artifact.imageUrl &&
   863→            msg.artifact.slides.length === 0 &&
   864→            !msg.visual

   867→              id: visualId,
   868→              request: msg.visualRequest ?? artifact.subject,
   869→              prompt: artifact.prompt,

   882→      }));
   883→    }
