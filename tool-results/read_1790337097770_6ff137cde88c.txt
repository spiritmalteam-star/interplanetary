    47→    47→/**

    53→    53→export function AkashicView() {
    54→    54→  const exitAkashic = useMirror((s) => s.exitAkashic);
    55→    55→  const language = useMirror((s) => s.language);

    58→    58→  const [record, setRecord] = useState<AkashicRecord | null>(null);

   103→   103→            resonance,

   105→   105→              asReply && record
   106→   106→                ? {
   107→   107→                    title: record.title,
   108→   108→                    era: record.era,

   123→   123→            data.entrance,

   125→   125→          ].slice(0, 4);

   128→   128→          title: data.title ?? "A Record Set Aside",
   129→   129→          era: data.era ?? "inscribed in an age the shelves remember",

   135→   135→      } finally {

   138→   138→      }

   152→   152→      await navigator.clipboard.writeText(recordText);

   157→   157→      toast.error(t("The copying did not take — select and copy by hand."));
   158→   158→    }
   159→   159→  };

   163→   163→    const shareTitle = record ? `Akashic Library — ${record.title}` : "Akashic Library";
   164→   164→    try {
   165→   165→      if (typeof navigator.share === "function") {
   166→   166→        await navigator.share({ title: shareTitle, text: recordText });
   167→   167→        return;
   168→   168→      }
   169→   169→      throw new Error("share unavailable");
   170→   170→    } catch (err) {
   171→   171→      if (err instanceof DOMException && err.name === "AbortError") return;
   172→   172→      try {
   173→   173→        await navigator.clipboard.writeText(recordText);
   174→   174→        toast.success(t("Sharing is quiet here — the record is copied instead."));
   175→   175→      } catch {
   176→   176→        toast.error(t("The copying did not take — select and copy by hand."));
   177→   177→      }
   178→   178→    }

   180→   180→

   188→   188→      audioRef.current?.pause();

   190→   190→      /* already stopped */
   191→   191→    }
   192→   192→    audioRef.current = null;
   193→   193→    setVoiceState("idle");
   194→   194→  }, []);
   195→   195→
   196→   196→  useEffect(() => stopVoice, [stopVoice]);
   197→   197→

   199→   199→    if (voiceState === "loading") return;
   200→   200→    if (voiceState === "playing") {
   201→   201→      stopVoice();
   202→   202→      return;
   203→   203→    }
   204→   204→    if (!record) return;
   205→   205→    /* The Librarian reads aloud himself — the old man's own voice,
   206→   206→       slower than the world, from behind the great desk. */

   209→   209→      const res = await fetch("/api/tts", {
   210→   210→        method: "POST",
   211→   211→        headers: { "Content-Type": "application/json" },

   213→   213→          text: `${record.title}. ${record.era}. ${record.record.replaceAll(
   214→   214→            /\s*\n\s*/g,
   215→   215→            " "
   216→   216→          )} ${record.seal}`,
   217→   217→          voice: "regent",
   218→   218→          pace: 0.82,
   219→   219→        }),

   222→   222→      const blob = await res.blob();

   224→   224→      audioRef.current = audio;
   225→   225→      audio.onended = () => {
   226→   226→        if (audioRef.current === audio) audioRef.current = null;
   227→   227→        setVoiceState("idle");

   230→   230→        if (audioRef.current === audio) audioRef.current = null;
   231→   231→        setVoiceState("idle");

   233→   233→      await audio.play();
   234→   234→      setVoiceState("playing");
   235→   235→    } catch {
   236→   236→      setVoiceState("idle");
   237→   237→      toast.error(t("The Librarian is quiet. Rest, then listen again."));
   238→   238→    }
   239→   239→  }, [record, voiceState, stopVoice, t]);
   240→   240→

   244→   244→  /* ---------------------------------------------------------------- */

   246→   246→    async (request: string) => {

   248→   248→      visualBusyRef.current = true;
   249→   249→      setVisual({ state: "pending", request });

   251→   251→      try {
   252→   252→        const res = await fetch("/api/visualize", {
   253→   253→          method: "POST",
   254→   254→          headers: { "Content-Type": "application/json" },
   255→   255→          body: JSON.stringify({
   256→   256→            language,
   257→   257→            message: request,
   258→   258→            history: record
   259→   259→              ? [
   260→   260→                  {
   261→   261→                    role: "mirror",
   262→   262→                    text: `${record.title}. ${record.era}. ${record.record}`,
   263→   263→                  },
   264→   264→                ]
   265→   265→              : [],
   266→   266→            ...(visualContextRef.current
   267→   267→              ? {
   268→   268→                  contextSubject: visualContextRef.current.subject,
   269→   269→                  previousMode: visualContextRef.current.mode,
   270→   270→                }
   271→   271→              : {}),
   272→   272→          }),
   273→   273→        });

   284→   284→          mode: artifact.mode,

   288→   288→        setVisual((prev) =>
   289→   289→          prev ? { ...prev, state: "error" } : { state: "error", request }
   290→   290→        );
   291→   291→      } finally {
   292→   292→        visualBusyRef.current = false;
   293→   293→      }
   294→   294→    },
   295→   295→    [language, record]
   296→   296→  );
   297→   297→
   298→   298→  /* ---------- composer ---------- */

   318→   318→    }
   319→   319→  };

   321→   321→  const send = (e: FormEvent) => {
   322→   322→    e.preventDefault();
   323→   323→    sendText(draft);
   324→   324→  };

   326→   326→  /* the voice becomes words on the desk — visible for a breath — then
   327→   327→     the resonance is sought on its own */

   329→   329→  sendTextRef.current = sendText;
   330→   330→  const draftRef = useRef(draft);
   331→   331→  draftRef.current = draft;
   332→   332→  const handleVoiceSubmit = (text: string) => {
   333→   333→    const prev = draftRef.current.trim();
   334→   334→    const finalText = prev ? `${prev} ${text}` : text;
   335→   335→    setDraft(finalText);
   336→   336→    window.setTimeout(() => sendTextRef.current(finalText), 650);
   337→   337→  };
   338→   338→
   339→   339→  const listening = voiceState === "playing";
   340→   340→  /* once a record lies open on the desk, everything written becomes a reply */
   341→   341→  const replying = Boolean(record) && !seeking;
   342→   342→
   343→   343→  return (
   344→   344→    <div
   345→   345→      className="akashic-world relative flex h-full flex-col overflow-hidden"
   346→   346→      data-testid="akashic-view"
   347→   347→    >
   348→   348→      {/* ---------- the wall of light codes ---------- */}
   349→   349→      <div aria-hidden="true" className="akashic-veil absolute inset-0" />
   350→   350→      <div
   351→   351→        aria-hidden="true"
   352→   352→        className="animate-drift-a pointer-events-none absolute -left-28 top-[-8%] size-[380px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,var(--gd)_14%,transparent),transparent_65%)] blur-3xl"
   353→   353→      />
   354→   354→      <div
   355→   355→        aria-hidden="true"
   356→   356→        className="animate-drift-c pointer-events-none absolute -right-24 bottom-[-10%] size-[340px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,#b06a2c_12%,transparent),transparent_65%)] blur-3xl"
   357→   357→      />
   358→   358→
   359→   359→      {/* ---------- threshold: return · the name · a new record ---------- */}
   360→   360→      <header className="relative z-20 flex items-center gap-2 border-b border-[color-mix(in_srgb,var(--gd)_20%,transparent)] bg-[color-mix(in_srgb,#0c0806_66%,transparent)] px-3 py-2.5 backdrop-blur-md sm:px-5">
   361→   361→        <button
   362→   362→          type="button"
   363→   363→          onClick={exitAkashic}
   364→   364→          data-testid="akashic-return"
   365→   365→          aria-label={t("Return from the Library")}
   366→   366→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--gd)_30%,transparent)] text-[12px] text-foreground/85 transition-all duration-300 hover:-translate-x-px hover:border-[var(--hairline-hover)] hover:text-foreground sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   367→   367→        >
   368→   368→          <ArrowLeft
   369→   369→            className="size-3.5 transition-transform duration-300 group-hover:-translate-x-0.5"
   370→   370→            aria-hidden="true"
   371→   371→          />
   372→   372→          <span className="hidden sm:inline">{t("Return")}</span>
   373→   373→        </button>
   374→   374→
   375→   375→        <div className="flex min-w-0 flex-1 flex-col items-center justify-center gap-0.5">
   376→   376→          <span className="flex items-center gap-2">
   377→   377→            <span className="akashic-halo flex size-6 shrink-0 items-center justify-center rounded-full border border-[color-mix(in_srgb,var(--gd)_45%,transparent)] bg-[color-mix(in_srgb,var(--gd)_10%,transparent)]">
   378→   378→              <Scroll className="size-3 text-[var(--gd)]" aria-hidden="true" />
   379→   379→            </span>
   380→   380→            <h1 className="font-ancient min-w-0 truncate text-[15px] font-semibold tracking-[0.14em] text-[var(--gd)] sm:text-[17px]">
   381→   381→              {t("The Akashic Library")}
   382→   382→            </h1>
   383→   383→          </span>
   384→   384→          <span className="mono-label hidden text-[9px] uppercase tracking-[0.26em] text-muted-foreground/70 sm:block">
   385→   385→            {t("resonance · records · the ancient one")}
   386→   386→          </span>
   387→   387→        </div>
   388→   388→
   389→   389→        <button
   390→   390→          type="button"
   391→   391→          onClick={() => void seek(null, false)}
   392→   392→          disabled={seeking}
   393→   393→          data-testid="akashic-new"
   394→   394→          aria-label={t("New record")}
   395→   395→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--gd)_30%,transparent)] text-[12px] text-foreground/85 transition-all duration-300 hover:-translate-y-px hover:border-[var(--hairline-hover)] hover:text-foreground disabled:cursor-not-allowed disabled:opacity-40 sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   396→   396→        >

   399→   399→            aria-hidden="true"
   400→   400→          />
   401→   401→          <span className="hidden sm:inline">{t("New record")}</span>
   402→   402→        </button>
   403→   403→      </header>
   404→   404→
   405→   405→      {/* ---------- the reading room: one papyrus, one record ---------- */}
   406→   406→      <div className="nice-scroll relative z-10 flex-1 overflow-y-auto">
   407→   407→        <div className="mx-auto flex w-full max-w-[780px] flex-col px-3 pb-10 pt-5 sm:px-6 sm:pt-7">

   411→   411→              <motion.div

   418→   418→                data-testid="akashic-invitation"
   419→   419→              >
   420→   420→                <span
   421→   421→                  aria-hidden="true"
   422→   422→                  className="ink-gold mb-5 block text-[22px] leading-none"

   424→   424→                  ❧
   425→   425→                </span>
   426→   426→                <p className="ink-hand mx-auto max-w-[460px] text-[18px] leading-[1.9] sm:text-[20px]">
   427→   427→                  {t(
   428→   428→                    "Set down a resonance — a name, a question, a feeling — and the Librarian will draw out the record it belongs to. Or leave the desk silent, and the Library will choose for you."
   429→   429→                  )}
   430→   430→                </p>
   431→   431→                <button
   432→   432→                  type="button"
   433→   433→                  onClick={() => void seek(null, false)}
   434→   434→                  data-testid="akashic-receive"

   436→   436→                >
   437→   437→                  <Feather className="size-4" aria-hidden="true" />
   438→   438→                  {t("Receive a record")}
   439→   439→                </button>
   440→   440→              </motion.div>
   441→   441→            )}
   442→   442→
   443→   443→            {/* the ink forming — the Librarian walks the shelves */}
   444→   444→            {seeking && (
   445→   445→              <motion.div
   446→   446→                key="seeking"
   447→   447→                initial={{ opacity: 0 }}
   448→   448→                animate={{ opacity: 1 }}
   449→   449→                exit={{ opacity: 0 }}
   450→   450→                transition={{ duration: 0.5 }}
   451→   451→                className="papyrus papyrus-frame relative mx-auto w-full rounded-2xl px-6 py-12 sm:px-12"
   452→   452→                aria-live="polite"
   453→   453→                aria-busy="true"
   454→   454→                data-testid="akashic-seeking"
   455→   455→              >
   456→   456→                <span
   457→   457→                  aria-hidden="true"
   458→   458→                  className="ink-gold mb-6 block text-center text-[22px] leading-none"
   459→   459→                >
