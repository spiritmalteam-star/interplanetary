    46→/* ---------- parsing the transmission into rich blocks ---------- */
    47→
    48→type Block =
    49→  | { kind: "opening"; text: string }
    50→  | { kind: "para"; text: string }
    51→  | { kind: "list"; items: string[] }
    52→  | { kind: "signature"; text: string };
    53→
    54→function parseBlocks(text: string): Block[] {
    55→  const raw = text
    56→    .split(/\n{2,}/)
    57→    .map((p) => p.trim())
    58→    .filter(Boolean);
    59→  const blocks: Block[] = [];
    60→
    61→  raw.forEach((p, i) => {
    62→    const isLast = i === raw.length - 1;
    63→    if (isLast && (p.startsWith("—") || p.startsWith("—"))) {
    64→      blocks.push({ kind: "signature", text: p });
    65→      return;
    66→    }
    67→    const lines = p.split("\n").map((l) => l.trim());
    68→    const isList =
    69→      lines.length > 0 &&
    70→      lines.every((l) => l.startsWith("- ") || l.startsWith("• "));
    71→    if (isList) {
    72→      blocks.push({
    73→        kind: "list",
    74→        items: lines.map((l) => l.replace(/^[-•]\s*/, "")),
    75→      });
    76→      return;
    77→    }
    78→    if (i === 0) {
    79→      blocks.push({ kind: "opening", text: p });
    80→      return;
    81→    }
    82→    blocks.push({ kind: "para", text: p });
    83→  });
    84→
    85→  return blocks;
    86→}

   390→            "The atelier is quiet — the vision could not be composed. Rest a breath, then ask again."
   391→          )}
   392→        </p>
   393→        {message.visualRequest && (
   394→          <button
   395→            type="button"
   396→            onClick={() =>
   397→              void askScopeVisual(activeMode, message.visualRequest ?? "", null)

   400→            className="focus-glow mt-2.5 flex h-9 items-center gap-2 rounded-full border px-4 text-[13px] text-foreground/90 transition-all duration-300 hover:-translate-y-px"
   401→            style={{
   402→              borderColor: "color-mix(in srgb, var(--ac) 35%, transparent)",
   403→            }}
   404→          >
   405→            <RotateCcw className="size-3.5" aria-hidden="true" />
   406→            {t("Be still and receive")}
   407→          </button>
   408→        )}
   409→      </div>
   410→    );
   411→  }
   412→  if (!message.artifact) return null;
   413→  const artifact = message.artifact;
   414→  if (!artifact.imageUrl && artifact.slides.length === 0) {
   415→    return (
   416→      <PreparedPromptFallback
   417→        artifact={artifact}
   418→        accent="var(--ac)"
   419→        testIdPrefix="scope-visual"
   420→        onPaint={() =>
   421→          void askScopeVisual(
   422→            activeMode,
   423→            message.visualRequest ?? artifact.subject,
   424→            {
   425→              id: message.id,
   426→              request: message.visualRequest ?? artifact.subject,
   427→              prompt: artifact.prompt,
   428→              subject: artifact.subject,
   429→              mode: artifact.mode,
   430→            }
   431→          )
   432→        }
   433→      />
   434→    );
   435→  }
   436→  return (
   437→    <VisualizationCard
   438→      artifact={artifact}
   439→      accent="var(--ac)"
   440→      testIdPrefix="scope-visual"
   441→      onRegenerate={() =>
   442→        void askScopeVisual(activeMode, message.visualRequest ?? artifact.subject, {
   443→          id: message.id,
   444→          request: message.visualRequest ?? artifact.subject,
   445→          prompt: artifact.prompt,
   446→          subject: artifact.subject,
   447→          mode: artifact.mode,
   448→        })
   449→      }
   450→    />
   451→  );
   452→}

  1026→/* ---------- main view: the chamber — one channel per scope ---------- */

  1028→export function TransmissionView() {
  1029→  const activeMode = useMirror((s) => s.activeMode);
  1030→  const session = useMirror((s) => s.sessions[s.activeMode]);
  1031→  const remedyStatus = useMirror((s) => s.remedyStatus);
  1032→  const t = useT();
  1033→
  1034→  const scope: Scope = activeMode;
  1035→  const meta = SCOPE_META[scope];
  1036→
  1037→  /* the pager's anchor: -1 = follow the latest */
  1038→  const [anchor, setAnchor] = useState(-1);
  1039→  const exchangeRefs = useRef<Array<HTMLDivElement | null>>([]);
  1040→
  1041→  const messagesLength = session.messages.length;
  1042→
  1043→  /* a new arrival (or a scope switch) resets the pager to the latest —
  1044→     the render-time reset pattern, no effect required */
  1045→  const [pagerKey, setPagerKey] = useState(`${messagesLength}:${activeMode}`);
  1046→  const nextPagerKey = `${messagesLength}:${activeMode}`;
  1047→  if (pagerKey !== nextPagerKey) {
  1048→    setPagerKey(nextPagerKey);
  1049→    setAnchor(-1);
  1050→  }
  1051→
  1052→  /* The apothecary serves the concern most recently spoken — the last
  1053→     exchange's question, held quietly until the thread rests. */
  1054→  const lastSpoken = useMemo(
  1055→    () =>
  1056→      scope === "healing"
  1057→        ? (session.messages.at(-1)?.query ?? null)
  1058→        : null,
  1059→    [scope, session.messages]
  1060→  );
  1061→
  1062→  /* The thread begins where it begins: opening, reloading or switching
  1063→     into a channel keeps the thread at its BEGINNING. New generations
  1064→     load from the TOP — the newest exchange settles its first line at
  1065→     the top of the view and the visitor reads downward from there.
  1066→     One law above all: the moment a suggestion is struck — even after
  1067→     the thread already carries conversations — the frame must meet
  1068→     THAT exchange, while it forms and again when it lands. */
  1069→  const topRef = useRef<HTMLDivElement | null>(null);
  1070→  const latestRef = useRef<HTMLDivElement | null>(null);
  1071→  const loadingRef = useRef<HTMLDivElement | null>(null);
  1072→  const errorRef = useRef<HTMLDivElement | null>(null);
  1073→  const status = session.status;
  1074→  const baseline = useRef({
  1075→    mode: activeMode,
  1076→    len: messagesLength,
  1077→    status,
  1078→    fresh: true,
  1079→  });
  1080→
  1081→  useEffect(() => {
  1082→    const b = baseline.current;
  1083→    const sameMode = b.mode === activeMode;
  1084→    const grew = sameMode && messagesLength !== b.len;
  1085→    const startedLoading =
  1086→      sameMode && !grew && status === "loading" && b.status !== "loading";
  1087→    const becameError =
  1088→      sameMode &&
  1089→      !grew &&
  1090→      !startedLoading &&
  1091→      status === "error" &&
  1092→      b.status !== "error";
  1093→
  1094→    baseline.current = {
  1095→      mode: activeMode,
  1096→      len: messagesLength,
  1097→      status,
  1098→      fresh: false,
  1099→    };
  1100→
  1101→    if (b.fresh || !sameMode) {
  1102→      /* First sight of the channel — arriving from the observatory the
  1103→         transmission may ALREADY be forming (a suggestion was struck
  1104→         and the channel mounted mid-reception): meet it where it
  1105→         forms, never wait above the older thread. A channel entering
  1106→         with a failed transmission meets the error; otherwise it
  1107→         rests at its beginning. */
  1108→      if (status === "loading") {
  1109→        loadingRef.current?.scrollIntoView({ block: "start" });
  1110→      } else if (status === "error") {
  1111→        errorRef.current?.scrollIntoView({ block: "start" });
  1112→      } else if (!sameMode) {
  1113→        topRef.current?.scrollIntoView({ block: "start" });
  1114→      }
  1115→      return;
  1116→    }
  1117→    if (grew) {
  1118→      latestRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
  1119→      /* the answer must own the frame — re-affirm once the entrance
  1120→         has settled, in case the first glide was preempted */
  1121→      const id = window.setTimeout(() => {
  1122→        latestRef.current?.scrollIntoView({
  1123→          behavior: "smooth",
  1124→          block: "start",
  1125→        });
  1126→      }, 450);
  1127→      return () => window.clearTimeout(id);
  1128→    }
  1129→    if (startedLoading) {
  1130→      loadingRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
  1131→      return;
  1132→    }
  1133→    if (becameError) {
  1134→      errorRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
  1135→    }
  1136→  }, [activeMode, messagesLength, status]);
  1137→
  1138→  /* the pager glides to the chosen exchange */
  1139→  const gotoAnchor = (n: number) => {
  1140→    setAnchor(n);
  1141→    if (n >= 1) {
  1142→      exchangeRefs.current[n - 1]?.scrollIntoView({
  1143→        behavior: "smooth",
  1144→        block: "start",
  1145→      });
  1146→    }
  1147→  };
  1148→
  1149→  return (
  1150→    <motion.section
