    84→    84→/* ---------- parsing the transmission into rich blocks ---------- */
    85→    85→
    86→    86→type Block =
    87→    87→  | { kind: "opening"; text: string }
    88→    88→  | { kind: "para"; text: string }
    89→    89→  | { kind: "list"; items: string[] }
    90→    90→  | { kind: "signature"; text: string };
    91→    91→
    92→    92→function parseBlocks(text: string): Block[] {
    93→    93→  const raw = text
    94→    94→    .split(/\n{2,}/)
    95→    95→    .map((p) => p.trim())
    96→    96→    .filter(Boolean);
    97→    97→  const blocks: Block[] = [];
    98→    98→
    99→    99→  raw.forEach((p, i) => {
   100→   100→    const isLast = i === raw.length - 1;
   101→   101→    if (isLast && (p.startsWith("—") || p.startsWith("—"))) {
   102→   102→      blocks.push({ kind: "signature", text: p });
   103→   103→      return;
   104→   104→    }
   105→   105→    const lines = p.split("\n").map((l) => l.trim());
   106→   106→    const isList =
   107→   107→      lines.length > 0 &&
   108→   108→      lines.every((l) => l.startsWith("- ") || l.startsWith("• "));
   109→   109→    if (isList) {
   110→   110→      blocks.push({
   111→   111→        kind: "list",
   112→   112→        items: lines.map((l) => l.replace(/^[-•]\s*/, "")),
   113→   113→      });
   114→   114→      return;
   115→   115→    }
   116→   116→    if (i === 0) {
   117→   117→      blocks.push({ kind: "opening", text: p });
   118→   118→      return;
   119→   119→    }
   120→   120→    blocks.push({ kind: "para", text: p });
   121→   121→  });
   122→   122→
   123→   123→  return blocks;
   124→   124→}
