   531→  askMirror: async (question, attachments) => {
   532→    const query = question.trim();
   533→    const mode = get().activeMode;
   534→    const session = get().sessions[mode];
   535→    if (!query || session.status === "loading") return;
   536→
   537→    set((s) => ({
   538→      view: "transmission",
   539→      mobileNavOpen: false,
   540→      sessions: {
   541→        ...s.sessions,
   542→        [mode]: {
   543→          ...s.sessions[mode],
   544→          status: "loading",
   545→          activeQuery: query,
   546→          draft: "",
   547→          error: null,
   548→        },
   549→      },
   550→    }));
   551→
   552→    try {
   553→      /* Context memory: the mirror remembers this channel's earlier
   554→         exchanges, so conversations deepen instead of restarting. */
   555→      const history = session.messages.slice(-6).map((m) => ({
   556→        q: m.query,
   557→        a: m.text,
   558→      }));
   559→      const payload = attachments ? attachmentsToPayload(attachments) : null;
   560→      const res = await fetch("/api/transmission", {
   561→        method: "POST",
   562→        headers: { "Content-Type": "application/json" },
   563→        body: JSON.stringify({
   564→          query,
   565→          mode,
   566→          scienceField: get().activeScienceField,
   567→          direction: get().activeDirection,
   568→          language: get().language,
   569→          history,
   570→          ...(payload ?? {}),
   571→        }),
   572→      });
   573→      const data = await res.json().catch(() => null);
   574→
   575→      if (!res.ok) {
   576→        throw new Error(
   577→          (data && data.error) ||
   578→            "The field is momentarily quiet. Rest, then try again."
   579→        );
   580→      }
   581→
   582→      set((s) => ({
   583→        sessions: {
   584→          ...s.sessions,
   585→          [mode]: {
   586→            ...s.sessions[mode],
   587→            status: "ready",
   588→            activeQuery: "",
   589→            messages: [
   590→              ...s.sessions[mode].messages,
   591→              {
   592→                id: nextMessageId(),
   593→                query,
   594→                text: data.transmission,
   595→                classification: data.classification,
   596→                createdAt: data.createdAt ?? new Date().toISOString(),
   597→                ...(attachments
   598→                  ? {
   599→                      attachments: {
   600→                        images: attachments.filter((a) => a.kind === "image")
   601→                          .length,
   602→                        docNames: attachments
   603→                          .filter((a) => a.kind === "document")
   604→                          .map((a) => a.name),
   605→                      },
   606→                    }
   607→                  : {}),
   608→              },
   609→            ],
   610→          },
   611→        },
   612→      }));
   613→    } catch (err) {
   614→      set((s) => ({
   615→        sessions: {
   616→          ...s.sessions,
   617→          [mode]: {
   618→            ...s.sessions[mode],
   619→            status: "error",
   620→            activeQuery: "",
   621→            error:
   622→              err instanceof Error
   623→                ? err.message
   624→                : "The field is momentarily quiet. Rest, then try again.",
   625→          },
   626→        },
   627→      }));
   628→    }
