   419→/* ---------------- the forge chat — the direct mirror line ---------------- */
   420→
   421→function ForgeBody({ text }: { text: string }) {
   422→  const paras = useMemo(
   423→    () =>
   424→      text
   425→        .split(/\n{2,}/)
   426→        .map((p) => p.trim())
   427→        .filter(Boolean),
   428→    [text]
   429→  );
   430→  return (
   431→    <div className="space-y-2.5">
   432→      {paras.map((p, i) => {
   433→        const isLast = i === paras.length - 1;
   434→        if (isLast && p.startsWith("—")) {
   435→          return (
   436→            <p
   437→              key={i}
   438→              className="mono-label pt-1 text-[11px] leading-relaxed"
   439→              style={{ color: "color-mix(in srgb, var(--scope-b) 82%, white)" }}
   440→            >
   441→              {p}
   442→            </p>
   443→          );
   444→        }
   445→        return (
   446→          <p
   447→            key={i}
   448→            className={cn(
   449→              "text-[14.5px] leading-[1.75] text-foreground/88",
   450→              i === 0 && "font-medium text-foreground/95"
   451→            )}
   452→          >
   453→            {p}
   454→          </p>
   455→        );
   456→      })}
   457→    </div>
   458→  );
   459→}

   488→function ForgeChat() {

   491→  const askForge = useMirror((s) => s.askForge);
   492→  const clearChannel = useMirror((s) => s.clearChannel);
   493→  const t = useT();
   494→  const threadRef = useRef<HTMLDivElement | null>(null);
   495→  const [phase, setPhase] = useState(0);
   496→
   497→  const { status, messages, draft, error, activeQuery } = forgeSession;
   498→  const loading = status === "loading";
   499→
   500→  useEffect(() => {
   501→    if (!loading) return;
   502→    const id = window.setInterval(
   503→      () => setPhase((p) => (p + 1) % forgeChatPhases.length),
   504→      1900
   505→    );
   506→    return () => window.clearInterval(id);

   511→    const el = threadRef.current;

   517→    const q = (text ?? draft).trim();

   519→    void askForge(q);

   521→
   522→  const canSend = draft.trim().length > 0 && !loading;
   523→
   524→  return (
   525→    <div
   526→      className="scope-frame-card relative flex flex-col overflow-hidden rounded-2xl glass"
   527→      data-testid="forge-chat"
   528→    >
   529→      <span className="scope-corner scope-corner-tl" aria-hidden="true" />
   530→      <span className="scope-corner scope-corner-tr" aria-hidden="true" />
   531→      <span className="scope-corner scope-corner-bl" aria-hidden="true" />
   532→      <span className="scope-corner scope-corner-br" aria-hidden="true" />
   533→
   534→      {/* header */}

   537→          <Hammer className="size-3.5" aria-hidden="true" />
   538→          {t("The Forge speaks")}
   539→        </h3>
   540→        {messages.length > 0 && (
   541→          <button
   542→            type="button"
   543→            onClick={() => clearChannel("forge")}
   544→            aria-label={t("Quiet the Forge")}
   545→            title={t("Quiet the Forge")}
   546→            className="focus-glow flex size-7 items-center justify-center rounded-full border hairline text-muted-foreground transition-all duration-300 hover:text-foreground"

   553→      {/* thread — scrolls inside its own pane, the workshop stays one screen */}

   556→        className="nice-scroll max-h-[46vh] min-h-[240px] flex-1 space-y-3.5 overflow-y-auto overscroll-contain px-4 py-4 sm:px-5 lg:max-h-[440px]"

   560→          <div className="flex h-full min-h-[220px] flex-col items-center justify-center text-center">
   561→            <Hammer
   562→              className="size-6 text-muted-foreground/40"
   563→              aria-hidden="true"
   564→            />
