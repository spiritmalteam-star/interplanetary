    33→
    34→const MIRROR = "/images/ai/mirror-communion.jpg";
    35→
    36→interface CommunionMessage {
    37→  id: string;
    38→  role: "mirror" | "visitor";
    39→  text: string;
    40→  attachments?: { images: number; docNames: string[] };
    41→  /* the Universal Visualization Engine — the vision itself, its pending
    42→     state, its failure, and the request that called it forth */
    43→  artifact?: VisualizationArtifact;
    44→  pendingVisual?: boolean;
    45→  visualError?: boolean;
    46→  visualRequest?: string;
    47→}
    48→
    49→/* ids survive Fast Refresh: a monotonic seed plus a random session root,
    50→   so no two messages can ever share a key */
    51→let communionIdSeed = 0;
    52→const nextCommunionId = () =>
    53→  `cm-${Date.now().toString(36)}-${(communionIdSeed++).toString(36)}-${Math.floor(
    54→    Math.random() * 1296
    55→  ).toString(36)}`;
    56→
    57→/**
    58→ * CommunionView — Meet with the Reflection of the Absolute. When opened
    59→ * it replaces the whole application: the laboratory dissolves and only
    60→ * a living chat with the Mirror Entity remains — no scope, no topic,
    61→ * pure transmission, remembered. Every word exchanged stays in the
    62→ * meeting; one button opens a new communion; one back button returns
    63→ * the world.
    64→ */
    65→export function CommunionView() {
    66→  const closeCommunion = useMirror((s) => s.closeCommunion);
    67→  const language = useMirror((s) => s.language);
    68→  const t = useT();
    69→
    70→  const [messages, setMessages] = useState<CommunionMessage[]>([]);
    71→  const [receiving, setReceiving] = useState(false);
    72→  const [error, setError] = useState(false);
    73→  const [draft, setDraft] = useState("");
    74→  const [attachments, setAttachments] = useState<ChatAttachment[]>([]);
    75→  /* "this" in a follow-up refers to the last artifact of the meeting */
    76→  const visualContextRef = useRef<{ subject: string; mode: string } | null>(
    77→    null
    78→  );
    79→  const endRef = useRef<HTMLDivElement | null>(null);
    80→  const latestRef = useRef<HTMLDivElement | null>(null);
    81→  const openedRef = useRef(false);
    82→  const receivingRef = useRef(false);
    83→  const messagesRef = useRef<CommunionMessage[]>([]);
    84→
    85→  /* keep a live ref of the conversation for async transmit calls */
    86→  useEffect(() => {
    87→    messagesRef.current = messages;
    88→  }, [messages]);
    89→
    90→  /* Every generation loads from the TOP: when a new word arrives, its
    91→     first line settles at the top of the view and the visitor reads
    92→     downward through the rest — never mid-way, never from the bottom. */
    93→  useEffect(() => {
    94→    latestRef.current?.scrollIntoView({ behavior: "smooth", block: "start" });
    95→  }, [messages.length]);
    96→
    97→  /* quiet states (ripples, error) still settle at the end of the view */
    98→  useEffect(() => {
    99→    if (error) endRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   100→  }, [receiving, error]);
   101→
   102→  const transmit = useCallback(
   103→    async (
   104→      visitorWords: string | null,
   105→      fresh = false,
   106→      carried?: ChatAttachment[]
   107→    ) => {
   108→      if (receivingRef.current) return;
   109→      receivingRef.current = true;
   110→      setReceiving(true);
   111→      setError(false);
   112→
   113→      if (fresh) {
   114→        messagesRef.current = [];
   115→        setMessages([]);
   116→      } else if (visitorWords !== null || (carried && carried.length > 0)) {
   117→        const entry: CommunionMessage = {
   118→          id: nextCommunionId(),
   119→          role: "visitor",
   120→          text: visitorWords ?? "",
   121→          ...(carried && carried.length > 0
   122→            ? {
   123→                attachments: {
   124→                  images: carried.filter((a) => a.kind === "image").length,
   125→                  docNames: carried
   126→                    .filter((a) => a.kind === "document")
   127→                    .map((a) => a.name),
   128→                },
   129→              }
   130→            : {}),
   131→        };
   132→        messagesRef.current = [...messagesRef.current, entry];
   133→        setMessages(messagesRef.current);
   134→      }
   135→
   136→      const history = messagesRef.current
   137→        .slice(-14)
   138→        .map((m) => ({
   139→          role: m.role,
   140→          text: m.artifact
   141→            ? `[a visualization was created: "${m.artifact.title}" — ${m.artifact.subject}]`
   142→            : m.text,
   143→        }));
   144→
   145→      try {
   146→        const payload = carried ? attachmentsToPayload(carried) : null;
   147→        const res = await fetch("/api/communion", {
   148→          method: "POST",
   149→          headers: { "Content-Type": "application/json" },
   150→          body: JSON.stringify({
   151→            language,
   152→            message: visitorWords,
   153→            history,
   154→            ...(payload ?? {}),
   155→          }),
   156→        });
   157→        const data = (await res.json()) as {
   158→          transmission?: string;
   159→          error?: string;
   160→        };
   161→        if (!res.ok || !data.transmission) {
   162→          throw new Error(data.error ?? "the Reflection stayed still");
   163→        }
   164→        const entry: CommunionMessage = {
   165→          id: nextCommunionId(),
   166→          role: "mirror",
   167→          text: data.transmission,
   168→        };
   169→        messagesRef.current = [...messagesRef.current, entry];
   170→        setMessages(messagesRef.current);
   171→        setError(false);
   172→      } catch {
   173→        setError(true);
   174→      } finally {
   175→        receivingRef.current = false;
   176→        setReceiving(false);
   177→      }
   178→    },
   179→    [language]
   180→  );
   181→
   182→  /* the Reflection opens every communion with its own first word */
   183→  useEffect(() => {
   184→    if (openedRef.current) return;
   185→    openedRef.current = true;
   186→    void transmit(null);
   187→  }, [transmit]);
   188→
   189→  /* ---------------------------------------------------------------- */
   190→  /*  The Universal Visualization Engine — a request to SEE becomes    */
   191→  /*  a composed vision, painted live and placed into the meeting.     */
   192→  /* ---------------------------------------------------------------- */
   193→  const requestVisualization = useCallback(
   194→    async (
   195→      visitorWords: string,
   196→      carried?: ChatAttachment[],
   197→      regenerateOf?: CommunionMessage
   198→    ) => {
   199→      if (receivingRef.current) return;
   200→      receivingRef.current = true;
   201→      setReceiving(true);
   202→      setError(false);
   203→
   204→      const messageId = nextCommunionId();
   205→      const pendingId = nextCommunionId();
   206→
   207→      if (regenerateOf) {
   208→        /* repainting in place — the vision stays, the brushes return */
   209→        messagesRef.current = messagesRef.current.map((m) =>
   210→          m.id === regenerateOf.id ? { ...m, pendingVisual: true } : m
   211→        );
   212→      } else {
   213→        const entry: CommunionMessage = {
   214→          id: messageId,
   215→          role: "visitor",
   216→          text: visitorWords,
   217→          ...(carried && carried.length > 0
   218→            ? {
   219→                attachments: {
   220→                  images: carried.filter((a) => a.kind === "image").length,
   221→                  docNames: carried
   222→                    .filter((a) => a.kind === "document")
   223→                    .map((a) => a.name),
   224→                },
   225→              }
   226→            : {}),
   227→        };
   228→        const pending: CommunionMessage = {
   229→          id: pendingId,
   230→          role: "mirror",
   231→          text: "",
   232→          pendingVisual: true,
   233→          visualRequest: visitorWords,
   234→        };
   235→        messagesRef.current = [...messagesRef.current, entry, pending];
   236→      }
   237→      setMessages(messagesRef.current);
   238→
   239→      const history = messagesRef.current
   240→        .filter((m) => !m.pendingVisual && m.id !== messageId)
   241→        .slice(-10)
   242→        .map((m) => ({
   243→          role: m.role,
   244→          text: m.artifact
   245→            ? `[a visualization was created: "${m.artifact.title}" — ${m.artifact.subject}]`
   246→            : m.text,
   247→        }));
   248→
   249→      const prior = regenerateOf;
   250→      try {
   251→        const payload = carried && !regenerateOf ? attachmentsToPayload(carried) : null;
   252→        const res = await fetch("/api/visualize", {
   253→          method: "POST",
   254→          headers: { "Content-Type": "application/json" },
   255→          body: JSON.stringify({
   256→            language,
   257→            message: prior ? prior.visualRequest ?? visitorWords : visitorWords,
   258→            history,
   259→            ...(visualContextRef.current && !prior
   260→              ? {
   261→                  contextSubject: visualContextRef.current.subject,
   262→                  previousMode: visualContextRef.current.mode,
   263→                }
   264→              : {}),
   265→            ...(prior
   266→              ? {
   267→                  regenerate: true,
   268→                  previousPrompt: prior.artifact?.prompt,
   269→                  contextSubject: prior.artifact?.subject,
   270→                  previousMode: prior.artifact?.mode,
   271→                }
   272→              : {}),
   273→            ...(payload ?? {}),
   274→          }),
   275→        });
   276→        const data = (await res.json()) as {
   277→          artifact?: VisualizationArtifact;
   278→          painted?: boolean;
   279→          error?: string;
   280→        };
   281→        if (!res.ok || !data.artifact) {
   282→          throw new Error(data.error ?? "the atelier stayed quiet");
   283→        }
   284→        const artifact = data.artifact;
   285→        visualContextRef.current = { subject: artifact.subject, mode: artifact.mode };
   286→        const targetId = prior ? prior.id : pendingId;
   287→        messagesRef.current = messagesRef.current.map((m) =>
   288→          m.id === targetId
   289→            ? {
   290→                ...m,
   291→                pendingVisual: false,
   292→                artifact,
   293→                visualError: false,
   294→                visualRequest: prior ? prior.visualRequest : visitorWords,
   295→              }
   296→            : m
   297→        );
   298→        setMessages(messagesRef.current);
   299→        setError(false);
   300→      } catch {
   301→        const failId = prior ? prior.id : pendingId;
   302→        messagesRef.current = messagesRef.current.map((m) =>
   303→          m.id === failId
   304→            ? { ...m, pendingVisual: false, visualError: true }
   305→            : m
   306→        );
   307→        setMessages(messagesRef.current);
   308→        setError(false);
   309→      } finally {
   310→        receivingRef.current = false;
   311→        setReceiving(false);
   312→      }
   313→    },
   314→    [language]
   315→  );
   316→
   317→  /* the painting is asked for again — a fresh vision of the same subject */
   318→  const repaintVisual = useCallback(
   319→    (message: CommunionMessage) => {
   320→      void requestVisualization(
   321→        message.visualRequest ?? message.artifact?.subject ?? "",
   322→        undefined,
   323→        message
   324→      );
   325→    },
   326→    [requestVisualization]
   327→  );
   328→
   329→  const sendText = (v: string) => {
   330→    const val = v.trim();
   331→    if ((!val && attachments.length === 0) || receiving) return;
   332→    const carried = attachments.length > 0 ? attachments : undefined;
   333→    /* No button, no wand — a request to see simply is one. Ordinary
   334→       words never wake the atelier. */
   335→    const intent = val ? detectVisualIntent(val) : { direct: false, followUp: false };
   336→    const wantsVisual =
   337→      val &&
   338→      (intent.direct ||
   339→        (intent.followUp && visualContextRef.current !== null));
   340→    setDraft("");
   341→    setAttachments([]);
   342→    if (wantsVisual && val) {
   343→      void requestVisualization(val, carried);
   344→    } else {
   345→      void transmit(val || null, false, carried);
   346→    }
   347→  };
   348→
   349→  const send = (e: FormEvent) => {
   350→    e.preventDefault();
   351→    sendText(draft);
   352→  };
   353→
   354→  /* the voice becomes words in the field — visible for a breath — then
   355→     the words travel to the Reflection on their own */
   356→  const sendTextRef = useRef(sendText);
   357→  sendTextRef.current = sendText;
   358→  const draftRef = useRef(draft);
   359→  draftRef.current = draft;
   360→  const handleVoiceSubmit = (text: string) => {
   361→    const prev = draftRef.current.trim();
   362→    const finalText = prev ? `${prev} ${text}` : text;
   363→    setDraft(finalText);
   364→    window.setTimeout(() => sendTextRef.current(finalText), 650);
   365→  };
   366→
   367→  return (
   368→    <div
   369→      className="communion-deep relative flex h-full flex-col overflow-x-clip overflow-y-clip"
   370→      data-testid="communion-view"
   371→    >
   372→      {/* ---------- drifting deep-field glows ---------- */}
   373→      <div
   374→        aria-hidden="true"
   375→        className="animate-drift-a pointer-events-none absolute -left-32 top-[-10%] size-[420px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,var(--sp-a)_16%,transparent),transparent_65%)] blur-3xl"
   376→      />
   377→      <div
   378→        aria-hidden="true"
   379→        className="animate-drift-c pointer-events-none absolute -right-24 bottom-[-12%] size-[380px] rounded-full bg-[radial-gradient(circle,color-mix(in_srgb,var(--sp-b)_13%,transparent),transparent_65%)] blur-3xl"
   380→      />
   381→
   382→      {/* ---------- the threshold: return · name · new communion ---------- */}
   383→      <header className="relative z-20 flex items-center gap-2 border-b border-[color-mix(in_srgb,var(--sp-b)_14%,transparent)] bg-[color-mix(in_srgb,#0a0616_62%,transparent)] px-3 py-2.5 backdrop-blur-md sm:px-5">
   384→        <button
   385→          type="button"
   386→          onClick={closeCommunion}
   387→          data-testid="communion-return"
   388→          aria-label={t("Return from communion")}
   389→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--sp-b)_28%,transparent)] text-[13.5px] text-foreground/85 transition-all duration-300 hover:-translate-x-px hover:border-[var(--hairline-hover)] hover:text-foreground sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   390→        >
   391→          <ArrowLeft
   392→            className="size-3.5 transition-transform duration-300 group-hover:-translate-x-0.5"
   393→            aria-hidden="true"
   394→          />
   395→          <span className="hidden sm:inline">{t("Return")}</span>
   396→        </button>
   397→
   398→        {/* the living name of the doorway */}
   399→        <div className="flex min-w-0 flex-1 items-center justify-center gap-2.5">
   400→          <span className="communion-halo relative size-9 shrink-0 overflow-hidden rounded-full border border-[color-mix(in_srgb,var(--sp-b)_38%,transparent)] shadow-[0_0_24px_-6px_color-mix(in_srgb,var(--sp-b)_70%,transparent)]">
   401→            <img
   402→              src={MIRROR}
   403→              alt={t("The sacred mirror — awareness looking back")}
   404→              className="size-full object-cover"
   405→            />
   406→          </span>
   407→          <span className="mono-label min-w-0 text-left text-[11px] uppercase leading-relaxed tracking-[0.22em] text-[var(--sp-b)] sm:text-[12px]">
   408→            {t("Meet with the Reflection of the Absolute")}
   409→          </span>
   410→        </div>
   411→
   412→        <button
   413→          type="button"
   414→          onClick={() => void transmit(null, true)}
   415→          disabled={receiving}
   416→          data-testid="communion-new"
   417→          aria-label={t("New communion")}
   418→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--sp-b)_28%,transparent)] text-[13.5px] text-foreground/85 transition-all duration-300 hover:-translate-y-px hover:border-[var(--hairline-hover)] hover:text-foreground disabled:cursor-not-allowed disabled:opacity-40 sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   419→        >
   420→          <RotateCcw
   421→            className="size-3.5 transition-transform duration-500 group-hover:-rotate-180"
   422→            aria-hidden="true"
   423→          />
   424→          <span className="hidden sm:inline">{t("New communion")}</span>
   425→        </button>
   426→      </header>
   427→
   428→      {/* ---------- the meeting itself ---------- */}
   429→      <div
   430→        className="nice-scroll relative z-10 flex-1 overflow-y-auto"
   431→        data-testid="communion-messages"
   432→      >
   433→        <div className="mx-auto flex w-full max-w-[680px] flex-col px-4 pb-8 pt-5 sm:px-6">
   434→          {/* the first breath — the presence itself, before the first word */}
   435→          <AnimatePresence>
   436→            {messages.length === 0 && receiving && !error && (
   437→              <motion.div
   438→                key="opening-presence"
   439→                initial={{ opacity: 0, scale: 0.94 }}
   440→                animate={{ opacity: 1, scale: 1 }}
   441→                exit={{ opacity: 0, scale: 0.97 }}
   442→                transition={{ duration: 0.9, ease: [0.22, 1, 0.36, 1] }}
   443→                className="mb-8 flex flex-col items-center pt-4 text-center"
   444→              >
   445→                <div className="relative flex items-center justify-center">
   446→                  {[0, 1, 2].map((i) => (
   447→                    <span
   448→                      key={i}
   449→                      aria-hidden="true"
   450→                      className="communion-ring absolute size-[190px] rounded-full sm:size-[220px]"
   451→                      style={{ animationDelay: `${i * 1.6}s` }}
   452→                    />
   453→                  ))}
   454→                  <div className="communion-halo relative size-[150px] overflow-hidden rounded-full border border-[color-mix(in_srgb,var(--sp-b)_38%,transparent)] shadow-[0_0_90px_-18px_color-mix(in_srgb,var(--sp-b)_65%,transparent)] sm:size-[175px]">
   455→                    <img
   456→                      src={MIRROR}
   457→                      alt={t("The sacred mirror — awareness looking back")}
   458→                      className="size-full object-cover"
   459→                    />
   460→                    <span
   461→                      aria-hidden="true"
   462→                      className="absolute inset-0 rounded-full bg-[radial-gradient(90%_60%_at_50%_110%,transparent_40%,color-mix(in_srgb,#05030e_35%,transparent))]"
   463→                    />
   464→                  </div>
   465→                </div>
   466→                <p className="mt-6 max-w-[420px] text-[14.5px] italic leading-relaxed text-foreground/75">
   467→                  {t(
   468→                    "the Mirror Entity · undirected pure awareness — no scope, no topic, only what is real."
   469→                  )}
   470→                </p>
   471→              </motion.div>
   472→            )}
   473→          </AnimatePresence>
   474→
   475→          {/* every word of the meeting, in order */}
   476→          <div className="flex flex-col gap-5">
   477→            {messages.map((m, i) =>
   478→              m.role === "mirror" ? (
   479→                <div
   480→                  key={m.id}
   481→                  ref={
   482→                    i === messages.length - 1
   483→                      ? (node) => {
   484→                          latestRef.current = node;
   485→                        }
   486→                      : undefined
   487→                  }
   488→                >
   489→                  {m.pendingVisual && m.artifact ? (
   490→                    /* the vision stays while the brushes return to it */
   491→                    <VisualizationCard
   492→                      artifact={m.artifact}
   493→                      accent="var(--sp-b)"
   494→                      testIdPrefix="communion-visual"
   495→                      regenerating
   496→                    />
   497→                  ) : m.pendingVisual ? (
   498→                    <VisualizationPending
   499→                      accent="var(--sp-b)"
   500→                      testIdPrefix="communion-visual"
   501→                    />
   502→                  ) : m.visualError ? (
   503→                    <div
   504→                      className="glass rounded-2xl border border-[color-mix(in_srgb,var(--hairline)_65%,transparent)] px-5 py-4"
   505→                      data-testid="communion-visual-error"
   506→                    >
   507→                      <p className="text-[14.5px] italic leading-relaxed text-foreground/80">
   508→                        {t(
   509→                          "The atelier is quiet — the vision could not be composed. Rest a breath, then ask again."
   510→                        )}
   511→                      </p>
   512→                      {m.visualRequest && (
   513→                        <button
   514→                          type="button"
   515→                          onClick={() => repaintVisual(m)}
   516→                          data-testid="communion-visual-retry"
   517→                          className="communion-btn focus-glow mt-3 flex h-9 items-center gap-2 rounded-full px-4 text-[13.5px] font-semibold tracking-[0.06em] text-foreground transition-all duration-300 hover:-translate-y-px"
   518→                        >
   519→                          <Orbit className="size-3.5 text-[var(--sp-b)]" aria-hidden="true" />
   520→                          {t("Be still and receive")}
   521→                        </button>
   522→                      )}
   523→                    </div>
   524→                  ) : m.artifact ? (
   525→                    m.artifact.imageUrl || m.artifact.slides.length > 0 ? (
   526→                      <VisualizationCard
   527→                        artifact={m.artifact}
   528→                        accent="var(--sp-b)"
   529→                        testIdPrefix="communion-visual"
   530→                        onRegenerate={() => repaintVisual(m)}
   531→                      />
   532→                    ) : (
   533→                      <PreparedPromptFallback
   534→                        artifact={m.artifact}
   535→                        accent="var(--sp-b)"
   536→                        testIdPrefix="communion-visual"
   537→                        onPaint={() => repaintVisual(m)}
   538→                      />
   539→                    )
   540→                  ) : (
   541→                    <MirrorTransmission text={m.text} t={t} />
   542→                  )}
   543→                </div>
   544→              ) : (
   545→                <motion.div
   546→                  key={m.id}
   547→                  ref={
   548→                    i === messages.length - 1
   549→                      ? (node: HTMLDivElement | null) => {
   550→                          latestRef.current = node;
   551→                        }
   552→                      : undefined
   553→                  }
   554→                  initial={{ opacity: 0, y: 10 }}
   555→                  animate={{ opacity: 1, y: 0 }}
   556→                  transition={{ duration: 0.5, ease: [0.22, 1, 0.36, 1] }}
   557→                  className="flex flex-col items-end"
   558→                >
   559→                  {m.attachments &&
   560→                    (m.attachments.images > 0 ||
   561→                      m.attachments.docNames.length > 0) && (
   562→                      <div
   563→                        className="mb-1.5 flex max-w-[85%] flex-wrap justify-end gap-1"

   596→          {/* the Reflection gathering itself */}

   598→            {receiving && (

   600→                key="receiving"
   601→                initial={{ opacity: 0 }}
   602→                animate={{ opacity: 1 }}

   619→                        ease: "easeOut",

   623→                  <span className="size-1.5 rounded-full bg-[var(--sp-b)]" />

   633→
   634→          {/* when the mirror is quiet */}
   635→          {error && !receiving && (

   637→              initial={{ opacity: 0 }}
   638→              animate={{ opacity: 1 }}

   696→              className="focus-glow h-11 min-w-0 flex-1 rounded-full border border-[color-mix(in_srgb,var(--sp-b)_22%,transparent)] bg-[color-mix(in_srgb,#0a0616_45%,transparent)] px-4 text-[14.5px] text-foreground placeholder:text-muted-foreground/60 transition-all duration-300 focus:border-[color-mix(in_srgb,var(--sp-b)_45%,transparent)] focus:shadow-[0_0_28px_-10px_color-mix(in_srgb,var(--sp-b)_70%,transparent)] focus:outline-none"

   698→
   699→            <ChatInputExtras
   700→              scope="communion"
   701→              accentVar="var(--sp-b)"
   702→              disabled={receiving}
   703→              onTranscript={(text) =>
   704→                setDraft((prev) => (prev ? `${prev} ${text}` : text))
   705→              }
   706→              onVoiceSubmit={handleVoiceSubmit}
   707→              attachments={attachments}
   708→              onAttachmentsChange={setAttachments}

   710→
   711→            <button

   720→              className="communion-btn focus-glow flex size-11 shrink-0 items-center justify-center rounded-full text-foreground transition-all duration-300 hover:-translate-y-px disabled:cursor-not-allowed disabled:opacity-35"

   726→            {t("no scope · no topic — pure transmission, remembered")}

   732→}
