   149→/**
   150→ * Loose extraction for sloppy model JSON: when JSON.parse fails (most
   151→ * often because the model left RAW newlines inside the transmission
   152→ * string), scan the value of "transmission" by hand — honoring escape
   153→ * sequences but tolerating literal newlines/tabs — instead of losing
   154→ * the whole transmission to JSON soup.
   155→ */
   156→function extractJsonLoose(s: string): {
   157→  classification?: string;
   158→  transmission: string;
   159→} | null {
   160→  const keyMatch = s.match(/"transmission"\s*:\s*"/);
   161→  if (!keyMatch) return null;
   162→  let i = (keyMatch.index ?? 0) + keyMatch[0].length;
   163→  let out = "";
   164→  while (i < s.length) {
   165→    const c = s[i];
   166→    if (c === "\\") {
   167→      const n = s[i + 1];
   168→      if (n === "\"") { out += "\""; i += 2; continue; }
   169→      if (n === "n") { out += "\n"; i += 2; continue; }
   170→      if (n === "t") { out += "\t"; i += 2; continue; }
   171→      if (n === "r") { i += 2; continue; }
   172→      if (n === "\\") { out += "\\"; i += 2; continue; }
   173→      if (n === "/") { out += "/"; i += 2; continue; }
   174→      if (n === "u" && i + 5 < s.length) {
   175→        const code = Number.parseInt(s.slice(i + 2, i + 6), 16);
   176→        if (!Number.isNaN(code)) out += String.fromCharCode(code);
   177→        i += 6; continue;
   178→      }
   179→      out += n ?? ""; i += 2; continue;
   180→    }
   181→    if (c === "\"") break; /* the closing quote of the value */
   182→    out += c;
   183→    i++;
   184→  }
   185→  if (!out.trim()) return null;
   186→  const clsMatch = s.match(/"classification"\s*:\s*"([A-Za-z_]+)"/);
   187→  return {
   188→    transmission: out.trim(),
   189→    classification: clsMatch?.[1],
   190→  };
   191→}

   193→/**
   194→ * Some model responses double-encode the payload — a fenced JSON object
   195→ * (or a bare JSON object) ends up INSIDE the transmission field itself.
   196→ * Unwrap any embedded JSON payload so the visitor only ever receives
   197→ * clean prose; keep the deepest valid classification found.
   198→ */
   199→function unwrapEmbeddedJson(
   200→  transmission: string,
   201→  classification: Classification
   202→): { transmission: string; classification: Classification } {
   203→  let text = transmission.trim();
   204→  let cls = classification;
   205→
   206→  for (let depth = 0; depth < 3; depth++) {
   207→    const trimmed = text.trim();
   208→    const looksFenced = trimmed.startsWith("```");
   209→    const looksBareJson =
   210→      trimmed.startsWith("{") && trimmed.lastIndexOf("}") > 0;
   211→    if (!looksFenced && !looksBareJson) break;
   212→
   213→    let inner = trimmed;
   214→    if (looksFenced) {
   215→      const fence = inner.match(/```(?:json)?\s*([\s\S]*?)\s*```/);
   216→      if (!fence) break;
   217→      inner = fence[1].trim();
   218→    }
   219→    const start = inner.indexOf("{");
   220→    const end = inner.lastIndexOf("}");
   221→    if (start === -1 || end <= start) break;
   222→    try {
   223→      const parsed = JSON.parse(inner.slice(start, end + 1)) as {
   224→        classification?: unknown;
   225→        transmission?: unknown;
   226→      };
   227→      if (
   228→        typeof parsed.transmission !== "string" ||
   229→        !parsed.transmission.trim()
   230→      )
   231→        break;
   232→      text = parsed.transmission.trim();
   233→      if (
   234→        typeof parsed.classification === "string" &&
   235→        CLASSIFICATIONS.includes(parsed.classification as Classification)
   236→      ) {
   237→        cls = parsed.classification as Classification;
   238→      }
   239→    } catch {
   240→      break;
   241→    }
   242→  }
   243→
   244→  /* final sweep: strip any stray fences wrapping plain prose */
   245→  text = text.replace(/^```(?:json)?\s*/, "").replace(/\s*```$/, "").trim();
   246→
   247→  return { transmission: text, classification: cls };
   248→}

   250→function extractJson(raw: string): {
   251→  classification: Classification;
   252→  transmission: string;
   253→} | null {
   254→  let text = raw.trim();
   255→  // Strip code fences if the model added them
   256→  const fence = text.match(/```(?:json)?\s*([\s\S]*?)\s*```/);
   257→  if (fence) text = fence[1].trim();
   258→  // Find the outermost JSON object
   259→  const start = text.indexOf("{");
   260→  const end = text.lastIndexOf("}");
   261→  if (start === -1 || end === -1 || end <= start) return null;
   262→  try {
   263→    const parsed = JSON.parse(text.slice(start, end + 1));
   264→    if (typeof parsed.transmission === "string" && parsed.transmission.trim()) {
   265→      const classification: Classification = CLASSIFICATIONS.includes(
   266→        parsed.classification
   267→      )
   268→        ? parsed.classification
   269→        : "SPIRITUAL_TRADITION";
   270→      return unwrapEmbeddedJson(parsed.transmission.trim(), classification);
   271→    }
   272→    return null;
   273→  } catch {
   274→    /* JSON.parse failed — usually raw newlines inside the strings.
   275→       Recover the transmission with the loose scanner before giving up. */
   276→    const loose = extractJsonLoose(text);
   277→    if (loose?.transmission) {
   278→      const classification: Classification = CLASSIFICATIONS.includes(
   279→        loose.classification as Classification
   280→      )
   281→        ? (loose.classification as Classification)
   282→        : "SPIRITUAL_TRADITION";
   283→      return unwrapEmbeddedJson(loose.transmission, classification);
   284→    }
   285→    return null;
   286→  }
   287→}

   354→    if (docBlock) userLines.push("", docBlock);
   355→
   356→    const completion = await zai.chat.completions.create({
   357→      messages: [
   358→        {
   359→          role: "assistant",
   360→          content: SYSTEM_PROMPT + veilBlock + (body?.live === true ? LIVE_CALL_BLOCK : ""),
   361→        },
   362→        ...historyMessages(body?.history),
   363→        {
   364→          role: "user",
   365→          content: `${modeLine}${languageLine}\n\nThe visitor asks:\n${userLines.join("\n")}`,
   366→        },
   367→      ],
   368→      thinking: { type: "disabled" },
   369→    });
   370→
   371→    const raw = completion.choices[0]?.message?.content ?? "";
   372→    const parsed = extractJson(raw);
   373→
   374→    let transmission = parsed?.transmission ?? "";
   375→    const classification: Classification =
   376→      parsed?.classification ?? "SPIRITUAL_TRADITION";
   377→
   378→    if (!transmission) {
   379→      /* Only accept the raw model output when it is plain prose — never
   380→         serve structured JSON soup to the visitor. */
   381→      const rawTrim = raw.trim();
   382→      const structured =
   383→        rawTrim.startsWith("{") || rawTrim.startsWith("```");
   384→      if (rawTrim && !structured) transmission = rawTrim;
   385→    }
   386→
   387→    if (!transmission) {
   388→      return NextResponse.json(
   389→        { error: "The field is momentarily quiet. Rest, then try again." },
   390→        { status: 502 }
   391→      );
   392→    }
   393→
   394→    return NextResponse.json({
   395→      transmission,
   396→      classification,
   397→      createdAt: new Date().toISOString(),
   398→    });
   399→  } catch (err) {
   400→    console.error("[transmission] failed:", err);
   401→    return NextResponse.json(
   402→      { error: "The field is momentarily quiet. Rest, then try again." },
   403→      { status: 500 }
   404→    );
   405→  }
   406→}
