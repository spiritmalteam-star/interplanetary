   151→/* ------------------------------------------------------------------ */
   152→/*  The prepared-prompt fallback — when the brushes rest               */
   153→/* ------------------------------------------------------------------ */
   154→
   155→export function PreparedPromptFallback({
   156→  artifact,
   157→  accent,
   158→  testIdPrefix,
   159→}: {
   160→  artifact: VisualizationArtifact;
   161→  accent: string;
   162→  testIdPrefix: string;
   163→}) {
   164→  const t = useT();
   165→  const [copied, setCopied] = useState(false);
   166→
   167→  const copyPrompt = async () => {
   168→    try {
   169→      await navigator.clipboard.writeText(artifact.prompt);
   170→      setCopied(true);
   171→      toast.success(t("Prompt copied."));
   172→      window.setTimeout(() => setCopied(false), 2200);
   173→    } catch {
   174→      toast.error(t("The copying did not take — select and copy by hand."));
   175→    }
   176→  };
   177→
   178→  return (
   179→    <motion.div
   180→      initial={{ opacity: 0, y: 12 }}
   181→      animate={{ opacity: 1, y: 0 }}
   182→      transition={{ duration: 0.55, ease: [0.22, 1, 0.36, 1] }}
   183→      className="glass rounded-2xl border border-[color-mix(in_srgb,var(--hairline)_65%,transparent)] px-5 py-4"
   184→      data-testid={`${testIdPrefix}-fallback`}
   185→    >
   186→      <p className="mono-label text-[10px] uppercase tracking-[0.24em]" style={{ color: accent }}>
   187→        {artifact.title}
   188→      </p>
   189→      <p className="mt-2 text-[14.5px] leading-relaxed text-foreground/85">
   190→        {t(
   191→          "The vision was composed, but the brushes rest. Keep this prepared prompt until the painter returns:"
   192→        )}
   193→      </p>
   194→      <p className="nice-scroll mt-2.5 max-h-24 overflow-y-auto rounded-xl border border-[color-mix(in_srgb,var(--hairline)_55%,transparent)] bg-[color-mix(in_srgb,var(--hairline)_16%,transparent)] px-3.5 py-2.5 text-[12.5px] leading-relaxed text-muted-foreground">
   195→        {artifact.prompt}
   196→      </p>
   197→      <button
   198→        type="button"
   199→        onClick={() => void copyPrompt()}
   200→        className="focus-glow mt-3 flex h-9 items-center gap-2 rounded-full border border-[color-mix(in_srgb,var(--hairline)_70%,transparent)] px-4 text-[13px] text-foreground/85 transition-all duration-300 hover:-translate-y-px hover:border-[var(--hairline-hover)]"

   210→/* ------------------------------------------------------------------ */
   211→/*  Diagram overlay — elegant nodes and threads over the artwork       */
   212→/* ------------------------------------------------------------------ */
   213→
   214→function DiagramOverlay({
   215→  artifact,
   216→  accent,
   217→}: {
   218→  artifact: VisualizationArtifact;
   219→  accent: string;
   220→}) {
   221→  const diagram = artifact.diagram;
   222→  const byId = useMemo(
   223→    () => new Map((diagram?.nodes ?? []).map((n) => [n.id, n])),
   224→    [diagram]
   225→  );
   226→  if (!diagram || diagram.nodes.length === 0) return null;
   227→
   228→  return (
   229→    <div className="pointer-events-none absolute inset-0" data-testid="visual-diagram-overlay">
   230→      {/* the threads */}
   231→      <svg
   232→        className="absolute inset-0 size-full"
   233→        viewBox="0 0 100 100"
   234→        preserveAspectRatio="none"
   235→        aria-hidden="true"
   236→      >
   237→        {diagram.edges.map(([a, b], i) => {
   238→          const na = byId.get(a);
   239→          const nb = byId.get(b);
   240→          if (!na || !nb) return null;
   241→          const mx = (na.x + nb.x) / 2;
   242→          const my = (na.y + nb.y) / 2 - Math.abs(nb.x - na.x) * 0.12 - 3;
   243→          return (
   244→            <path
   245→              key={i}
   246→              d={`M ${na.x} ${na.y} Q ${mx} ${my} ${nb.x} ${nb.y}`}
   247→              fill="none"
   248→              stroke={accent}
   249→              strokeWidth={1}
   250→              strokeOpacity={0.55}
   251→              vectorEffect="non-scaling-stroke"
   252→              strokeDasharray="3 3"
   253→            />
   254→          );
   255→        })}
   256→      </svg>
   257→      {/* the councils themselves */}
   258→      {diagram.nodes.map((n, i) => (
   259→        <motion.span
   260→          key={n.id}
   261→          initial={{ opacity: 0, scale: 0.85 }}
   262→          animate={{ opacity: 1, scale: 1 }}
   263→          transition={{ duration: 0.5, delay: 0.25 + i * 0.09, ease: [0.22, 1, 0.36, 1] }}
   264→          className="absolute -translate-x-1/2 -translate-y-1/2 rounded-full border px-2.5 py-1 text-[10.5px] font-medium leading-tight backdrop-blur-md sm:text-[11.5px]"
   265→          style={{
   266→            left: `${n.x}%`,
   267→            top: `${n.y}%`,
   268→            maxWidth: "30%",
   269→            borderColor: `color-mix(in srgb, ${accent} 55%, transparent)`,
   270→            background: "color-mix(in srgb, #08050f 62%, transparent)",
   271→            color: "rgba(255,255,255,0.94)",
   272→            boxShadow: `0 0 18px -8px color-mix(in srgb, ${accent} 80%, transparent)`,
   273→          }}
   274→        >
   275→          <span className="line-clamp-2">{n.label}</span>
   276→        </motion.span>
   277→      ))}
   278→    </div>
   279→  );
   280→}

   329→/* ------------------------------------------------------------------ */
   330→/*  The presentation deck — each slide with its own artwork            */
   331→/* ------------------------------------------------------------------ */
   332→
   333→function PresentationDeck({
   334→  artifact,
   335→  accent,
   336→  testIdPrefix,
   337→}: {
   338→  artifact: VisualizationArtifact;
   339→  accent: string;
   340→  testIdPrefix: string;
   341→}) {
   342→  const t = useT();
   343→  const [index, setIndex] = useState(0);
   344→  const slides = artifact.slides;
   345→  const total = slides.length;
   346→  const go = (dir: 1 | -1) =>
   347→    setIndex((i) => (i + dir + total) % total);
   348→
   349→  const slide = slides[index];
   350→  if (!slide) return null;
   351→
   352→  return (
   353→    <div data-testid={`${testIdPrefix}-deck`}>
   354→      <div className="relative overflow-hidden rounded-xl border border-[color-mix(in_srgb,var(--hairline)_60%,transparent)]">
   355→        <AnimatePresence mode="wait" initial={false}>
   356→          <motion.div
   357→            key={index}
   358→            initial={{ opacity: 0, scale: 1.015 }}
   359→            animate={{ opacity: 1, scale: 1 }}
   360→            exit={{ opacity: 0 }}
   361→            transition={{ duration: 0.45, ease: [0.22, 1, 0.36, 1] }}
   362→          >
   363→            {slide.imageUrl ? (
   364→              <img
   365→                src={slide.imageUrl}
   366→                alt={slide.title}
   367→                loading="lazy"
   368→                className="aspect-[7/4] w-full object-cover"
   369→              />
   370→            ) : (
   371→              <div
   372→                className="flex aspect-[7/4] w-full items-center justify-center"
   373→                style={{
   374→                  background: `radial-gradient(circle at 30% 20%, color-mix(in srgb, ${accent} 22%, transparent), transparent 60%), linear-gradient(160deg, #0a0716, #120a24)`,
   375→                }}
   376→              >
   377→                <span className="mono-label text-[11px] tracking-[0.3em]" style={{ color: accent }}>
   378→                  {String(index + 1).padStart(2, "0")}
   379→                </span>
   380→              </div>
   381→            )}
   382→          </motion.div>
   383→        </AnimatePresence>
   384→        {total > 1 && (
   385→          <>
   386→            <button
   387→              type="button"
   388→              onClick={() => go(-1)}
   389→              aria-label={t("Previous slide")}
   390→              data-testid={`${testIdPrefix}-deck-prev`}
   391→              className="focus-glow absolute left-2 top-1/2 flex size-9 -translate-y-1/2 items-center justify-center rounded-full border border-white/20 bg-black/35 text-white/90 backdrop-blur-md transition-all duration-300 hover:bg-black/55"
   392→            >
   393→              <ChevronLeft className="size-4" aria-hidden="true" />
   394→            </button>
   395→            <button
   396→              type="button"
   397→              onClick={() => go(1)}
   398→              aria-label={t("Next slide")}
   399→              data-testid={`${testIdPrefix}-deck-next`}
   400→              className="focus-glow absolute right-2 top-1/2 flex size-9 -translate-y-1/2 items-center justify-center rounded-full border border-white/20 bg-black/35 text-white/90 backdrop-blur-md transition-all duration-300 hover:bg-black/55"
   401→            >
   402→              <ChevronRight className="size-4" aria-hidden="true" />
   403→            </button>
   404→          </>
   405→        )}
   406→      </div>
   407→
   408→      <AnimatePresence mode="wait" initial={false}>
   409→        <motion.div
   410→          key={`cap-${index}`}
   411→          initial={{ opacity: 0, y: 6 }}
   412→          animate={{ opacity: 1, y: 0 }}
   413→          exit={{ opacity: 0, y: -6 }}
   414→          transition={{ duration: 0.35, ease: "easeOut" }}
   415→          className="mt-3"
   416→        >
   417→          <p className="text-[15px] font-semibold text-foreground/95">{slide.title}</p>
   418→          {slide.body && (
   419→            <p className="mt-1 text-[14px] leading-relaxed text-muted-foreground">
   420→              {slide.body}
   421→            </p>
   422→          )}
   423→        </motion.div>
   424→      </AnimatePresence>
   425→
   426→      <div className="mt-3 flex items-center justify-between">
   427→        <span className="mono-label text-[10px] tracking-[0.2em] text-muted-foreground/70">
   428→          {t("Slide {n} of {total}", { n: index + 1, total })}
   429→        </span>
   430→        <span className="flex items-center gap-1.5">
   431→          {slides.map((_, i) => (
   432→            <button
   433→              key={i}
   434→              type="button"
   435→              onClick={() => setIndex(i)}
   436→              aria-label={`${t("Slide {n} of {total}", { n: i + 1, total })}`}
   437→              className={cn(
   438→                "size-1.5 rounded-full transition-all duration-300",
   439→                i === index ? "w-4" : "opacity-40"
   440→              )}
   441→              style={{ background: i === index ? accent : "currentColor" }}
   442→            />
   443→          ))}
   444→        </span>
   445→      </div>
   446→    </div>
   447→  );
   448→}

   450→/* ------------------------------------------------------------------ */
   451→/*  The card itself                                                    */

   453→
   454→export function VisualizationCard({
   455→  artifact,
   456→  accent,
   457→  testIdPrefix,
   458→  onRegenerate,
   459→  regenerating = false,
   460→}: {
   461→  artifact: VisualizationArtifact;
   462→  accent: string;
   463→  testIdPrefix: string;
   464→  onRegenerate?: () => void;
   465→  regenerating?: boolean;
   466→}) {
   467→  const t = useT();
   468→  const [loaded, setLoaded] = useState(false);
   469→  const [ratio, setRatio] = useState<string | null>(null);
   470→  const [expanded, setExpanded] = useState(false);
   471→  const modeLabel = MODE_LABELS[artifact.mode];
   472→  const deckRef = useRef<HTMLDivElement>(null);
   473→
   474→  /* body scroll lock while expanded */
   475→  useEffect(() => {
   476→    if (!expanded) return;
   477→    const prev = document.documentElement.style.overflow;
   478→    document.documentElement.style.overflow = "hidden";
   479→    return () => {
   480→      document.documentElement.style.overflow = prev;
   481→    };
   482→  }, [expanded]);
   483→
   484→  /* Escape closes · arrows turn the deck while expanded */
   485→  useEffect(() => {
   486→    if (!expanded) return;
   487→    const onKey = (e: KeyboardEvent) => {
   488→      if (e.key === "Escape") setExpanded(false);
   489→      if (artifact.mode === "presentation" && artifact.slides.length > 1) {
   490→        if (e.key === "ArrowRight")
   491→          deckRef.current
   492→            ?.querySelector<HTMLButtonElement>(`[data-testid="${testIdPrefix}-deck-next"]`)
   493→            ?.click();
   494→        if (e.key === "ArrowLeft")
   495→          deckRef.current
   496→            ?.querySelector<HTMLButtonElement>(`[data-testid="${testIdPrefix}-deck-prev"]`)
   497→            ?.click();
   498→      }
   499→    };
   500→    window.addEventListener("keydown", onKey);
   501→    return () => window.removeEventListener("keydown", onKey);
   502→  }, [expanded, artifact.mode, artifact.slides.length, testIdPrefix]);
   503→
   504→  const showArtwork =
   505→    artifact.imageUrl !== null && artifact.mode !== "presentation";
   506→  const hasPanels = artifact.panels.length > 0;
   507→
   508→  const artwork = (large: boolean) => (
   509→    <div
   510→      className="relative overflow-hidden rounded-xl border border-[color-mix(in_srgb,var(--hairline)_60%,transparent)]"
   511→      style={{
   512→        aspectRatio:
   513→          ratio ?? (artifact.mode === "encyclopedia" || artifact.mode === "science" ? "4 / 3" : "7 / 4"),
   514→      }}
   515→      data-testid={large ? `${testIdPrefix}-artwork-large` : `${testIdPrefix}-artwork`}
   516→    >
   517→      {!loaded && (
   518→        <div className="shimmer-bar absolute inset-0" aria-hidden="true" />
   519→      )}
   520→      {artifact.imageUrl && (
   521→        <img
   522→          src={artifact.imageUrl}
   523→          alt={artifact.title}
   524→          loading="lazy"
   525→          onLoad={(e) => {
   526→            const img = e.currentTarget;
   527→            if (img.naturalWidth > 0 && img.naturalHeight > 0) {
   528→              setRatio(`${img.naturalWidth} / ${img.naturalHeight}`);
   529→            }
   530→            setLoaded(true);
   531→          }}
   532→          className={cn(
   533→            "size-full object-cover transition-opacity duration-700",
   534→            loaded ? "opacity-100" : "opacity-0"
   535→          )}
   536→        />
   537→      )}
   538→      {!large && (
   539→        <>
   540→          <DiagramOverlay artifact={artifact} accent={accent} />
   541→          <AnnotationMarkers artifact={artifact} accent={accent} />
   542→        </>
   543→      )}
   544→    </div>
   545→  );
   546→
   547→  return (
   548→    <motion.div
   549→      initial={{ opacity: 0, y: 14 }}
   550→      animate={{ opacity: 1, y: 0 }}
   551→      transition={{ duration: 0.7, ease: [0.22, 1, 0.36, 1] }}

   762→                            className={cn(

   764→                              i === 0 &&
   765→                                artifact.panels.length > 2 &&

   769→                            <p
