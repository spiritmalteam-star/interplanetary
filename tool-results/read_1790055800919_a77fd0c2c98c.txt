   514→   514→/* --- tiny external store around localStorage (React-recommended) --- */
   515→   515→
   516→   516→let ledgerSnapshot = "[]";
   517→   517→const ledgerListeners = new Set<() => void>();
   518→   518→
   519→   519→function subscribeLedger(cb: () => void) {
   520→   520→  ledgerListeners.add(cb);
   521→   521→  return () => {
   522→   522→    ledgerListeners.delete(cb);
   523→   523→  };
   524→   524→}
   525→   525→
   526→   526→function getLedgerSnapshot() {
   527→   527→  return ledgerSnapshot;
   528→   528→}
   529→   529→
   530→   530→function getServerLedger() {
   531→   531→  return "[]";
   532→   532→}
   533→   533→
   534→   534→function writeLedger(next: LedgerEntry[]) {
   535→   535→  ledgerSnapshot = JSON.stringify(next);
   536→   536→  try {
   537→   537→    localStorage.setItem(LEDGER_KEY, ledgerSnapshot);
   538→   538→  } catch {
   539→   539→    /* storage unavailable */
   540→   540→  }
   541→   541→  ledgerListeners.forEach((l) => l());
   542→   542→}
   543→   543→
   544→   544→function hydrateLedger() {
   545→   545→  try {
   546→   546→    const raw = localStorage.getItem(LEDGER_KEY);
   547→   547→    if (raw && raw !== ledgerSnapshot) {
   548→   548→      ledgerSnapshot = raw;
   549→   549→      ledgerListeners.forEach((l) => l());
   550→   550→    }
   551→   551→  } catch {
   552→   552→    /* storage unavailable */
   553→   553→  }
   554→   554→}
   555→   555→
   556→   556→function parseLedger(json: string): LedgerEntry[] {
   557→   557→  try {
   558→   558→    const parsed = JSON.parse(json) as LedgerEntry[];
   559→   559→    return Array.isArray(parsed) ? parsed : [];
   560→   560→  } catch {
   561→   561→    return [];
   562→   562→  }
   563→   563→}
