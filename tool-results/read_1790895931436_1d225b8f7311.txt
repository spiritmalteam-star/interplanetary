    59→    59→  const [record, setRecord] = useState<AkashicRecord | null>(null);

   103→   103→            language,
   104→   104→            resonance,

   107→   107→                ? {
   108→   108→                    title: record.title,

   167→   167→        await navigator.share({ title: shareTitle, text: recordText });
   168→   168→        return;
   169→   169→      }
   170→   170→      throw new Error("share unavailable");
   171→   171→    } catch (err) {
   172→   172→      if (err instanceof DOMException && err.name === "AbortError") return;

   174→   174→        await navigator.clipboard.writeText(recordText);
   175→   175→        toast.success(t("Sharing is quiet here — the record is copied instead."));

   177→   177→        toast.error(t("The copying did not take — select and copy by hand."));

   195→   195→  }, []);

   202→   202→      stopVoice();
   203→   203→      return;
   204→   204→    }

   215→   215→            /\s*\n\s*/g,

   218→   218→          voice: "regent",
   219→   219→          pace: 0.82,

   223→   223→      const blob = await res.blob();

   234→   234→      await audio.play();
   235→   235→      setVoiceState("playing");
   236→   236→    } catch {
   237→   237→      setVoiceState("idle");
   238→   238→      toast.error(t("The Librarian is quiet. Rest, then listen again."));
   239→   239→    }

   241→   241→

   245→   245→  /* ---------------------------------------------------------------- */

   249→   249→      visualBusyRef.current = true;

   252→   252→      try {
   253→   253→        const res = await fetch("/api/visualize", {
   254→   254→          method: "POST",
   255→   255→          headers: { "Content-Type": "application/json" },
   256→   256→          body: JSON.stringify({
   257→   257→            language,
   258→   258→            message: request,
   259→   259→            history: record

   261→   261→                  {
   262→   262→                    role: "mirror",
   263→   263→                    text: `${record.title}. ${record.era}. ${record.record}`,
   264→   264→                  },
   265→   265→                ]
   266→   266→              : [],
   267→   267→            ...(visualContextRef.current
   268→   268→              ? {
   269→   269→                  contextSubject: visualContextRef.current.subject,

   289→   289→        setVisual((prev) =>
   290→   290→          prev ? { ...prev, state: "error" } : { state: "error", request }

   292→   292→      } finally {
   293→   293→        visualBusyRef.current = false;
   294→   294→      }

   298→   298→

   318→   318→      void seek(val || null, Boolean(record), carried);
   319→   319→    }
   320→   320→  };

   322→   322→  const send = (e: FormEvent) => {
   323→   323→    e.preventDefault();
   324→   324→    sendText(draft);
   325→   325→  };

   327→   327→  /* the voice becomes words on the desk — visible for a breath — then

   329→   329→  const sendTextRef = useRef(sendText);
   330→   330→  sendTextRef.current = sendText;
   331→   331→  const draftRef = useRef(draft);
   332→   332→  draftRef.current = draft;
   333→   333→  const handleVoiceSubmit = (text: string) => {
   334→   334→    const prev = draftRef.current.trim();
   335→   335→    const finalText = prev ? `${prev} ${text}` : text;
   336→   336→    setDraft(finalText);
   337→   337→    window.setTimeout(() => sendTextRef.current(finalText), 650);
   338→   338→  };
   339→   339→
   340→   340→  const listening = voiceState === "playing";
   341→   341→  /* once a record lies open on the desk, everything written becomes a reply */
   342→   342→  const replying = Boolean(record) && !seeking;
   343→   343→
   344→   344→  return (
   345→   345→    <div
   346→   346→      className="relative flex h-full flex-col overflow-hidden bg-background"
   347→   347→      data-testid="akashic-view"
   348→   348→    >
   349→   349→      {/* ---------- threshold: return · the name · a new record ---------- */}
   350→   350→      <header className="relative z-20 flex items-center gap-2 border-b hairline bg-background/80 px-3 py-2.5 backdrop-blur-md sm:px-5">
   351→   351→        <button
   352→   352→          type="button"
   353→   353→          onClick={exitAkashic}
   354→   354→          data-testid="akashic-return"
   355→   355→          aria-label={t("Return from the Library")}
   356→   356→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border hairline bg-card/60 text-[12px] text-foreground/85 transition-all duration-300 hover:-translate-x-px hover:border-[var(--hairline-hover)] hover:text-foreground sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   357→   357→        >
   358→   358→          <ArrowLeft
   359→   359→            className="size-3.5 transition-transform duration-300 group-hover:-translate-x-0.5"
   360→   360→            aria-hidden="true"
   361→   361→          />
   362→   362→          <span className="hidden sm:inline">{t("Return")}</span>
   363→   363→        </button>
   364→   364→
   365→   365→        <div className="flex min-w-0 flex-1 flex-col items-center justify-center gap-0.5">
   366→   366→          <span className="flex items-center gap-2">
   367→   367→            <span className="akashic-halo flex size-6 shrink-0 items-center justify-center rounded-full border hairline bg-card/60">
   368→   368→              <Scroll className="size-3 text-muted-foreground" aria-hidden="true" />
   369→   369→            </span>
   370→   370→            <h1 className="ink-title min-w-0 truncate text-[15px] font-semibold tracking-[0.14em] sm:text-[17px]">
   371→   371→              {t("The Akashic Library")}
   372→   372→            </h1>
   373→   373→          </span>
   374→   374→          <span className="mono-label hidden text-[9px] uppercase tracking-[0.26em] text-muted-foreground/70 sm:block">
   375→   375→            {t("resonance · records · the ancient one")}
   376→   376→          </span>
   377→   377→        </div>
   378→   378→
   379→   379→        <button
   380→   380→          type="button"
   381→   381→          onClick={() => void seek(null, false)}
   382→   382→          disabled={seeking}
   383→   383→          data-testid="akashic-new"
   384→   384→          aria-label={t("New record")}
   385→   385→          className="focus-glow group flex size-10 shrink-0 items-center justify-center gap-2 rounded-full border hairline bg-card/60 text-[12px] text-foreground/85 transition-all duration-300 hover:-translate-y-px hover:border-[var(--hairline-hover)] hover:text-foreground disabled:cursor-not-allowed disabled:opacity-40 sm:size-auto sm:justify-start sm:px-3 sm:py-2"
   386→   386→        >
   387→   387→          <RotateCcw
   388→   388→            className="size-3.5 transition-transform duration-500 group-hover:-rotate-180"
   389→   389→            aria-hidden="true"
   390→   390→          />
   391→   391→          <span className="hidden sm:inline">{t("New record")}</span>
   392→   392→        </button>
   393→   393→      </header>
   394→   394→
   395→   395→      {/* ---------- the reading room: one sheet, one record ---------- */}
   396→   396→      <div className="nice-scroll relative z-10 flex-1 overflow-y-auto">
   397→   397→        <div className="mx-auto flex w-full max-w-[780px] flex-col px-3 pb-10 pt-5 sm:px-6 sm:pt-7">
   398→   398→          <AnimatePresence mode="wait">
   399→   399→            {/* the unopened shelf — an invitation, before the first record */}
   400→   400→            {!record && !seeking && !error && (
   401→   401→              <motion.div
   402→   402→                key="invitation"
   403→   403→                initial={{ opacity: 0, y: 14 }}
   404→   404→                animate={{ opacity: 1, y: 0 }}

   415→   415→                </span>

   418→   418→                    "Set down a resonance — a name, a question, a feeling — and the Librarian will draw out the record it belongs to. Or leave the desk silent, and the Library will choose for you."

   431→   431→            )}

   433→   433→            {/* the ink forming — the Librarian walks the shelves */}

   453→   453→                  {[86, 70, 92, 60, 78].map((w, i) => (

   459→   459→                  ))}
   460→   460→                </div>

   463→   463→                    ? t("the Librarian weighs your reply...")

   468→   468→
   469→   469→            {/* when the shelves keep their silence */}
   470→   470→            {error && !seeking && (
   471→   471→              <motion.div
